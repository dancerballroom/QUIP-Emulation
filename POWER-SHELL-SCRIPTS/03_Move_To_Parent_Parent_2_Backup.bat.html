03_Move_To_Parent_Parent_2_Backup.bat


REM 03_Move_To_Parent_Parent_2_Backup.bat
@Echo Off
Echo File Name including Path:
Echo %1
Echo ---
Echo File Name without Extension
SET FilenameNoExt=%~n1
Echo %FilenameNoExt%
Echo ---
Echo Filenames of Child
SET ChildName_MP4="%~n1.mp4"
SET ChildName_SRT="%~n1.srt"
Echo Child MP4 is %ChildName_MP4%
Echo Child SRT is %ChildName_SRT%
Echo ---
REM # REM Exit 0
REM # REM DEBUG
cd
dir
REM # REM CD Parent
pushd ..
REM # REM Create ..\Backup or NOT if Exists
if exist Backup\ (
  echo Yes 
) else (
  mkdir Backup
)
REM # REM Create ..\Backup or NOT if Exists
if exist DONE\ (
  echo Yes 
) else (
  mkdir DONE
)
REM # DEBUG
cd
dir
Echo Next Copy All to Parent Backup
REM # REM Parent to Backup
Echo Move Parent Movie-1 to Backup
Move "%FilenameNoExt%".* Backup
REM # REM CD Child
popd
REM # DEBUG
Echo CUR-DIR
REM # DEBUG
cd
dir 
REM # Move Movie.mp4 and Movie.srt to Parent
Echo Next CUR-DIR to Parent
Move "%FilenameNoExt%".* ..\DONE
Echo DONE
Echo ---
REM # End of Program
echo End of Program
Echo ---
REM 03_Move_To_Parent_Parent_2_Backup.bat
        REM EOL
