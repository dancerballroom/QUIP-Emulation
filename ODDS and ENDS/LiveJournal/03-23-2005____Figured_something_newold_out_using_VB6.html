Here's my ActiveX DLL stuff:
Option Explicit

Public Function Listing_W3SVC() As String
    Listing_W3SVC = Dir("C:\WINDOWS\system32\Logfiles\W3SVC1\*.log")
End Function

Public Function Listing_FTPSVC() As String
    Listing_FTPSVC = Dir("C:\WINDOWS\system32\Logfiles\MSFTPSVC1\*.log")
End Function

Public Function Read_W3SVC_LogFile() As String
    Dim f As String, g As String
    Dim out1 As String
    out1 = ""
    f = "C:\WINDOWS\system32\Logfiles\W3SVC1\ex" & Format(Now, "yymmdd") & ".log"
    
    Open f For Input As #1
    Do While Not EOF(1)   ' Loop until end of file.
        Input #1, g
        If InStr(g, "GetLog.asp") < 1 Then
            out1 = out1 & g & "<BR> <BR> "
        End If
    Loop
    Close #1
    
    Read_W3SVC_LogFile = out1
End Function

Public Function Read_FTPSVC_LogFile() As String
    Dim f As String, g As String
    Dim out2 As String
    out2 = ""
    f = "C:\WINDOWS\system32\Logfiles\MSFTPSVC1\ex" & Format(Now, "yymmdd") & ".log"
    
    Open f For Input As #2
    Do While Not EOF(2)   ' Loop until end of file.
        Input #2, g
        out2 = out2 & g & " <BR>  <BR> "
    Loop
    Close #2
    
    Read_FTPSVC_LogFile = out2
End Function

TECH NOTE:  If you compile an ActiveX DLL used by the web more than once, you will 
have to kill the dllhost.exe process using CNTRL+ALT+DEL about 3 to 4 times
before you get passed the Compile Error "Permission Denied: object in use", or
restart IIS, or un/register your dll manually using REGSVR32.exe in Windows/Systems32.


Here's my Asp under IIS file

HTML
BODY
LANGUAGE="VBSCRIPT"

    ' Note that the VB ActiveX Project above was called WEBLOG
    ' The Class I created in the ActiveX Project was called LOG_CLASS
    Set objDLL=Server.CreateObject("WEBLOG.LOG_CLASS")
    Response.Write " B W3SVC:     "

    ' Now I use my objDLL to reference my Functions
    Response.write objDLL.Listing_W3SVC
    Response.Write "</B> <BR>  <BR>  <PRE>"

    Response.write objDLL.Read_W3SVC_LogFile

    Response.Write "</PRE>  <HR>"
    Response.Write " <B> MSFTPSVC:  "

    Response.write objDLL.Listing_FTPSVC

    Response.Write "</B>  <BR>  <BR>  <PRE> "

    Response.write objDLL.Read_FTPSVC_LogFile

    Response.Write " </PRE>  <BR> "
    
    Set objDLL = Nothing

BODY
HTML

There is a trick to setting up Logging, of course. You have to right
click on the W3SVC (Default Web Site) and MSFTPSVC (Default FTP Site),
Then click the properties box next to "W3C Extended Log Format", then
click the "Extended Properties" tab, then choose what you want reported.
Apply, OK, Enable Logging, Apply, OK.

The logfiles are located in directories under:
C:\WINDOWS\system32\Logfiles\

You can delete these, of course.
