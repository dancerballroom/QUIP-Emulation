REM AAA-CROP-Start-End.bat
@Echo Off
setlocal enableDelayedExpansion
set /p timeStart="Start -> (00:00:00) :> "
set /p timeEnd="End   -> (00:00:00) :> "
REM echo "%~n1.mp4 -> %~n1_.mp4"
Set "ext=mp4"
Set "opts=-v quiet"
Set "opts=%opts% -print_format "compact=print_section=0:nokey=1:escape=csv""
Set "opts=%opts% -show_entries "format=duration""
If Exist *.%ext% (If Not Exist "Trimmed\" MD Trimmed)
REM For %%a In (*.%ext%) Do Call :Sub "%%~a"
For %%a In ("%~n1.mp4") Do Call :Sub "%%~a"
Exit /B

REM Continued
:Sub
For /f "Tokens=1* Delims=." %%a In ('FFProbe %opts% %1') Do (Set /A "ws=!timeEnd!" & Set "ps=%%b")
Set/A hh=ws/(60*60), lo=ws%%(60*60), mm=lo/60, ss=lo%%60
echo passed SET
If %hh% Lss 10 Set hh=0%hh%
If %mm% Lss 10 Set mm=0%mm%
If %ss% Lss 10 Set ss=0%ss%

        REM FFMpeg Commands
FFMpeg -itsoffset -!timeEnd! -i %1 "Trimmed\%~1.eng.srt"
FFMpeg -i %1 -ss !timeStart! -to !timeEnd! -c:v copy -c:a copy "Trimmed\%~1"
