Extract_Chapters.bat

@ECHO OFF
SETLOCAL ENABLEDELAYEDEXPANSION

IF "%~1" == "" (
     CALL :HELP
        EXIT
)
Set FILENAME="%~d1%~p1%~n1.mp4"
Set CHAPTERS="%~d1%~p1%~n1-Chapters.txt"

REM             TRUE = SHOW DEBUGGING 
set /A TRUE=1
REM             FALSE = HIDE DEBUGGING
set /A FALSE=0
REM SHOW/HIDE DEBUGGING
set /A DEBUG=!FALSE!

CALL :RUN_FFPROBE
exit

:RUN_FFPROBE

for /f "delims=" %%i in ('ffprobe -v quiet -print_format ini -show_chapters !FILENAME! ^| findstr /C:"title" /C:"start_time"') do (
    set my_variable=%%i
    REM echo !my_variable!
    REM echo.
    for /F "tokens=1,2 delims==^=" %%a in ("!my_variable!") do (
        REM echo Key: %%a
        REM echo Value: %%b
        if "%%a" EQU "title" (
            REM Echo %%b
            SET CHAPTER[!chapter_count!][2]= %%b
        )
        if "%%a" EQU "start_time" (
            set /a chapter_count+=1
            SET SPC=
            if !chapter_count! LEQ 9 (
                SET SPC=0
            )
            set /a Time_S=%%b 2> nul
            set /a Time_S=!Time_S! * 1000
            
            CALL :MilliSec2TimeSig !Time_S! Timex TimexC
            REM powershell -command "Write-Host -NoNewLine '!SPC!!chapter_count!. !Timex! '"
            REM Echo !SPC!!chapter_count!. !Timex!
            SET CHAPTER[!chapter_count!][1]=!SPC!!chapter_count!. !Timex!
        )
    )
)
type NUL > !CHAPTERS!
FOR /L %%c IN (1,1,!chapter_count!) DO (
    Echo !CHAPTER[%%c][1]! !CHAPTER[%%c][2]! >> !CHAPTERS!
    Echo !CHAPTER[%%c][1]! !CHAPTER[%%c][2]!
)
REM Echo !CHAPTERS!
type !CHAPTERS! | clip
pause
Exit /b

:HELP
    ECHO ==============================================================  
    ECHO ==== MP4 File not Specified
    ECHO ==== Extract_Chapters.bat "Filename.mp4"
    ECHO ====   Drag/Drop "Filename.mp4" on Extract_Chapters.bat
    ECHO ==============================================================  
pause
Exit /b

REM Function MilliSec2TimeSig
REM INPUT:  %1 (Milli-Seconds)
REM          CALL :MilliSec2TimeSig !Milli-Seconds! Timex TimexC
REM OUTPUT: Timex (Time-Signature: 12:34:56.789)
REM OUTPUT: TimexC (Time-Signature: 12:34:56,789)
:MilliSec2TimeSig
    Set /A MS_TOT=%1
    REM Echo MS_TOT: !MS_TOT!
    Set MS_EXT=!MS_TOT:~-3,3!
    if !DEBUG! EQU !TRUE! Echo Mili-Seconds Extracted: !MS_EXT!
    Set /A Rem_Minus_MS=!MS_TOT!-MS_EXT
    if !DEBUG! EQU !TRUE! Echo Total Minus MS: !Rem_Minus_MS!
    Set /A HH_EXT=(!Rem_Minus_MS!/(60*60*1000))
    if !DEBUG! EQU !TRUE! Echo Hours Extracted: !HH_EXT!
    Set /A Rem_Minus_HH=(!Rem_Minus_MS!%%(60*60*1000))
    if !DEBUG! EQU !TRUE! Echo Total Minus HH: !Rem_Minus_HH!
    Set /A MM_EXT=(!Rem_Minus_HH!/(60*1000))
    if !DEBUG! EQU !TRUE! Echo Minutes Extracted: !MM_EXT!
    Set /A Rem_Minus_MM=(!Rem_Minus_HH!%%(60*1000))
    if !DEBUG! EQU !TRUE! Echo Total Seconds SS: !Rem_Minus_MM!
    Set /A SS_EXT=(!Rem_Minus_MM!/1000)
    if !DEBUG! EQU !TRUE! Echo Seconds Extracted: !SS_EXT!

    if !HH_EXT! lss 10 set HH_EXT=0!HH_EXT!
    if !MM_EXT! lss 10 set MM_EXT=0!MM_EXT!
    if !SS_EXT! lss 10 set SS_EXT=0!SS_EXT!
    REM if !MS_EXT! lss 10 set MS_EXT=00!MS_EXT!
    REM MOD ERROR Results in .0 instead of .000
    REM ALSO Above 00!MS_EXT! Results in .000000 instead of .000

    if !DEBUG! EQU !TRUE! Echo Time Extracted: !HH_EXT!:!MM_EXT!:!SS_EXT!,!MS_EXT!
    Set Timex=!HH_EXT!:!MM_EXT!:!SS_EXT!
    Set TimexC=!HH_EXT!:!MM_EXT!:!SS_EXT!
Exit /b
