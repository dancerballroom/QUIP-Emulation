<script runat="server">
    protected void Page_Load(object sender, EventArgs e)
    {
        // Ensure state is synced from the URL
        if (!string.IsNullOrEmpty(Request.QueryString["path"]))
        {
            AppState.CurrentFolder = Request.QueryString["path"];
        }
    }


    protected void btnCreate_Click(object sender, EventArgs e)
    {
        string folderName = txtFolderName.Text.Trim();
        
        if (!string.IsNullOrEmpty(folderName)) 
        {
            string root = Server.MapPath("~/uploads/notes/");
            // Use AppState.CurrentFolder
            string newFolderPath = Path.Combine(root, AppState.CurrentFolder.Replace("/", "\\").TrimStart('\\'), folderName);
            
            if (!Directory.Exists(newFolderPath)) 
            {
                Directory.CreateDirectory(newFolderPath);
            }
            
            // Redirect back to the browser, maintaining the current folder context via AppState
            Response.Redirect("FileBrowser.aspx?path=" + HttpUtility.UrlEncode(AppState.CurrentFolder));
        }
    }
</script>
