<script type="text/javascript">
    // History State Arrays for Undo / Redo
    var undoStack = [];
    var redoStack = [];


    // Strip editor-only code box action buttons from the loaded preview content
    function stripCodeBoxActionIcons() {
        var container = document.getElementById('preview-container');
        if (!container) return;


        container.querySelectorAll('.code-box-action-bar').forEach(function(actionBar) {
            actionBar.remove();
        });


        container.querySelectorAll('.code-box-read-btn, .code-box-highlight-btn, .code-box-copy-btn, .code-box-save-btn').forEach(function(actionButton) {
            actionButton.remove();
        });


        container.querySelectorAll('pre.code-box-content').forEach(function(codeBox) {
            if (codeBox.style.paddingRight === '280px') {
                codeBox.style.paddingRight = '';
            }
        });
    }


    // Capture initial document state when the DOM is fully loaded
    window.addEventListener('DOMContentLoaded', function() {
        stripCodeBoxActionIcons();
        saveState();
    });


    // Saves current innerHTML state of preview-container into history
    function saveState() {
        var container = document.getElementById('preview-container');
        if (!container) return;


        var currentState = container.innerHTML;


        // Prevent saving duplicate consecutive states
        if (undoStack.length === 0 || undoStack[undoStack.length - 1] !== currentState) {
            undoStack.push(currentState);
            // Clear redo stack whenever new modifications occur
            redoStack = [];
        }
    }


    // Reverts preview content to previous history snapshot
    function undoAction() {
        var container = document.getElementById('preview-container');
        if (!container || undoStack.length <= 1) return; // Need at least initial state + 1 state to undo


        // Move current state to redo stack
        var currentState = undoStack.pop();
        redoStack.push(currentState);


        // Apply previous state
        container.innerHTML = undoStack[undoStack.length - 1];
    }


    // Reapplies previously undone snapshot
    function redoAction() {
        var container = document.getElementById('preview-container');
        if (!container || redoStack.length === 0) return;


        // Pop state from redo stack and push onto undo stack
        var nextState = redoStack.pop();
        undoStack.push(nextState);


        // Apply state
        container.innerHTML = nextState;
    }


    // Helper function: Extracts the ID from URL query string to use as default filename base
    function getDefaultFileName() {
        var urlParams = new URLSearchParams(window.location.search);
        var fileId = urlParams.get('ID') || urlParams.get('id') || 'Document.html';
        
        // Strip .html or .htm extension if present
        if (fileId.toLowerCase().endsWith('.html')) {
            fileId = fileId.substring(0, fileId.length - 5);
        } else if (fileId.toLowerCase().endsWith('.htm')) {
            fileId = fileId.substring(0, fileId.length - 4);
        }
        return fileId;
    }


    // Change display margins on screen
    function adjustMargins(val) {
        document.getElementById('preview-container').style.padding = val;
    }


    // Helper: Shared Native "Save As" / Prompt execution for Blobs/Files
    async function saveFileWithDialog(blobData, defaultFullName, extension, description, mimeType) {
        // Method 1: Modern Native OS "Save As" Dialog (File System Access API)
        if ('showSaveFilePicker' in window) {
            try {
                var acceptObj = {};
                acceptObj[mimeType] = ['.' + extension];


                const handle = await window.showSaveFilePicker({
                    suggestedName: defaultFullName,
                    types: [{
                        description: description,
                        accept: acceptObj
                    }]
                });
                const writable = await handle.createWritable();
                await writable.write(blobData);
                await writable.close();
                return; // Completed native save
            } catch (err) {
                if (err.name === 'AbortError') {
                    return; // User clicked Cancel
                }
                // Fall through to Method 2 if blocked by browser policy
            }
        }


        // Method 2: Custom JavaScript Prompt Fallback
        var userFileName = prompt("Save As - Enter filename:", defaultFullName);
        
        if (userFileName === null) {
            return; // User canceled
        }


        userFileName = userFileName.trim();
        if (userFileName === '') {
            userFileName = defaultFullName;
        }


        if (!userFileName.toLowerCase().endsWith('.' + extension)) {
            userFileName += '.' + extension;
        }


        // Trigger file save using FileSaver.js
        saveAs(blobData, userFileName);
    }


    // Export PDF with "Save As" Dialog
    async function exportPDF() {
        var defaultFullName = getDefaultFileName() + '.pdf';
        var element = document.getElementById('preview-container');
        
        var opt = {
            margin: 0.5,
            image: { type: 'jpeg', quality: 0.98 },
            html2canvas: { scale: 2 },
            jsPDF: { unit: 'in', format: 'letter', orientation: 'portrait' }
        };


        // Generate PDF as a Blob without auto-downloading
        html2pdf().set(opt).from(element).output('blob').then(async function(pdfBlob) {
            await saveFileWithDialog(
                pdfBlob, 
                defaultFullName, 
                'pdf', 
                'PDF Document (*.pdf)', 
                'application/pdf'
            );
        });
    }


    // Export DOCX with "Save As" Dialog
    async function exportDOCX() {
        var defaultFullName = getDefaultFileName() + '.docx';
        var content = document.getElementById('preview-container').innerHTML;
        var htmlString = '<!DOCTYPE html><html><head><meta charset="utf-8"></head><body>' + content + '</body></html>';
        
        // Convert HTML to DOCX Blob
        var docxBlob = htmlDocx.asBlob(htmlString);


        await saveFileWithDialog(
            docxBlob, 
            defaultFullName, 
            'docx', 
            'Word Document (*.docx)', 
            'application/vnd.openxmlformats-officedocument.wordprocessingml.document'
        );
    }


    // Export TXT with "Save As" Dialog
    async function exportTXT() {
        var textContent = document.getElementById('preview-container').innerText;
        var defaultFullName = getDefaultFileName() + '.txt';
        var txtBlob = new Blob([textContent], { type: "text/plain;charset=utf-8" });


        await saveFileWithDialog(
            txtBlob, 
            defaultFullName, 
            'txt', 
            'Text Files (*.txt)', 
            'text/plain'
        );
    }


    // Navigates back to FilingCabinet.aspx in the current window using current URL parameters
    function goBackToFilingCabinet() {
        if (!confirm("Do you want to leave the Print Preview Dialogue?")) {
            return;
        }


        var urlParams = new URLSearchParams(window.location.search);
        
        // Read parameters passed to PrintPreview.aspx
        var fileId = urlParams.get('ID') || urlParams.get('id') || '';
        var folderName = urlParams.get('Folder') || urlParams.get('folder') || '';


        // Strip .html or .htm extension if present to restore original id parameter
        if (fileId.toLowerCase().endsWith('.html')) {
            fileId = fileId.substring(0, fileId.length - 5);
        } else if (fileId.toLowerCase().endsWith('.htm')) {
            fileId = fileId.substring(0, fileId.length - 4);
        }


        // Build return URL
        var targetUrl = 'FilingCabinet.aspx';
        var queryParts = [];


        if (fileId) {
            queryParts.push('id=' + encodeURIComponent(fileId));
        }
        if (folderName) {
            queryParts.push('Folder=' + encodeURIComponent(folderName));
        }


        if (queryParts.length > 0) {
            targetUrl += '?' + queryParts.join('&');
        }


        // Navigate in current window/page
        window.location.href = targetUrl;
    }


    // Adjusts font size of highlighted text or the entire document when nothing is selected
    function changeFontSize(deltaPixels) {
        var container = document.getElementById('preview-container');
        var selection = window.getSelection();


        if (!container) return;


        var range;
        var shouldSelectEntireDocument = false;


        if (selection.rangeCount > 0 && !selection.isCollapsed) {
            range = selection.getRangeAt(0);


            // If the current selection is not inside the preview container, fall back to the whole document
            if (!container.contains(range.commonAncestorContainer)) {
                shouldSelectEntireDocument = true;
            }
        } else {
            shouldSelectEntireDocument = true;
        }


        if (shouldSelectEntireDocument) {
            selection.removeAllRanges();
            range = document.createRange();
            range.selectNodeContents(container);
            selection.addRange(range);
        }


        range = selection.getRangeAt(0);
        if (!container.contains(range.commonAncestorContainer)) {
            return;
        }


        var selectedContent = range.extractContents();
        var span = document.createElement('span');


        var parentElem = range.commonAncestorContainer.nodeType === 1
            ? range.commonAncestorContainer
            : range.commonAncestorContainer.parentNode;


        var currentSize = parseFloat(window.getComputedStyle(parentElem).fontSize) || 16;
        var newSize = Math.max(8, currentSize + deltaPixels);


        span.style.fontSize = newSize + 'px';
        span.appendChild(selectedContent);
        range.insertNode(span);


        selection.removeAllRanges();
        var newRange = document.createRange();
        newRange.selectNodeContents(span);
        selection.addRange(newRange);


        saveState();
    }
