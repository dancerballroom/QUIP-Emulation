    private static string SanitizeFileNameCore(string value)
    {
        if (string.IsNullOrWhiteSpace(value))
        {
            return "";
        }


        string cleaned = value.Trim().Replace(" ", "_");
        char[] invalidChars = Path.GetInvalidFileNameChars();
        foreach (char c in invalidChars)
        {
            cleaned = cleaned.Replace(c, '_');
        }


        return cleaned.Trim('_', '.');
    }


    private static string EnsureUniqueFileName(string folderPath, string preferredFileName)
    {
        string baseName = Path.GetFileNameWithoutExtension(preferredFileName);
        string extension = Path.GetExtension(preferredFileName);
        string candidate = preferredFileName;
        int index = 2;


        while (File.Exists(Path.Combine(folderPath, candidate)))
        {
            candidate = baseName + "_" + index + extension;
            index++;
        }


        return candidate;
    }


    private static string BuildFinalHtmlFromEditorInput(string noteHtml, string pageTitle)
    {
        noteHtml = (noteHtml ?? "").Replace("\r\n", "\n").Replace("\r", "\n");
        noteHtml = noteHtml.Trim('\n', '\r', ' ');


        string[] lines = noteHtml.Split(new[] { '\n' }, StringSplitOptions.None);
        bool insidePre = false;


        for (int k = 0; k < lines.Length; k++)
        {
            string trimmedLine = lines[k].Trim();


            if (trimmedLine.IndexOf("<pre", StringComparison.OrdinalIgnoreCase) >= 0)
            {
                lines[k] = "        " + lines[k].TrimStart(' ', '\t');
                insidePre = true;
                continue;
            }


            if (trimmedLine.IndexOf("</pre>", StringComparison.OrdinalIgnoreCase) >= 0)
            {
                lines[k] = "        " + lines[k].TrimStart(' ', '\t');
                insidePre = false;
                continue;
            }


            if (insidePre)
            {
                continue;
            }


            if (string.IsNullOrWhiteSpace(lines[k]))
            {
                lines[k] = "";
            }
            else
            {
                lines[k] = "        " + lines[k].TrimStart(' ', '\t');
            }
        }


        string nestedBodyHtml = string.Join("\r\n", lines);


        return
            "<!DOCTYPE html>\r\n" +
            "<html>\r\n" +
            "    <head>\r\n" +
            "        <meta charset='utf-8' />\r\n" +
            "        <title>" + pageTitle + "</title>\r\n" +
            "    </head>\r\n" +
            "    <body>\r\n" +
            nestedBodyHtml + "\r\n" +
            "    </body>\r\n" +
            "</html>";
    }


    private static string BuildFinalHtmlFromImportedInput(string rawImportedText, string pageTitle)
    {
        string source = rawImportedText ?? "";


        System.Text.RegularExpressions.Match bodyMatch =
            System.Text.RegularExpressions.Regex.Match(
                source,
                @"<body[^>]*>(?<body>[\s\S]*?)</body>",
                System.Text.RegularExpressions.RegexOptions.IgnoreCase
            );


        string bodyHtml;
        if (bodyMatch.Success)
        {
            bodyHtml = bodyMatch.Groups["body"].Value;
        }
        else
        {
            string normalized = source.Replace("\r\n", "\n").Replace("\r", "\n");
            bodyHtml = HttpUtility.HtmlEncode(normalized).Replace("\n", "<br />\r\n");
        }


        return
            "<!DOCTYPE html>\r\n" +
            "<html>\r\n" +
            "    <head>\r\n" +
            "        <meta charset='utf-8' />\r\n" +
            "        <title>" + pageTitle + "</title>\r\n" +
            "    </head>\r\n" +
            "    <body>\r\n" +
            bodyHtml + "\r\n" +
            "    </body>\r\n" +
            "</html>";
    }


    private static string DecodeBase64Utf8(string value)
    {
        if (string.IsNullOrEmpty(value))
        {
            return "";
        }


        try
        {
            byte[] bytes = Convert.FromBase64String(value);
            return Encoding.UTF8.GetString(bytes);
        }
        catch
        {
            return "";
        }
    }


    protected void btnCreate_Click(object sender, EventArgs e)
    {
        string currentFolder = AppState.CurrentFolder ?? "";
        string importQueueRaw = hfImportQueue.Value;


        if (!string.IsNullOrWhiteSpace(importQueueRaw))
        {
            try
            {
                JavaScriptSerializer serializer = new JavaScriptSerializer();
                List<ImportedFileItem> importQueue = serializer.Deserialize<List<ImportedFileItem>>(importQueueRaw);


                if (importQueue != null && importQueue.Count > 0)
                {
                    string root = Server.MapPath("~/uploads/notes/");
                    string folderPath = Path.Combine(root, currentFolder.Replace("/", "\\").TrimStart('\\'));
                    Directory.CreateDirectory(folderPath);


                    int createdCount = 0;


                    foreach (ImportedFileItem importedFile in importQueue)
                    {
                        if (importedFile == null || string.IsNullOrWhiteSpace(importedFile.name))
                        {
                            continue;
                        }


                        string sourceName = Path.GetFileNameWithoutExtension(importedFile.name.Trim());
                        string safeName = SanitizeFileNameCore(sourceName);


                        if (string.IsNullOrWhiteSpace(safeName))
                        {
                            continue;
                        }


                        string preferredFileName = safeName + ".html";
                        string finalFileName = EnsureUniqueFileName(folderPath, preferredFileName);
                        string finalTitle = Path.GetFileNameWithoutExtension(finalFileName);
                        string sourceContent = DecodeBase64Utf8(importedFile.contentBase64);
                        string finalHtml = BuildFinalHtmlFromImportedInput(sourceContent, finalTitle);


                        File.WriteAllText(Path.Combine(folderPath, finalFileName), finalHtml, Encoding.UTF8);
                        createdCount++;
                    }


                    if (createdCount > 0)
                    {
                        Response.Redirect("FileBrowser.aspx?path=" + HttpUtility.UrlEncode(currentFolder));
                        return;
                    }
                }
            }
            catch
            {
                // Fall through to classic single-note create if queue payload is malformed.
            }
        }


        string fileName = txtFileName.Text.Trim();
        
        if (!string.IsNullOrEmpty(fileName)) 
        {
            // SANITIZATION HATCH: Instantly replace any space with an underscore
            string sanitizedFileName = fileName.Replace(" ", "_");


            // Update the global state via AppState using the sanitized name
            AppState.CurrentNote = sanitizedFileName;
            
            // Redirect to the editor, passing the clean, underscore-filled name via QueryString
            Response.Redirect("FilingCabinet.aspx?id=" + HttpUtility.UrlEncode(sanitizedFileName) + 
                              "&folder=" + HttpUtility.UrlEncode(AppState.CurrentFolder));
        }
    }

<script>
    (function () {
        var queue = [];
        var pendingReads = 0;
        var hiddenQueueField = document.getElementById('<%= hfImportQueue.ClientID %>');
        var createBtn = document.getElementById('<%= btnCreate.ClientID %>');
        var dropZone = document.getElementById('importDropZone');
        var listEl = document.getElementById('importQueueList');
        var emptyEl = document.getElementById('importQueueEmpty');
        var picker = document.getElementById('filePicker');
        var chooseBtn = document.getElementById('btnChooseFiles');
        var clearBtn = document.getElementById('btnClearQueue');


        function toBase64Utf8(input) {
            try {
                return btoa(unescape(encodeURIComponent(input)));
            } catch (err) {
                return '';
            }
        }


        function syncQueueToForm() {
            hiddenQueueField.value = JSON.stringify(queue);
            emptyEl.style.display = queue.length ? 'none' : 'block';


            while (listEl.firstChild) {
                listEl.removeChild(listEl.firstChild);
            }


            for (var i = 0; i < queue.length; i++) {
                var li = document.createElement('li');
                li.textContent = (i + 1) + '. ' + queue[i].name;
                listEl.appendChild(li);
            }


            createBtn.disabled = pendingReads > 0;
            createBtn.value = pendingReads > 0 ? 'Reading Files...' : 'Create';
        }


        function queueFiles(fileList) {
            if (!fileList || !fileList.length) {
                return;
            }


            for (var i = 0; i < fileList.length; i++) {
                (function (file) {
                    pendingReads++;
                    syncQueueToForm();


                    var reader = new FileReader();
                    reader.onload = function (evt) {
                        var rawText = evt && evt.target ? String(evt.target.result || '') : '';
                        queue.push({
                            name: file.name,
                            contentBase64: toBase64Utf8(rawText)
                        });
                    };


                    reader.onerror = function () {
                        // Ignore unreadable files and continue with others.
                    };


                    reader.onloadend = function () {
                        pendingReads = Math.max(0, pendingReads - 1);
                        syncQueueToForm();
                    };


                    reader.readAsText(file);
                })(fileList[i]);
            }
        }


        function suppressDefaults(evt) {
            evt.preventDefault();
            evt.stopPropagation();
        }


        ['dragenter', 'dragover', 'dragleave', 'drop'].forEach(function (eventName) {
            dropZone.addEventListener(eventName, suppressDefaults, false);
        });


        ['dragenter', 'dragover'].forEach(function (eventName) {
            dropZone.addEventListener(eventName, function () {
                dropZone.classList.add('drag-over');
            }, false);
        });


        ['dragleave', 'drop'].forEach(function (eventName) {
            dropZone.addEventListener(eventName, function () {
                dropZone.classList.remove('drag-over');
            }, false);
        });


        dropZone.addEventListener('drop', function (evt) {
            var files = evt.dataTransfer ? evt.dataTransfer.files : null;
            queueFiles(files);
        }, false);


        chooseBtn.addEventListener('click', function () {
            picker.click();
        });


        picker.addEventListener('change', function () {
            queueFiles(picker.files);
            picker.value = '';
        });


        clearBtn.addEventListener('click', function () {
            queue = [];
            syncQueueToForm();
        });


        syncQueueToForm();
    })();
