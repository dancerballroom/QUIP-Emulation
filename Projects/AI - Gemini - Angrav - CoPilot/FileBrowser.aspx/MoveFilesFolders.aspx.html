<%@ Page Language="C#" AutoEventWireup="true" ResponseEncoding="utf-8" %>
<%@ Import Namespace="System.IO" %>
<%@ Import Namespace="System.Collections.Generic" %>


<script runat="server">
    // Absolute path line replaced with Relative MapPath
    public string RootNotesPath
    {
        get
        {
            return Server.MapPath("~/uploads/notes/");
        }
    }


    public string CurrentDestFolder = "";
    public string ItemToMoveName = "";
    public bool IsMovingFile = false;
    
    // Maintain state across links
    public string QueryId = "";
    public string QueryFolder = "";
    public string QueryMoveFolder = "";


    // =========================================================================
    // UNIFIED PIN STORAGE HELPERS (Points directly to /uploads/pinned_notes.txt)
    // =========================================================================
    private List<string> GetPinnedPaths()
    {
        string pinFilePath = Server.MapPath("~/uploads/pinned_notes.txt");
        if (File.Exists(pinFilePath))
        {
            return new List<string>(File.ReadAllLines(pinFilePath));
        }
        return new List<string>();
    }


    private void SavePinnedPaths(List<string> paths)
    {
        string pinFilePath = Server.MapPath("~/uploads/pinned_notes.txt");
        File.WriteAllLines(pinFilePath, paths);
    }
    // =========================================================================


    protected void Page_Load(object sender, EventArgs e)
    {
        // 1. Capture what item is being targeted for a move
        QueryId = Request.QueryString["id"] ?? "";
        QueryFolder = Request.QueryString["folder"] ?? "";
        QueryMoveFolder = Request.QueryString["moveFolder"] ?? "";


        if (!string.IsNullOrEmpty(QueryId))
        {
            IsMovingFile = true;
            ItemToMoveName = QueryId + ".html";
        }
        else if (!string.IsNullOrEmpty(QueryMoveFolder))
        {
            IsMovingFile = false;
            ItemToMoveName = Path.GetFileName(QueryMoveFolder);
        }


        // 2. Track the Destination tree browsing path
        if (Request.QueryString["path"] != null)
        {
            CurrentDestFolder = Request.QueryString["path"];
        }
        else
        {
            // Default the browser drop window to the item's original folder location
            CurrentDestFolder = IsMovingFile ? QueryFolder : Path.GetDirectoryName(QueryMoveFolder).Replace("\\", "/");
        }


        // 3. Handle Form Button Submissions
        string action = Request.QueryString["action"];
        
        // --- NEW RESTORE ACTION ROUTINE ---
    if (action == "restore")
        {
            try
            {
                string targetRestoreFolder = Request.QueryString["destFolder"] ?? "";
                
                // Build file paths mapping from the Trash directory
                string sourceTrashPath = Path.Combine(Server.MapPath("~/uploads/notes/Trash/"), ItemToMoveName);
                
                // Clean the timestamp off the restored file name if one exists (e.g., Note_20260710_051835.html -> Note.html)
                string cleanFileName = ItemToMoveName;
                if (System.Text.RegularExpressions.Regex.IsMatch(ItemToMoveName, @"_\d{8}_\d{6}\.html$", System.Text.RegularExpressions.RegexOptions.IgnoreCase))
                {
                    cleanFileName = System.Text.RegularExpressions.Regex.Replace(ItemToMoveName, @"_\d{8}_\d{6}\.html$", ".html", System.Text.RegularExpressions.RegexOptions.IgnoreCase);
                }
                string destinationPath = Path.Combine(RootNotesPath, targetRestoreFolder.Replace("/", "\\").TrimStart('\\'), cleanFileName);


                if (File.Exists(sourceTrashPath))
                {
                    // Ensure destination subdirectory structure exists
                    string destinationDirectory = Path.GetDirectoryName(destinationPath);
                    if (!Directory.Exists(destinationDirectory))
                    {
                        Directory.CreateDirectory(destinationDirectory);
                    }


                    // Move file out of Trash back home with its clean name
                    File.Move(sourceTrashPath, destinationPath);


                    // Scrub the original tracking entry from trash_manifest.txt using the messy filename it was logged under
                    string manifestPath = Server.MapPath("~/uploads/trash_manifest.txt");
                    if (File.Exists(manifestPath))
                    {
                        var remainingLines = File.ReadAllLines(manifestPath)
                            .Where(line => !line.StartsWith(ItemToMoveName + "=", StringComparison.OrdinalIgnoreCase))
                            .ToArray();


                        File.WriteAllLines(manifestPath, remainingLines);
                    }
                }
                
                // Redirect user back to the Trash view showing the item has successfully vanished from trash
                Response.Redirect("FileBrowser.aspx?folder=TRASH&status=restored", true);
                return;
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine("Error restoring file: " + ex.Message);
            }
        }        
        else if (action == "execute")
        {
            string sourcePath = IsMovingFile 
                ? Path.Combine(RootNotesPath, QueryFolder, ItemToMoveName)
                : Path.Combine(RootNotesPath, QueryMoveFolder);


            string targetPath = Path.Combine(RootNotesPath, CurrentDestFolder, ItemToMoveName);


            try
            {
                List<string> pinnedPaths = GetPinnedPaths();
                bool pinsWereUpdated = false;


                if (IsMovingFile && File.Exists(sourcePath))
                {
                    if (!File.Exists(targetPath)) 
                    {
                        // -------------------------------------------------------------
                        // SCENARIO 1: PIN PATH CORRECTOR FOR SINGLE FILE MOVE
                        // -------------------------------------------------------------
                        string oldFileRel = string.IsNullOrEmpty(QueryFolder) ? ItemToMoveName : QueryFolder + "/" + ItemToMoveName;
                        string newFileRel = string.IsNullOrEmpty(CurrentDestFolder) ? ItemToMoveName : CurrentDestFolder + "/" + ItemToMoveName;


                        string oldFileMatch = oldFileRel.Replace("\\", "/").Trim('/');
                        string newFilePrefix = newFileRel.Replace("\\", "/").Trim('/');


                        for (int i = 0; i < pinnedPaths.Count; i++)
                        {
                            bool hasLeadingSlash = pinnedPaths[i].StartsWith("/") || pinnedPaths[i].StartsWith("\\");
                            string cleanPin = pinnedPaths[i].Replace("\\", "/").Trim('/');


                            if (cleanPin.Equals(oldFileMatch, StringComparison.OrdinalIgnoreCase))
                            {
                                pinnedPaths[i] = (hasLeadingSlash ? "/" : "") + newFilePrefix;
                                pinsWereUpdated = true;
                            }
                        }


                        // Physical disk execution
                        File.Move(sourcePath, targetPath);
                    }
                }
                else if (!IsMovingFile && Directory.Exists(sourcePath))
                {
                    // Prevent moving a folder inside itself
                    if (!targetPath.StartsWith(sourcePath, StringComparison.OrdinalIgnoreCase))
                    {
                        if (!Directory.Exists(targetPath)) 
                        {
                            // -------------------------------------------------------------
                            // SCENARIO 2: PIN PATH CORRECTOR FOR FOLDER MOVE
                            // -------------------------------------------------------------
                            string oldFolderMatch = QueryMoveFolder.Replace("\\", "/").Trim('/');
                            string cleanDest = CurrentDestFolder.Replace("\\", "/").Trim('/');
                            string newFolderPrefix = string.IsNullOrEmpty(cleanDest) 
                                ? ItemToMoveName 
                                : cleanDest + "/" + ItemToMoveName;


                            for (int i = 0; i < pinnedPaths.Count; i++)
                            {
                                bool hasLeadingSlash = pinnedPaths[i].StartsWith("/") || pinnedPaths[i].StartsWith("\\");
                                string cleanPin = pinnedPaths[i].Replace("\\", "/").Trim('/');


                                // Scenario A: An inner file or nested folder path matches
                                if (cleanPin.StartsWith(oldFolderMatch + "/", StringComparison.OrdinalIgnoreCase))
                                {
                                    string remainingPath = cleanPin.Substring(oldFolderMatch.Length);
                                    pinnedPaths[i] = (hasLeadingSlash ? "/" : "") + newFolderPrefix + remainingPath;
                                    pinsWereUpdated = true;
                                }
                                // Scenario B: The exact moved folder itself was pinned
                                else if (cleanPin.Equals(oldFolderMatch, StringComparison.OrdinalIgnoreCase))
                                {
                                    pinnedPaths[i] = (hasLeadingSlash ? "/" : "") + newFolderPrefix;
                                    pinsWereUpdated = true;
                                }
                            }


                            // Physical disk execution
                            Directory.Move(sourcePath, targetPath);
                        }
                    }
                }


                // Save pin tracking file changes if modifications occurred
                if (pinsWereUpdated)
                {
                    SavePinnedPaths(pinnedPaths);
                }
            }
            catch (Exception ex) { /* Optional: Handle error feedback */ }


            // Take user back to the main browser at the newly updated target directory
            Response.Redirect("FileBrowser.aspx?path=" + Server.UrlEncode(CurrentDestFolder));
        }
        else if (action == "cancel")
        {
            // Escape back out to safety
            string returnPath = IsMovingFile ? QueryFolder : Path.GetDirectoryName(QueryMoveFolder).Replace("\\", "/");
            Response.Redirect("FileBrowser.aspx?path=" + Server.UrlEncode(returnPath));
        }
    }


    public string GetParentPath()
    {
        if (string.IsNullOrEmpty(CurrentDestFolder)) return "";
        int lastSlash = CurrentDestFolder.LastIndexOf('/');
        if (lastSlash == -1) return "";
        return CurrentDestFolder.Substring(0, lastSlash);
    }

            // Safety Net: Drop back to root if the requested folder doesn't exist
            if (!Directory.Exists(physicalDir))
            {
                CurrentDestFolder = "";
                physicalDir = RootNotesPath;
            }
            
            foreach (string dir in Directory.GetDirectories(physicalDir)) {
                string dirName = Path.GetFileName(dir);
                string newPath = string.IsNullOrEmpty(CurrentDestFolder) ? dirName : CurrentDestFolder + "/" + dirName;
                
                // Keep the Trash folder visible structure hidden out of the target destination browser loop selector window
                if (dirName.Equals("Trash", StringComparison.OrdinalIgnoreCase)) continue;
