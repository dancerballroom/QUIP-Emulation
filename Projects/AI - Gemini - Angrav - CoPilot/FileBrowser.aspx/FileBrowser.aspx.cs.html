protected void Page_Load(object sender, EventArgs e)
{
    Response.ContentEncoding = System.Text.Encoding.UTF8;
    string PinsFile = Path.Combine(Server.MapPath("~/uploads/"), "pinned_notes.txt");


    // Intercept Route Action Hooks
    if (Request.QueryString["action"] == "rename") { HandleRename(); }
    else if (Request.QueryString["action"] == "renameFolder") { HandleRenameFolder(); }
    else if (Request.QueryString["action"] == "pin")
    {
        string fileToPin = Request.QueryString["file"];
        File.AppendAllLines(PinsFile, new[] { fileToPin });
        Response.Redirect("FileBrowser.aspx?path=" + HttpUtility.UrlEncode(AppState.CurrentFolder));
    }
    else if (Request.QueryString["action"] == "unpin")
    {
        string fileToUnpin = Request.QueryString["file"];
        if (File.Exists(PinsFile))
        {
            var lines = File.ReadAllLines(PinsFile).Where(l => l != fileToUnpin).ToList();
            File.WriteAllLines(PinsFile, lines);
        }
        Response.Redirect("FileBrowser.aspx?path=" + HttpUtility.UrlEncode(AppState.CurrentFolder));
    }
    else if (Request.QueryString["action"] == "emptyTrash")
    {
        // ... (Your native empty trash code remains unmodified here) ...
    }
    
    // =======================================================
    // 📊 NEW EXPORT MULTI-STAGE STEP CONTROLLERS
    // =======================================================
    else if (Request.QueryString["action"] == "export_start")
    {
        Response.Clear();
        string trackingId = HandleExportWebsiteAsync();
        Response.Write(trackingId);
        Response.End();
    }
    else if (Request.QueryString["action"] == "export_progress")
    {
        Response.Clear();
        Response.ContentType = "application/json";
        Response.Write(string.Format("{{\"files\":{0}, \"dirs\":{1}, \"pct\":{2}}}", 
            Session["ExpFiles"] ?? 0, Session["ExpDirs"] ?? 0, Session["ExpPct"] ?? 0));
        Response.End();
    }
    else if (Request.QueryString["action"] == "export_download")
    {
        // Check both potential parameter tracking names to ensure the ID is never null
        string id = Request.QueryString["id"] ?? Request.QueryString["shortId"];
        DownloadFinishedExport(id);
    }


    // State Sync
    AppState.CurrentNote = "";
    string rawPath = Request.QueryString["path"];
    if (rawPath != null) AppState.CurrentFolder = rawPath;
    string resolvedPath = Path.GetFullPath(Path.Combine(RootNotesPath, AppState.CurrentFolder));
    if (!resolvedPath.StartsWith(RootNotesPath)) AppState.CurrentFolder = ""; 
}




private string HandleExportWebsiteAsync()
{
    // Initialize standard session tracking storage fields
    Session["ExpFiles"] = 0;
    Session["ExpDirs"] = 0;
    Session["ExpPct"] = 5;


    string sourcePath = Server.MapPath("~/uploads/notes");
    string shortId = Path.GetRandomFileName().Substring(0, 4);
    string targetWorkPath = Server.MapPath("~/uploads/exp_" + shortId);


    if (!Directory.Exists(sourcePath)) return "ERROR_SOURCE_MISSING";


    // 1. Scan totals to establish a baseline for progress bar math
    int totalFiles = Directory.GetFiles(sourcePath, "*.*", SearchOption.AllDirectories).Length;
    Session["TotalExpFiles"] = totalFiles == 0 ? 1 : totalFiles;


    // 2. Perform tracked local disk staging transfer
    _rootPath = targetWorkPath;
    CopyDirectoryWithTracking(sourcePath, targetWorkPath);
    Session["ExpPct"] = 40;


    // 3. Compile custom HTML static site files
    GenerateIndices(targetWorkPath, true);
    Session["ExpPct"] = 70;
    
    GenerateSearchIndexFile(targetWorkPath);
    GenerateSearchHtmlPage(targetWorkPath);
    Session["ExpPct"] = 85;


    // 4. Archive staging cluster to ZIP output format
    // FIX: Save Zip cleanly outside the staging directory path as 'website_xxxx.zip'
    string zipPath = Server.MapPath(string.Format("~/uploads/website_{0}.zip", shortId));
    if (File.Exists(zipPath)) File.Delete(zipPath);
    
    ZipFile.CreateFromDirectory(targetWorkPath, zipPath);
    Session["ExpPct"] = 100;


    // FIX: Wrap directory deletion in a try-catch block so file I/O locks won't crash the worker thread
    try 
    { 
        if (Directory.Exists(targetWorkPath)) 
        {
            Directory.Delete(targetWorkPath, true); 
        }
    } 
    catch { }


    return shortId; // Returns ID token to JavaScript matching the website_xxxx.zip layout
}




private void CopyDirectoryWithTracking(string sourceDir, string destinationDir)
{
    Directory.CreateDirectory(destinationDir);


    foreach (string file in Directory.GetFiles(sourceDir))
    {
        string name = Path.GetFileName(file);
        File.Copy(file, Path.Combine(destinationDir, name), true);
        
        // Step calculations
        int currentFiles = (int)(Session["ExpFiles"] ?? 0) + 1;
        Session["ExpFiles"] = currentFiles;


        int total = (int)(Session["TotalExpFiles"] ?? 1);
        int piecePct = (int)(((double)currentFiles / total) * 35); // Allocates 35% of bar space to copy task
        Session["ExpPct"] = 5 + piecePct;
    }


    foreach (string subDir in Directory.GetDirectories(sourceDir))
    {
        Session["ExpDirs"] = (int)(Session["ExpDirs"] ?? 0) + 1;
        string name = Path.GetFileName(subDir);
        CopyDirectoryWithTracking(subDir, Path.Combine(destinationDir, name));
    }
}




private void DownloadFinishedExport(string shortId)
{
    // Clean string characters against local folder escaping hijacking vectors
    if (string.IsNullOrEmpty(shortId) || shortId.Length > 6) return;
    
    // FIX: Match the 'website_{0}.zip' naming convention used in HandleExportWebsiteAsync
    string zipPath = Server.MapPath(string.Format("~/uploads/website_{0}.zip", shortId));


    Response.Clear();


    // Safety Fallback: If Windows or the browser hits this code too fast and it's not ready
    if (!File.Exists(zipPath))
    {
        Response.ContentType = "text/html";
        Response.Write("<h3>Backend Error: The temporary ZIP file could not be found on the server. Path checked: " + zipPath + "</h3>");
        Response.End();
        return;
    }


    try
    {
        // Get the exact creation time of this specific zip file to build the filename string
        DateTime creationTime = File.GetCreationTime(zipPath);
        string timestampedName = string.Format("Website_Export_{0}.zip", creationTime.ToString("yyyyMMdd_HHmmss"));


        Response.ContentType = "application/zip";
        
        // Push the file name header down the live pipeline channel securely
        Response.AddHeader("Content-Disposition", string.Format("attachment; filename={0}", timestampedName));
        Response.TransmitFile(zipPath);
        Response.Flush();
    }
    catch (Exception ex)
    {
        Response.ContentType = "text/html";
        Response.Write("<h3>Stream Transmission Error: " + ex.Message + "</h3>");
    }
    finally
    {
        // Completely strip output archive from disk array ONLY after successful transmission or clean fail
        try 
        { 
            if (File.Exists(zipPath)) 
            {
                File.Delete(zipPath); 
            }
        } 
        catch { }
        
        Response.End();
    }
}

    protected void HandleRename() 
    {
        string oldName = Request.QueryString["old"];
        string newName = Request.QueryString["new"];
        
        string targetFolder = Request.QueryString["folder"] ?? AppState.CurrentFolder;
        string dirPath = Path.Combine(RootNotesPath, targetFolder.Replace("/", "\\").TrimStart('\\'));
        
        string oldPath = Path.Combine(dirPath, oldName);
        string newPath = Path.Combine(dirPath, newName.EndsWith(".html", StringComparison.OrdinalIgnoreCase) ? newName : newName + ".html");
        if (File.Exists(oldPath) && !string.IsNullOrWhiteSpace(newName)) File.Move(oldPath, newPath);
        Response.Redirect("FileBrowser.aspx?path=" + HttpUtility.UrlEncode(AppState.CurrentFolder));
    }


    protected void HandleRenameFolder() 
    {
        string oldName = Request.QueryString["old"];
        string newName = Request.QueryString["new"];
        string dirPath = Path.Combine(RootNotesPath, AppState.CurrentFolder.Replace("/", "\\").TrimStart('\\'));
        string oldPath = Path.Combine(dirPath, oldName);
        string newPath = Path.Combine(dirPath, newName);
        
        if (Directory.Exists(oldPath) && !string.IsNullOrWhiteSpace(newName)) 
        {
            try 
            {
                string currentParent = AppState.CurrentFolder ?? ""; 
                string oldRelPath = string.IsNullOrEmpty(currentParent) ? oldName : currentParent + "/" + oldName;
                string newRelPath = string.IsNullOrEmpty(currentParent) ? newName : currentParent + "/" + newName;


                string oldMatchTarget = oldRelPath.Replace("\\", "/").Trim('/');
                string newPrefix = newRelPath.Replace("\\", "/").Trim('/');


                List<string> pinnedPaths = GetPinnedPaths(); 
                bool pinsWereUpdated = false;


                for (int i = 0; i < pinnedPaths.Count; i++)
                {
                    bool hasLeadingSlash = pinnedPaths[i].StartsWith("/") || pinnedPaths[i].StartsWith("\\");
                    string cleanPin = pinnedPaths[i].Replace("\\", "/").Trim('/');


                    if (cleanPin.StartsWith(oldMatchTarget + "/", StringComparison.OrdinalIgnoreCase))
                    {
                        string remainingPath = cleanPin.Substring(oldMatchTarget.Length);
                        pinnedPaths[i] = (hasLeadingSlash ? "/" : "") + newPrefix + remainingPath;
                        pinsWereUpdated = true;
                    }
                    else if (cleanPin.Equals(oldMatchTarget, StringComparison.OrdinalIgnoreCase))
                    {
                        pinnedPaths[i] = (hasLeadingSlash ? "/" : "") + newPrefix;
                        pinsWereUpdated = true;
                    }
                }


                if (pinsWereUpdated)
                {
                    SavePinnedPaths(pinnedPaths); 
                }            
            }
            catch 
            { 
                // Protection lock fallback
            }


            Directory.Move(oldPath, newPath);
        }
        
        Response.Redirect("FileBrowser.aspx?path=" + HttpUtility.UrlEncode(AppState.CurrentFolder));
    }


    protected string GetOriginalLocation(string fileName)
    {
        try
        {
            string uploadsDir = Server.MapPath("~/uploads/");
            string manifestPath = Path.Combine(uploadsDir, "trash_manifest.txt");


            if (!File.Exists(manifestPath)) return "";


            string match = File.ReadLines(manifestPath)
                               .FirstOrDefault(line => line.StartsWith(fileName + "=", StringComparison.OrdinalIgnoreCase));


            if (!string.IsNullOrEmpty(match))
            {
                int eqIndex = match.IndexOf('=');
                if (eqIndex > 0) return match.Substring(eqIndex + 1).Trim();
            }
        }
        catch (Exception ex)
        {
            System.Diagnostics.Debug.WriteLine("Error reading trash manifest: " + ex.Message);
        }
        return "";
    }

    protected string GetHtmlPreview(string filePath)
    {
        try
        {
            if (!File.Exists(filePath)) return string.Empty;


            string content = File.ReadAllText(filePath);
            content = Regex.Replace(content, @"<(script|style)[^>]*?>.*?</\1>", string.Empty, RegexOptions.IgnoreCase | RegexOptions.Singleline);
            string textOnly = Regex.Replace(content, @"<[^>]*>", string.Empty);
            textOnly = HttpUtility.HtmlDecode(textOnly);


            using (StringReader reader = new StringReader(textOnly))
            {
                string line;
                while ((line = reader.ReadLine()) != null)
                {
                    line = line.Trim();
                    if (!string.IsNullOrEmpty(line))
                    {
                        return line.Length > 60 ? line.Substring(0, 57) + "..." : line;
                    }
                }
            }
        }
        catch { /* Fail gracefully */ }
        return "Empty note.";
    }

    protected void HandleExportWebsite()
    {
        string sourcePath = Server.MapPath("~/uploads/notes");
        
        // FIX: Using a super short, randomized 4-character folder name inside /uploads/ instead of AppData+GUID
        string shortId = Path.GetRandomFileName().Substring(0, 4);
        _rootPath = Server.MapPath("~/uploads/exp_" + shortId);
        string zipPath = _rootPath + ".zip";
    
        if (!Directory.Exists(sourcePath))
        {
            Response.Write("Error: Source directory does not exist.");
            Response.End();
            return;
        }
    
        try
        {
            CopyDirectory(sourcePath, _rootPath);
    
            GenerateIndices(_rootPath, true);
            GenerateSearchIndexFile(_rootPath);
            GenerateSearchHtmlPage(_rootPath);
    
            ZipFile.CreateFromDirectory(_rootPath, zipPath);
    
            Response.Clear();
            Response.ContentType = "application/zip";
            Response.AddHeader("Content-Disposition", string.Format("attachment; filename=Website_Export_{0}.zip", DateTime.Now.ToString("yyyyMMdd")));
            Response.TransmitFile(zipPath);
            Response.Flush();
            Response.End();
        }
        catch (Exception ex)
        {
            Response.Write("An error occurred during export: " + ex.Message);
        }
        finally
        {
            if (Directory.Exists(_rootPath)) { Directory.Delete(_rootPath, true); }
            if (File.Exists(zipPath)) { File.Delete(zipPath); }
        }
    }


protected void GenerateIndices(string currentDir, bool isRoot)
{
    string[] subDirs = Directory.GetDirectories(currentDir);
    string[] files = Directory.GetFiles(currentDir);


    StringBuilder html = new StringBuilder();
    
    html.AppendLine("<!DOCTYPE html>");
    html.AppendLine("<html lang='en'>"); // Good practice to include lang
    html.AppendLine("<head>");
    html.AppendLine("    <meta charset='utf-8'>"); // Ensure this is right at the top of the head!
    
    // FIX: Replaced $"..." with string.Format()
    html.AppendLine(string.Format("<title>Index of {0}</title>", GetExportRelativeUrl(currentDir)));
    
    html.AppendLine("<style>body{font-family:sans-serif;padding:20px;line-height:1.6;} ul{list-style-type:none;padding-left:0;} li{margin:8px 0;} a{text-decoration:none;color:#0066cc;} a:hover{text-decoration:underline;}</style>");
    html.AppendLine("</head><body>");
    
    // FIX: Replaced $"..." with string.Format()
    html.AppendLine(string.Format("<h2>Directory Listing for: {0}</h2><hr><ul>", GetExportRelativeUrl(currentDir)));


    if (!isRoot)
    {
        html.AppendLine("<li>📁 <a href='../'><strong>[..] Up to Parent Directory</strong></a></li>");
    }
    else
    {
        html.AppendLine("<li>🔍 <a href='./search.html' style='color:#d9534f;'><strong>Search Entire Directory</strong></a></li><br>");
    }


    foreach (string dir in subDirs)
    {
        DirectoryInfo di = new DirectoryInfo(dir);
        // FIX: Replaced $"..." with string.Format()
        html.AppendLine(string.Format("<li>📁 <a href='{0}/'>{0}/</a></li>", di.Name));
        GenerateIndices(dir, false); 
    }


    foreach (string file in files)
    {
        FileInfo fi = new FileInfo(file);
        
        if (fi.Name.Equals("index.html", StringComparison.OrdinalIgnoreCase) || 
            fi.Name.Equals("search.html", StringComparison.OrdinalIgnoreCase) || 
            fi.Name.Equals("search-index.js", StringComparison.OrdinalIgnoreCase))
        {
            continue;
        }


        // FIX: Replaced $"..." with string.Format()
        html.AppendLine(string.Format("<li>📄 <a href='{0}' target='_blank'>{0}</a></li>", fi.Name));


        _searchIndex.Add(new SearchItem
        {
            Name = fi.Name,
            RelativePath = GetExportRelativeUrl(file)
        });
    }


    // Scroll to the very bottom of the GenerateIndices method and ensure it looks like this:
    html.AppendLine("</ul></body></html>");
    
    // FIX: Explicitly enforce standard UTF-8 stream writer formatting rules
    File.WriteAllText(Path.Combine(currentDir, "index.html"), html.ToString(), new System.Text.UTF8Encoding(true));


}


protected void GenerateSearchIndexFile(string rootPath)
{
    StringBuilder js = new StringBuilder();
    js.AppendLine("// Auto-generated search database\nconst searchIndex = [");
    foreach (var item in _searchIndex)
    {
        // FIX: Replaced $"..." with string.Format()
        js.AppendLine(string.Format("    {{ name: '{0}', path: '.{1}' }},", item.Name.Replace("'", "\\'"), item.RelativePath.Replace("'", "\\'")));
    }
    js.AppendLine("];");
    File.WriteAllText(Path.Combine(rootPath, "search-index.js"), js.ToString(), Encoding.UTF8);
}




    protected string GetExportRelativeUrl(string fullPath)
    {
        if (fullPath.Equals(_rootPath, StringComparison.OrdinalIgnoreCase)) return "/";
        string relative = fullPath.Substring(_rootPath.Length).Replace('\\', '/');
        return relative.StartsWith("/") ? relative : "/" + relative;
    }




protected void GenerateSearchHtmlPage(string rootPath)
{

    StringBuilder sb = new StringBuilder();
    
    // 🎨 This builds the clean, responsive search page template structure

    sb.AppendLine("    <meta charset='UTF-8'>");
    sb.AppendLine("    <meta name='viewport' content='width=device-width, initial-scale=1.0'>");
    sb.AppendLine("    <title>Search Site Index</title>");

    sb.AppendLine("<body>");
    sb.AppendLine("    <a class='back-btn' href='index.html'>⬅️ Back to Index</a>");

    sb.AppendLine("    <input type='text' id='searchBox' class='search-box' placeholder='Type to search names or paths...' oninput='doSearch()'>");
    sb.AppendLine("    <div id='searchResults'></div>");
    sb.AppendLine("    <script src='search-index.js'></script>");
    sb.AppendLine("    <script>");
    sb.AppendLine("        function doSearch() {");
    sb.AppendLine("            const q = document.getElementById('searchBox').value.toLowerCase();");
    sb.AppendLine("            const res = document.getElementById('searchResults');");
    sb.AppendLine("            res.innerHTML = '';");

    sb.AppendLine("            const filtered = searchIndex.filter(item => item.name.toLowerCase().includes(q) || item.path.toLowerCase().includes(q));");
    sb.AppendLine("            filtered.forEach(item => {");
    sb.AppendLine("                const div = document.createElement('div');");
    sb.AppendLine("                div.className = 'result-item';");
    sb.AppendLine("                div.innerHTML = `<a href='${item.path}'>${item.name}</a><br><small style='color:#888'>${item.path}</small>`;");
    sb.AppendLine("                res.appendChild(div);");
    sb.AppendLine("            });");
    sb.AppendLine("            if(filtered.length === 0) { res.innerHTML = '<p style=\"color:#888\">No matches found</p>'; }");
    sb.AppendLine("        }");
    sb.AppendLine("    </script>");
    sb.AppendLine("</body>");
    sb.AppendLine("</html>");


    // Force UTF-8 encoding on the output stream writer


    File.WriteAllText(searchHtmlPath, sb.ToString(), new System.Text.UTF8Encoding(true));


}


    protected static void CopyDirectory(string sourceDir, string destinationDir)
    {
        DirectoryInfo dir = new DirectoryInfo(sourceDir);
        Directory.CreateDirectory(destinationDir);


        foreach (FileInfo file in dir.GetFiles())
        {
            file.CopyTo(Path.Combine(destinationDir, file.Name), true);
        }
        foreach (DirectoryInfo subDir in dir.GetDirectories())
        {
            CopyDirectory(subDir.FullName, Path.Combine(destinationDir, subDir.Name));
        }
    }
