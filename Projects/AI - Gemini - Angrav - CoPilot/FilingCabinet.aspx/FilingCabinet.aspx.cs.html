using System;
using System.IO;
using System.Web;
using System.Web.UI;
using System.Web.UI.WebControls;


public partial class FilingCabinet : System.Web.UI.Page
{
    protected string RootNotesPath
    {
        get { return Server.MapPath("~/uploads/notes/"); }
    }


    protected string CurrentFolder = "";
    protected string SavedNoteHtml = "";
    protected string NoteId = "";
    protected string DisplayFullPath = "";
    protected string PublicNoteUrl = ""; 
    protected global::System.Web.UI.WebControls.TextBox txtEditor;
    protected string BackUrl = "FileBrowser.aspx";




protected void Page_Load(object sender, EventArgs e)
    {
        // 1. Process QueryString parameters first to update AppState and CurrentFolder
        if (Request.QueryString["folder"] != null)
            AppState.CurrentFolder = Request.QueryString["folder"];
            
        if (Request.QueryString["id"] != null)
            AppState.CurrentNote = Request.QueryString["id"];


        CurrentFolder = AppState.CurrentFolder;
        string fileName = AppState.CurrentNote;


        // --- DYNAMIC BACK BUTTON ROUTING ---
        if (!IsPostBack)
        {
            // Default fallback
            BackUrl = "FileBrowser.aspx?path=" + Server.UrlEncode(CurrentFolder);
        
            // 1. Explicit shortcut parameter check
            if (Request.QueryString["src"] == "pm")
            {
                // Preserve the current project folder path if available
                BackUrl = "ProjectManagement.aspx?folder=" + Server.UrlEncode("projects/" + CurrentFolder);
            }
            // 2. HTTP Referrer check (for direct page links)
            else if (Request.UrlReferrer != null)
            {
                string referrerUrl = Request.UrlReferrer.ToString().ToLower();


                if (referrerUrl.Contains("projectmanagement.aspx") || referrerUrl.Contains("searchmgr.aspx"))
                {
                    // Preserve whatever query string ProjectManagement.aspx originally had!
                    // (e.g. "ProjectManagement.aspx?path=Breakfast+and+Lunch")
                    BackUrl = Request.UrlReferrer.PathAndQuery;
                }
                else if (referrerUrl.Contains("filebrowser.aspx") || referrerUrl.Contains("search.aspx"))
                {
                    BackUrl = "FileBrowser.aspx?path=" + Server.UrlEncode(CurrentFolder);
                }
            }
        }
        // ----------------------------------


        if (string.IsNullOrWhiteSpace(fileName))
        {
            fileName = "NewNote_" + DateTime.Now.ToString("yyyyMMdd_HHmmss") + ".html";
            AppState.CurrentNote = fileName;
        }


        if (!fileName.EndsWith(".html", StringComparison.OrdinalIgnoreCase))
        {
            fileName += ".html";
            AppState.CurrentNote = fileName;
        }


        NoteId = fileName;


        string cleanFolder = CurrentFolder.Replace("/", "\\").TrimStart('\\');
        string fullPath = Path.Combine(Server.MapPath("~/uploads/notes/"), cleanFolder, NoteId);


        DisplayFullPath = fullPath;


        string urlFolder = CurrentFolder.Replace("\\", "/").Trim('/');
        if (!string.IsNullOrEmpty(urlFolder)) {
            PublicNoteUrl = "uploads/notes/" + urlFolder + "/" + NoteId;
        } else {
            PublicNoteUrl = "uploads/notes/" + NoteId;
        }


        if (File.Exists(fullPath))
        {
            SavedNoteHtml = ExtractBodyHtml(File.ReadAllText(fullPath));
        }
        else
        {
            SavedNoteHtml = "";
        }
    }






    protected void btnSave_Click(object sender, EventArgs e)
    {
        string noteHtml = Request.Unvalidated.Form["hfEditorHtml"];
        string savedFolder = AppState.CurrentFolder;
        string savedNoteId = AppState.CurrentNote;


        if (string.IsNullOrWhiteSpace(savedNoteId))
        {
            savedNoteId = "NewNote_" + DateTime.Now.ToString("yyyyMMdd_HHmmss") + ".html";
            AppState.CurrentNote = savedNoteId;
        }


        if (!savedNoteId.EndsWith(".html", StringComparison.OrdinalIgnoreCase))
        {
            savedNoteId += ".html";
            AppState.CurrentNote = savedNoteId;
        }


        string targetFolder = Path.Combine(RootNotesPath, savedFolder.TrimStart('/', '\\'));
        Directory.CreateDirectory(targetFolder);
        string targetPath = Path.Combine(targetFolder, savedNoteId);




        // --- BEAUTIFUL FILE GENERATOR ENGINE ---
        // Clean up editor content line endings
        noteHtml = noteHtml.Replace("\r\n", "\n").Replace("\r", "\n");
        
        // Trim empty lines from the very beginning and very end of the raw editor input
        noteHtml = noteHtml.Trim('\n', '\r', ' ');
        
        // SMART INDENTER: Split into lines and only indent lines that actually have content.
        // We track if we are inside a <pre> block so we don't destroy pre-formatted indentation!
        string[] lines = noteHtml.Split(new[] { '\n' }, StringSplitOptions.None);
        bool insidePre = false;
        
        for (int k = 0; k < lines.Length; k++)
        {
            string trimmedLine = lines[k].Trim();
        
            // Check if we are entering or leaving a <pre> block
            // (Using IndexOf to catch lines like <pre class="..."> or </pre>)
            if (trimmedLine.IndexOf("<pre", StringComparison.OrdinalIgnoreCase) >= 0)
            {
                // The <pre> opening tag itself gets standard indentation
                lines[k] = "        " + lines[k].TrimStart(' ', '\t');
                insidePre = true;
                continue; 
            }
            
            if (trimmedLine.IndexOf("</pre>", StringComparison.OrdinalIgnoreCase) >= 0)
            {
                // The </pre> closing tag itself gets standard indentation
                lines[k] = "        " + lines[k].TrimStart(' ', '\t');
                insidePre = false;
                continue;
            }
        
            // If we are INSIDE the <pre> block, DO NOT touch the indentation at all!
            if (insidePre)
            {
                // Leave the line exactly as-is (preserves all tabs/spaces/newlines)
                continue; 
            }
        
            // Normal HTML Line processing:
            if (string.IsNullOrWhiteSpace(lines[k]))
            {
                lines[k] = "";
            }
            else
            {
                // Trim any existing leading spaces/tabs first, THEN add exactly 8 spaces!
                lines[k] = "        " + lines[k].TrimStart(' ', '\t');
            }
        }
        String nestedBodyHtml = string.Join("\r\n", lines);


        // Get a clean version of the file name without ".html" to use as the page Title
        string cleanTitle = Path.GetFileNameWithoutExtension(savedNoteId);


        // Construct a perfectly structured, readable web page
        string finalCleanHtml = 
            "<!DOCTYPE html>\r\n" +
            "<html>\r\n" +
            "    <head>\r\n" +
            "        <meta charset='utf-8' />\r\n" +
            "        <title>" + cleanTitle + "</title>\r\n" +
            "    </head>\r\n" +
            "    <body>\r\n" +
            nestedBodyHtml + "\r\n" +
            "    </body>\r\n" +
            "</html>";


        File.WriteAllText(targetPath, finalCleanHtml, System.Text.Encoding.UTF8);
        // ---------------------------------------


        Response.Redirect(
            "FilingCabinet.aspx?id=" + HttpUtility.UrlEncode(savedNoteId) +
            "&folder=" + HttpUtility.UrlEncode(savedFolder) + "&saved=true",
            true
        );
    }
    
    
    protected string ExtractBodyHtml(string fullHtml)
    {
        if (string.IsNullOrWhiteSpace(fullHtml)) return "";


        System.Text.RegularExpressions.Match match =
            System.Text.RegularExpressions.Regex.Match(
                fullHtml,
                @"<body[^>]*>(?<body>[\s\S]*?)</body>",
                System.Text.RegularExpressions.RegexOptions.IgnoreCase
            );


        // Extract the content, clean up excess baseline indentations so it edits seamlessly
        if (match.Success)
        {
            string bodyContent = match.Groups["body"].Value;
            return bodyContent.Trim('\r', '\n');
        }
        
        return fullHtml;
    }


    protected void btnSendToTrash_Click(object sender, EventArgs e)
    {
        string fileName = Request.QueryString["id"]; 
        string folderPath = Request.QueryString["folder"];
        
        if (string.IsNullOrEmpty(fileName)) return;


        if (!fileName.EndsWith(".html", StringComparison.OrdinalIgnoreCase) && !fileName.EndsWith(".htm", StringComparison.OrdinalIgnoreCase))
        {
            fileName += ".html";
        }


        try
        {
            string cleanSubFolder = !string.IsNullOrEmpty(folderPath) ? folderPath.Replace("/", "\\").TrimStart('\\') : "";
            string sourcePath = Server.MapPath("~/uploads/notes/" + (string.IsNullOrEmpty(cleanSubFolder) ? "" : cleanSubFolder + "/") + fileName);


            if (File.Exists(sourcePath))
            {
                string trashDirectory = Server.MapPath("~/uploads/notes/Trash");
                if (!Directory.Exists(trashDirectory))
                {
                    Directory.CreateDirectory(trashDirectory);
                }


                string pureFileName = Path.GetFileNameWithoutExtension(sourcePath);
                string extension = Path.GetExtension(sourcePath);
                string timestamp = DateTime.Now.ToString("yyyyMMdd_HHmmss");
                
                string trashFileName = pureFileName + "_" + timestamp + extension;
                string trashDestinationPath = Path.Combine(trashDirectory, trashFileName);


                string uploadsDirectory = Server.MapPath("~/uploads/");
                if (!Directory.Exists(uploadsDirectory))
                {
                    Directory.CreateDirectory(uploadsDirectory);
                }
                string manifestPath = Path.Combine(uploadsDirectory, "trash_manifest.txt");
                string originalFolderMapping = folderPath ?? ""; 
                
                File.AppendAllText(manifestPath, trashFileName + "=" + originalFolderMapping + Environment.NewLine);


                File.Move(sourcePath, trashDestinationPath);


                Response.Redirect("FileBrowser.aspx?path=" + Server.UrlEncode(folderPath) + "&status=deleted");
            }
            else
            {
                ClientScript.RegisterStartupScript(this.GetType(), "alert", "alert('Error: The target file could not be located on the server.');", true);
            }
        }
        catch (Exception ex)
        {
            ClientScript.RegisterStartupScript(this.GetType(), "alert", "alert('File deletion failed: " + ex.Message.Replace("'", "\\'") + "');", true);
        }
    }
