    function processColumnSelection(rect) {
        let sel = window.getSelection();
        sel.removeAllRanges();
        
        let firstRange = null;
        let lastRange = null;
        
        let treeWalker = document.createTreeWalker(editorBox, NodeFilter.SHOW_TEXT, null, false);
        let tempRange = document.createRange();
        let node;
        
        while(node = treeWalker.nextNode()) {
            let text = node.textContent;
            for (let i = 0; i < text.length; i++) {
                tempRange.setStart(node, i);
                tempRange.setEnd(node, i + 1);
                let charRect = tempRange.getBoundingClientRect();
                
                if (charRect.left >= rect.left && charRect.right <= rect.right &&
                    charRect.top >= rect.top && charRect.bottom <= rect.bottom) {
                    
                    if (!firstRange) {
                        firstRange = tempRange.cloneRange();
                    }
                    lastRange = tempRange.cloneRange();
                }
            }
        }


        if (firstRange && lastRange) {
            let finalRange = document.createRange();
            finalRange.setStart(firstRange.startContainer, firstRange.startOffset);
            finalRange.setEnd(lastRange.endContainer, lastRange.endOffset);
            sel.addRange(finalRange);
        }
    }

function trimTrailingWhitespaceWithoutFlattening(codeBox) {
    if (!codeBox) return;


    function findLastTextNode(node) {
        if (!node) return null;
        if (node.nodeType === Node.TEXT_NODE) return node;


        var childNodes = node.childNodes || [];
        for (var i = childNodes.length - 1; i >= 0; i--) {
            var foundNode = findLastTextNode(childNodes[i]);
            if (foundNode) return foundNode;
        }
        return null;
    }


    var lastTextNode = findLastTextNode(codeBox);
    if (!lastTextNode) return;


    var trimmedValue = lastTextNode.nodeValue.replace(/\s+$/, '');
    if (trimmedValue !== lastTextNode.nodeValue) {
        lastTextNode.nodeValue = trimmedValue;
    }


    if (lastTextNode.nodeValue.length === 0 && lastTextNode.parentNode) {
        lastTextNode.parentNode.removeChild(lastTextNode);
    }
}

function applyFormattingToCurrentLine(styleType, value) {
    const selection = window.getSelection();
    if (!selection.rangeCount) return;


    const range = selection.getRangeAt(0);
    const startNode = range.startContainer;


    // 1. Find the current block/line element (e.g., a <div> or <p> inside a contenteditable area)
    let currentLine = startNode.nodeType === Node.TEXT_NODE ? startNode.parentNode : startNode;
    while (currentLine && currentLine.contentEditable !== "true" && 
           window.getComputedStyle(currentLine).display !== "block" && 
           currentLine.tagName !== "DIV" && currentLine.tagName !== "P") {
        currentLine = currentLine.parentNode;
    }


    if (!currentLine) return;


    // Save the cursor offset position so we can return it later
    const originalOffset = range.startOffset;


    // 2. Programmatically "Highlight" (Select) the current line
    const lineRange = document.createRange();
    lineRange.selectNodeContents(currentLine);
    selection.removeAllRanges();
    selection.addRange(lineRange);


    // 3. Clear existing formatting inside this line first
    if (typeof clearSelectionFormatting === "function") {
        clearSelectionFormatting();
    }


    // 4. Apply the new Font Size or Color
    if (styleType === "fontSize") {
        document.execCommand("fontSize", false, value);
        
        // Optional: If you use exact pixel/rem sizes instead of legacy execCommand (1-7) sizes
        // currentLine.style.fontSize = value; 
    } else if (styleType === "color") {
        document.execCommand("foreColor", false, value);
    }


    // 5. "Un-highlight" by collapsing the selection back to a single blinking cursor
    // This moves the cursor back to its original relative position on the line
    try {
        const resetRange = document.createRange();
        resetRange.setStart(startNode, originalOffset);
        resetRange.collapse(true);
        selection.removeAllRanges();
        selection.addRange(resetRange);
    } catch (e) {
        // Fallback: if structure changed completely, drop cursor at the end of the line
        selection.collapseToEnd();
    }
}

//Utility Function: clearSelectionFormatting()
function clearSelectionFormatting(e) {
    if (e) {
        e.preventDefault();
        e.stopPropagation();
    }


    console.log("-> clearSelectionFormatting() triggered");


    if (typeof isEditorLocked !== 'undefined' && isEditorLocked) {
        return;
    }


    var editorBox = document.getElementById("editorBox");
    var selection = window.getSelection();
    
    if (!selection.rangeCount || selection.isCollapsed) {
        // Fallback: Clear formatting on the single current element parent if nothing highlighted
        var range = selection.getRangeAt(0);
        var currentNode = range.startContainer;
        var targetEl = currentNode.nodeType === 3 ? currentNode.parentNode : currentNode;


        if (targetEl && targetEl !== editorBox && targetEl.tagName !== "DIV" && targetEl.tagName !== "P") {
            targetEl.removeAttribute("style");
        }
    } else {
        // Text IS successfully highlighted
        try {
            document.execCommand("removeFormat", false, null);
            document.execCommand("unlink", false, null);
            
            var range = selection.getRangeAt(0);
            var clonedContents = range.cloneContents();
            
            // Line-safe styling sweep
            var allElements = clonedContents.querySelectorAll("*");
            allElements.forEach(function(el) {
                var tagName = el.tagName.toUpperCase();
                if (tagName === "DIV" || tagName === "P" || tagName === "BR") {
                    el.style.color = ""; 
                    el.style.backgroundColor = "";
                } else {
                    el.removeAttribute("style");
                    el.removeAttribute("color");
                    el.removeAttribute("face");
                    el.removeAttribute("size");
                }
            });


            range.deleteContents();
            range.insertNode(clonedContents);
            console.log("Line-safe DOM style sweep executed successfully.");
        } catch (err) {
            console.error("Format clearing engine ran into an execution error:", err);
        }
    }


    // ========================================================
    // LINE REPAIR ENGINE: Rebuilds line-by-line capability
    // ========================================================
    if (editorBox) {
        // Force raw top-level loose text fragments to wrap cleanly into structural block lines
        var nodes = Array.from(editorBox.childNodes);
        nodes.forEach(function(node) {
            // If we find raw, unwrapped text nodes at the root level of the editor
            if (node.nodeType === Node.TEXT_NODE && node.nodeValue.trim() !== "") {
                var divWrapper = document.createElement("div");
                editorBox.insertBefore(divWrapper, node);
                divWrapper.appendChild(node);
            }
            // If we find loose inline elements (like a naked span or b tag) that lost their line blocks
            else if (node.nodeType === Node.ELEMENT_NODE && ["SPAN", "B", "I", "U", "FONT", "A"].includes(node.tagName)) {
                var divWrapper = document.createElement("div");
                editorBox.insertBefore(divWrapper, node);
                divWrapper.appendChild(node);
            }
        });
        console.log("Editor line-by-line infrastructure repaired.");
    }


    // Update backend data bridge field sync
    var hfBridge = document.getElementById("hfEditorHtml");
    if (hfBridge && editorBox) {
        hfBridge.value = encodeURIComponent(editorBox.innerHTML);
    }
    
    if (typeof setDirty === "function") {
        setDirty();
    }
}

// Helper function to convert RGB colors back to Hex format for accurate matching
function rgbToHex(rgb) {
    if (!rgb || rgb.indexOf("rgb") === -1) return rgb;
    var match = rgb.match(/^rgba?\((\d+),\s*(\d+),\s*(\d+)/);
    if (!match) return rgb;
    function hex(x) {
        return ("0" + parseInt(x).toString(16)).slice(-2);
    }
    return "#" + hex(match[1]) + hex(match[2]) + hex(match[3]);
}


function cycleCurrentLineColor() {
    // 1. Get the user's cursor position selection
    var selection = window.getSelection();
    if (!selection.rangeCount) return;


    var range = selection.getRangeAt(0);
    var currentNode = range.startContainer;
    var editorBox = document.getElementById("editorBox");


    // 2. Drill up to find the element block representing the current line
    var currentLineElement = currentNode.nodeType === 1 ? currentNode : currentNode.parentNode;
    
    while (currentLineElement && currentLineElement.parentNode !== editorBox && currentLineElement !== editorBox) {
        currentLineElement = currentLineElement.parentNode;
    }


    // 3. Robust Fallback: If text is loose/separated by <br> tags, wrap the target text in a styled span
    if (currentLineElement === editorBox || !currentLineElement) {
        var nextColorHex = LINE_COLORS[1]; // Default to Red if no previous color found
        
        // Let's check what color the text currently is by default
        var activeColor = window.getComputedStyle(editorBox).color;
        var activeHex = rgbToHex(activeColor).toUpperCase();
        var matchIndex = LINE_COLORS.findIndex(function(c) { return c.toUpperCase() === activeHex; });
        if (matchIndex !== -1) {
            nextColorHex = LINE_COLORS[(matchIndex + 1) % LINE_COLORS.length];
        }


        // Execute a native foreground color command to cleanly handle loose text/selections
        document.execCommand("foreColor", false, nextColorHex);
        
        // Sync up the hidden bridge field safely
        var hfBridge = document.getElementById("hfEditorHtml");
        if (hfBridge) {
            hfBridge.value = encodeURIComponent(editorBox.innerHTML);
        }
        return; 
    }


    // 4. Standard Block Element Route (div, p, h1, etc.)
    if (currentLineElement && currentLineElement !== editorBox) {
        var currentStyleColor = window.getComputedStyle(currentLineElement).color;
        var currentHex = rgbToHex(currentStyleColor).toUpperCase();


        var currentColorIndex = LINE_COLORS.findIndex(function(color) {
            return color.toUpperCase() === currentHex;
        });


        var nextColorIndex = (currentColorIndex + 1) % LINE_COLORS.length;
        var nextColorHex = LINE_COLORS[nextColorIndex];


        // Apply style modification
        currentLineElement.style.color = nextColorHex;


        // Sync up the hidden bridge field safely
        var hfBridge = document.getElementById("hfEditorHtml");
        if (hfBridge) {
            hfBridge.value = encodeURIComponent(editorBox.innerHTML);
        }
    }
}

function applySyntaxHighlighting(e, styleFileType) {
    if (e) {
        e.preventDefault();
        e.stopPropagation();
    }


    if (typeof isEditorLocked !== 'undefined' && isEditorLocked) return;


    // 1. Synchronously capture text boundaries
    var selection = window.getSelection();
    if (!selection.rangeCount || selection.isCollapsed) {
        console.warn("Syntax highlighting aborted: No code text was highlighted.");
        return;
    }


    var preservedRange = selection.getRangeAt(0).cloneRange();
    var extractionContainer = document.createElement("div");
    extractionContainer.appendChild(preservedRange.cloneContents());


    // Swap line breaks explicitly out of DOM nodes
    var brTags = extractionContainer.querySelectorAll("br");
    for (var b = 0; b < brTags.length; b++) {
        brTags[b].parentNode.replaceChild(document.createTextNode("\n"), brTags[b]);
    }
    var blockElements = extractionContainer.querySelectorAll("div, p");
    for (var d = 0; d < blockElements.length; d++) {
        var block = blockElements[d];
        if (block.nextSibling) block.appendChild(document.createTextNode("\n"));
    }


    var rawCodeText = extractionContainer.textContent || extractionContainer.innerText || "";
    if (!rawCodeText.trim()) return;


    console.log("Fetching formatting rules for: " + styleFileType);


    // 2. Query Handler
    fetch('GetSyntaxRules.ashx?type=' + encodeURIComponent(styleFileType))
        .then(function(response) {
            if (!response.ok) throw new Error("Configuration file fetch failed.");
            return response.json();
        })
        .then(function(stylingRules) {


            // 3. Convert snippet text characters safely into HTML entities
            var escapedHtml = rawCodeText
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;");


            // 4. Sort keys longest-to-shortest
            var sortedKeys = Object.keys(stylingRules).sort(function(a, b) {
                return b.length - a.length;
            });


// 5. UNIFIED REPLACEMENT ENGINE
            sortedKeys.forEach(function(tokenKey) {
                var styleData = stylingRules[tokenKey];
                
                var cssString = "";
                if (styleData.color) cssString += "color:" + styleData.color + ";";
                if (styleData.fontWeight) cssString += "font-weight:" + styleData.fontWeight + ";";
                if (styleData.fontStyle) cssString += "font-style:" + styleData.fontStyle + ";";


                // Clean backslashes from JSON parsing
                var cleanKey = tokenKey.replace(/\\"/g, '"');


                // Extract the core word, removing any brackets or slashes (e.g., "<head>" or "head" both become "head")
                var pureWord = cleanKey.replace(/[<>\/!]/g, "").split(" ")[0];
                if (!pureWord) return;


                var safeTagName = pureWord.replace(/&/g, "&amp;");
                var escapedToken = safeTagName.replace(/[-\/\\^$*+?.()|[\]{}]/g, '\\$&');
                var regexPattern = "";


                // Check if this key belongs to a standard HTML tag
                var htmlTags = ["html", "body", "head", "title", "meta", "link", "script", "div", "span", "p", "a", "img"];
                
                if (htmlTags.indexOf(pureWord.toLowerCase()) !== -1 || cleanKey.indexOf("<") !== -1) {
                    // Matches opening tags with attributes (<meta charset="UTF-8">) AND closing tags (</head>)
                    regexPattern = "(&lt;" + escapedToken + "\\b(?:\\s[^&]*?)?&gt;|&lt;\\/" + escapedToken + "\\b&gt;)";
                } else {
                    // Standard text block word boundary check for standalone attributes (class, href, action)
                    regexPattern = "\\b" + escapedToken + "\\b(?!(?=[^<]*>))";
                }


                var unifiedRegex = new RegExp(regexPattern, "gi");
                escapedHtml = escapedHtml.replace(unifiedRegex, function(match) {
                    return '<span style="' + cssString + '">' + match + '</span>';
                });
            });




            // 6. RENDER BACK TO DOM
            var templateContainer = document.createElement("span");
            templateContainer.style.whiteSpace = "pre-wrap";
            templateContainer.style.display = "inline-block";
            templateContainer.style.width = "100%";
            templateContainer.innerHTML = escapedHtml;


            preservedRange.deleteContents();
            preservedRange.insertNode(templateContainer);


            // Re-establish cursor layout positions
            selection.removeAllRanges();
            var postHighlightRange = document.createRange();
            postHighlightRange.selectNodeContents(templateContainer);
            postHighlightRange.collapse(false); 
            selection.addRange(postHighlightRange);


            // Save state fields
            var hfBridge = document.getElementById("hfEditorHtml");
            var editorBox = document.getElementById("editorBox");
            if (hfBridge && editorBox) {
                hfBridge.value = encodeURIComponent(editorBox.innerHTML);
            }
            if (typeof setDirty === "function") setDirty();
        })
        .catch(function(err) {
            console.error("Syntax processor engine threw an execution error:", err);
        });
}








// Draggable Window Utility Engine
function makePanelDraggable(panel, header) {
    var pos1 = 0, pos2 = 0, pos3 = 0, pos4 = 0;
    
    if (header) {
        header.onmousedown = dragMouseDown;
    }


    function dragMouseDown(e) {
        e = e || window.event;
        // Ignore clicks on the close button when trying to drag
        if (e.target.tagName.toLowerCase() === 'button') return;
        
        e.preventDefault();
        // Get initial mouse position layout
        pos3 = e.clientX;
        pos4 = e.clientY;
        document.onmouseup = closeDragElement;
        document.onmousemove = elementDrag;
    }


    function elementDrag(e) {
        e = e || window.event;
        e.preventDefault();
        // Calculate dynamic positional adjustments
        pos1 = pos3 - e.clientX;
        pos2 = pos4 - e.clientY;
        pos3 = e.clientX;
        pos4 = e.clientY;
        // Apply offset changes straight to the panel styling vectors
        panel.style.top = (panel.offsetTop - pos2) + "px";
        panel.style.left = (panel.offsetLeft - pos1) + "px";
    }


    function closeDragElement() {
        // Drop listeners safely when mouse button is released
        document.onmouseup = null;
        document.onmousemove = null;
    }
}

function beautifyHTML(htmlString) {
    //alert("I am running the NEW beautify code!");
    if (!htmlString) return "";


    // ==========================================
    // SANITIZATION ENGINE
    // Convert Unicode non-breaking spaces and &nbsp; entities to normal spaces
    // ==========================================
    htmlString = htmlString.replace(/\u00A0/g, " ");
    htmlString = htmlString.replace(/&nbsp;/g, " ");


    // ==========================================
    // SAFE STAIRCASE HEALER 
    // Only fixes raw, single-line staircase bugs.
    // Does NOT match once the code has newlines/indents.
    // ==========================================
    htmlString = htmlString.replace(/<div><br><\/div><div>/gi, "<div><br></div><div>"); 
    htmlString = htmlString.replace(/<div><br><div>/gi, "<div><br></div><div>");


    // 1. PRESERVE BLOCK ENGINE: Protect <pre> blocks from being formatted internally
    var preservedBlocks = [];
    var preservedCounter = 0;
    
    var cleanHtml = htmlString.replace(/<pre[^>]*>([\s\S]*?)<\/pre>/gi, function(match) {
        var placeholder = "";
        preservedBlocks.push({ placeholder: placeholder, original: match });
        return placeholder;
    });


    // ==========================================
    // 2. BULLETPROOF TOKENIZER
    // Manually parse the HTML to safely isolate tags with complex styles/attributes
    // ==========================================
    var tokens = [];
    var lastIndex = 0;
    var tagRegExp = /<[^>]+>/g;
    var match;


    while ((match = tagRegExp.exec(cleanHtml)) !== null) {
        // Grab any text content before this tag
        var textBefore = cleanHtml.substring(lastIndex, match.index);
        if (textBefore.trim() !== "") {
            tokens.push(textBefore);
        }
        // Grab the tag itself
        tokens.push(match[0]);
        lastIndex = tagRegExp.lastIndex;
    }
    // Grab any remaining trailing text
    var remainingText = cleanHtml.substring(lastIndex);
    if (remainingText.trim() !== "") {
        tokens.push(remainingText);
    }


    var formattedHtml = "";
    var indentLevel = 0;
    var indentString = "    "; // 4 spaces


    // Define structural elements that deserve their own lines and indentation
    var blockElements = [
        "html", "body", "head", "div", "p", "table", "thead", "tbody", "tr", "ul", "ol", "li", "section", 
        "header", "footer", "h1", "h2", "h3", "h4", "h5", "h6", "blockquote", "form", "select", "option",
        "!doctype", "br"
    ];


    // Explicitly define tags that MUST stay inline and NEVER affect formatting or spacing math
    var inlineElements = [
        "span", "a", "u", "b", "i", "em", "strong", "img", "small", "code", "sub", "sup"
    ];


    function isBlock(tagName) {
        var lowerTag = tagName.toLowerCase();
        if (inlineElements.indexOf(lowerTag) !== -1) return false;
        return blockElements.indexOf(lowerTag) !== -1;
    }


    for (var i = 0; i < tokens.length; i++) {
        var rawToken = tokens[i];
        var trimmedToken = rawToken.trim();


        if (!trimmedToken) continue;


        // Case A: It's an HTML Tag
        if (trimmedToken.charAt(0) === "<" && trimmedToken.charAt(trimmedToken.length - 1) === ">") {
            var token = trimmedToken;
            var isClosing = token.indexOf("</") === 0;
            var isSelfClosing = token.indexOf("/>") !== -1 || 
                                token.indexOf("<!") === 0 || 
                                /^(<br|<hr|<img|<input|<meta|<link)/i.test(token);


            var tagNameMatch = token.match(/<\/?([a-zA-Z0-9\-!]+)/);
            var tagName = tagNameMatch ? tagNameMatch[1].toLowerCase() : "";


            // Ignore totally inline helper tags to keep styling safe
            if (inlineElements.indexOf(tagName) !== -1) {
                formattedHtml += token;
                continue;
            }


            if (isClosing) {
                if (isBlock(tagName)) {
                    indentLevel = Math.max(0, indentLevel - 1);
                    formattedHtml += "\n" + indentString.repeat(indentLevel);
                }
                formattedHtml += token;
            } else {
                if (isBlock(tagName)) {
                    if (formattedHtml.length > 0) {
                        formattedHtml += "\n";
                    }
                    formattedHtml += indentString.repeat(indentLevel);
                    formattedHtml += token;
                    if (!isSelfClosing && tagName !== "!doctype") {
                        indentLevel++;
                    }
                } else {
                    formattedHtml += token;
                }
            }
        } 
        // Case B: It's actual text/content
        else {
            // Clean internal spacing
            var cleanedText = trimmedToken.replace(/[\t\r\n]+/g, " ").replace(/ {2,}/g, " ");
            if (cleanedText) {
                formattedHtml += cleanedText;
            }
        }
    }


    // 3. BLANK LINE ENGINE: Collapse excessive newlines
    formattedHtml = formattedHtml.replace(/\n([ \t]*\n){2,}/g, "\n\n");


    // 4. RESTORE BLOCK ENGINE: Re-inject preserved pre blocks
    for (var j = 0; j < preservedBlocks.length; j++) {
        formattedHtml = formattedHtml.replace(preservedBlocks[j].placeholder, preservedBlocks[j].original);
    }


    // ==========================================
    // 5. BODY TRIM ENGINE & CLEANUP 
    // Limits empty space at the top and bottom of the body
    // ==========================================
    
    // A. Collapse excessive space right after <body> (Set exactly 1 blank line space)
    formattedHtml = formattedHtml.replace(/(<body[^>]*>)\s*\n/gi, "$1\n\n");
    
    // B. Clean up empty divs at the bottom so they don't produce trailing run-on lines
    formattedHtml = formattedHtml.replace(/<div>\s*<\/div>/gi, "<div></div>");


    // C. LASER BOTTOM TRIMMER:
    // Finds only the *last* physical </body> tag to prevent splitting on syntax-highlighted code.
    var bodyCloseIndex = formattedHtml.toLowerCase().lastIndexOf("</body>");
    if (bodyCloseIndex !== -1) {
        var beforeBodyClose = formattedHtml.substring(0, bodyCloseIndex);
        var afterBodyClose = formattedHtml.substring(bodyCloseIndex); // Includes "</body>" and everything after


        // Trim ALL trailing whitespaces, newlines, and carriage returns completely from the content before </body>
        beforeBodyClose = beforeBodyClose.replace(/[\s\r\n\u00A0]+$/g, "");


        // Reconstruct with EXACTLY 2 blank lines (3 newlines total) before </body> to keep it tight and clean
        formattedHtml = beforeBodyClose + "\n\n\n" + afterBodyClose;
    }


    return formattedHtml.trim();
}

function getCaretCharacterOffsetWithin(element) {
    var caretOffset = 0;
    var doc = element.ownerDocument || element.document;
    var win = doc.defaultView || doc.parentWindow;
    var sel;
    if (typeof win.getSelection != "undefined") {
        sel = win.getSelection();
        if (sel.rangeCount > 0) {
            var range = win.getSelection().getRangeAt(0);
            var preCaretRange = range.cloneRange();
            preCaretRange.selectNodeContents(element);
            preCaretRange.setEnd(range.startContainer, range.startOffset);
            caretOffset = preCaretRange.toString().length;
        }
    } else if ( (sel = doc.selection) && sel.type != "Control") {
        var textRange = sel.createRange();
        var preCaretTextRange = doc.body.createTextRange();
        preCaretTextRange.moveToElementText(element);
        preCaretTextRange.setEndPoint("EndToStart", textRange);
        caretOffset = preCaretTextRange.text.length;
    }
    return caretOffset;
}


function toggleColumnFormatting() {
    if (isEditorLocked) return;
    
    var selection = window.getSelection();
    if (selection.rangeCount === 0) return;
    
    var range = selection.getRangeAt(0);
    var editorBox = document.getElementById("editorBox");
    if (!editorBox) return;


    // --- STEP 1: DETECT IF WE ARE ALREADY INSIDE COLUMNS (TOGGLE OFF) ---
    var startNode = range.startContainer;
    var columnContainer = startNode.nodeType === Node.ELEMENT_NODE 
        ? startNode.closest('.editor-two-columns') 
        : startNode.parentElement.closest('.editor-two-columns');


    if (columnContainer) {
        // --- TOGGLE OFF: Unwrap content and flow back into normal layout ---
        var parent = columnContainer.parentNode;
        var fragment = document.createDocumentFragment();
        
        // Move all children out of the column box and into our fragment
        while (columnContainer.firstChild) {
            fragment.appendChild(columnContainer.firstChild);
        }
        
        // Replace the column container with the flat content
        parent.replaceChild(fragment, columnContainer);
        
        setDirty();
        editorBox.focus();
        return;
    }


    // --- STEP 2: TOGGLE ON (CREATE TWO COLUMNS) ---
    if (selection.isCollapsed) {
        alert("Please highlight the text/paragraphs you want to format into columns.");
        return;
    }


    // Capture selection content
    var fragmentContents = range.cloneContents();
    
    // Ensure we actually have content to work with
    if (!fragmentContents.textContent.trim()) return;


    // Create our column wrapper
    var colWrapper = document.createElement("div");
    colWrapper.className = "editor-two-columns";


    // If the selection is just loose inline text, wrap it in a paragraph block inside the columns 
    // to keep the editor's block styling clean.
    var hasBlockElements = fragmentContents.querySelector('p, div, h1, h2, h3, h4, pre');
    if (!hasBlockElements) {
        var p = document.createElement("p");
        p.appendChild(fragmentContents);
        colWrapper.appendChild(p);
    } else {
        colWrapper.appendChild(fragmentContents);
    }


    // Create the document fragment to inject
    var finalFragment = document.createDocumentFragment();
    
    // Ensure there's a paragraph break before the columns so the user can edit above it
    var topEscape = document.createElement("div");
    topEscape.innerHTML = "<br>";
    finalFragment.appendChild(topEscape);


    finalFragment.appendChild(colWrapper);


    // Ensure there's an escape paragraph below the columns for normal editing
    var bottomEscape = document.createElement("div");
    bottomEscape.innerHTML = "<br>";
    finalFragment.appendChild(bottomEscape);


    // Delete the selection and insert the fresh column block
    range.deleteContents();
    range.insertNode(finalFragment);


    // Place the cursor in the bottom escape line so the user can continue typing easily
    var newRange = document.createRange();
    newRange.setStart(bottomEscape, 0);
    newRange.collapse(true);
    selection.removeAllRanges();
    selection.addRange(newRange);


    setDirty();
    editorBox.focus();
}

async function saveCodeBoxText(button) {
    var container = button.closest('.code-box-container');
    if (!container) return;
    var contentBlock = container.querySelector('.code-box-content');
    if (!contentBlock) return;
    var rawText = contentBlock.innerText || contentBlock.textContent || "";
    if (!rawText.trim()) return;


    function getLastMeaningfulLine(node) {
        if (!node) return "";


        var textValue = "";
        if (node.nodeType === Node.TEXT_NODE) {
            textValue = node.textContent || "";
        } else if (node.nodeType === Node.ELEMENT_NODE) {
            if (node.tagName && node.tagName.toLowerCase() === 'br') {
                return "";
            }
            textValue = node.innerText || node.textContent || "";
        }


        textValue = textValue.replace(/\u00A0/g, ' ').trim();
        if (!textValue) return "";


        var lines = textValue.split(/\r?\n/).filter(function(line) {
            return line.trim().length > 0;
        });


        return lines.length > 0 ? lines[lines.length - 1].trim() : "";
    }


    // 1. Look for the last visible line immediately above the code box container
    var suggestedTitle = "";
    var prevNode = container.previousSibling;
    
    while (prevNode) {
        var lineCandidate = getLastMeaningfulLine(prevNode);
        if (lineCandidate) {
            suggestedTitle = lineCandidate;
            break;
        }
        prevNode = prevNode.previousSibling;
    }


    // Fallback: If no text is found above the box, use the first line inside the box
    if (!suggestedTitle) {
        var boxLines = rawText.trim().split(/\r?\n/).filter(function(line) {
            return line.trim().length > 0;
        });
        suggestedTitle = boxLines.length > 0 ? boxLines[0].trim() : "code_box_export";
    }


    // Clean up filename reserved characters
    var defaultFilename = suggestedTitle.replace(/[/\\?%*:|"<>]/g, '').trim();
    if (!defaultFilename) {
        defaultFilename = "code_box_export.txt";
    }


    // Determine file extension for the dialog filter
    var ext = "txt";
    var match = defaultFilename.match(/\.([a-zA-Z0-9]+)$/);
    if (match) {
        ext = match[1].toLowerCase();
    } else {
        defaultFilename += ".txt";
    }


    // 2. Try using the Native Windows Save As Dialog (File System Access API)
    if ('showSaveFilePicker' in window) {
        try {
            var options = {
                suggestedName: defaultFilename,
                types: [{
                    description: 'Text/Code Files',
                    accept: { 'text/plain': ['.' + ext, '.txt', '.js', '.css', '.html', '.aspx', '.json'] }
                }]
            };


            // Triggers standard Windows File Explorer "Save As" Dialog
            var handle = await window.showSaveFilePicker(options);
            var writable = await handle.createWritable();
            await writable.write(rawText);
            await writable.close();
            return; // Successfully saved via native Windows dialog
        } catch (err) {
            // User cancelled the Windows dialog
            if (err.name === 'AbortError') {
                return; 
            }
            console.warn('Native save dialog failed or rejected, falling back to download:', err);
        }
    }


    // 3. Fallback for legacy browsers or non-HTTPS contexts
    var fileName = prompt("Save file as:", defaultFilename);
    if (!fileName) return;


    var blob = new Blob([rawText], { type: "text/plain;charset=utf-8" });
    var link = document.createElement("a");
    link.href = URL.createObjectURL(blob);
    link.download = fileName;
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    URL.revokeObjectURL(link.href);
}
