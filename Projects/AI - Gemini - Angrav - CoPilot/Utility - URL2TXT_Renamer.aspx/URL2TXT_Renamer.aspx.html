    <script type="text/javascript">
        const dropZone = document.getElementById('drop-zone');
        const txtCurrent = document.getElementById('<%= txtCurrent.ClientID %>');
        const txtConverted = document.getElementById('<%= txtConverted.ClientID %>');


        ['dragenter', 'dragover', 'dragleave', 'drop'].forEach(eventName => {
            dropZone.addEventListener(eventName, e => {
                e.preventDefault();
                e.stopPropagation();
            }, false);
        });


        ['dragenter', 'dragover'].forEach(eventName => {
            dropZone.addEventListener(eventName, () => {
                dropZone.classList.add('hover');
            }, false);
        });


        ['dragleave', 'drop'].forEach(eventName => {
            dropZone.addEventListener(eventName, () => {
                dropZone.classList.remove('hover');
            }, false);
        });


        dropZone.addEventListener('drop', e => {
            const files = e.dataTransfer.files;
            let currentContent = txtCurrent.value;
            
            for (let i = 0; i < files.length; i++) {
                let name = files[i].name;
                if (name.toLowerCase().endsWith('.url')) {
                    name = name.substring(0, name.length - 4);
                }
                currentContent += (currentContent ? '\n' : '') + name;
            }
            txtCurrent.value = currentContent;
        });


function convertFileNames() {
            const rawText = txtCurrent.value;
            if (!rawText.trim()) return;


            const lines = rawText.split(/\r?\n/);
            const processedLines = lines.map(line => {
                let s = line;


                // Strip out a leading number index wrapped in parentheses, like "(1) " or "( 20 )"
                s = s.replace(/^\(\s*\d+\s*\)\s*/g, '');


                // Remove the word "YouTube" completely, ignoring uppercase/lowercase differences
                s = s.replace(/YouTube/gi, '');


                // Remove the word "Manhwa" completely, ignoring uppercase/lowercase differences
                s = s.replace(/Manhwa/gi, '');


                // Remove the word "Recap" completely, ignoring uppercase/lowercase differences
                s = s.replace(/Recap/gi, '');


                // Convert any pound/hashtag sign (#) to the literal word "Number"
                s = s.replace(/#/g, 'Number');


                // Convert any ampersand symbol (&) to the literal word "And"
                s = s.replace(/&/g, 'And');


                // Replace any long chain of three or more consecutive periods (...) with a single hyphen
                s = s.replace(/\.{3,}/g, '-');


                // Strip out everything except normal letters, numbers, spaces, periods, hyphens, and parentheses
                s = s.replace(/[^\w\s\.\-\(\)]/g, '');


                // Capitalize the first letter of every single word and lowercase the rest (Title Case formatting)
                s = s.replace(/\b([a-zA-Z])([a-zA-Z0-9]*)\b/g, function(match, first, rest) {
                    return first.toUpperCase() + rest.toLowerCase();
                });


                // Strip out exclamation points entirely
                s = s.replace(/!/g, '');


                // Insert a clean space before the word "And" if it is stuck to the end of another word
                s = s.replace(/([^\s])(And)\b/g, '$1 $2');


                // Insert a clean space before the word "Number" if it is stuck to the end of another word
                s = s.replace(/([^\s])(Number)\b/g, '$1 $2');


                // Trim away any leftover trailing periods or hyphens from the end of the text string
                s = s.replace(/[\.\-]+$/g, '');
                s = s.trim();


                // Strip off a trailing hyphen if it has whitespace before it at the end of the line
                s = s.replace(/\s+\-+$/g, '');


                // Double check and clean up any trailing periods or hyphens left over after the previous space adjustments
                s = s.replace(/[\.\-]+$/g, '');
                s = s.trim();


                return s;
            });


            txtConverted.value = processedLines.join('\n');
        }
        
        const crcTable = [];
        for (let n = 0; n < 256; n++) {
            let c = n;
            for (let k = 0; k < 8; k++) {
                c = ((c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1));
            }
            crcTable[n] = c;
        }


        async function downloadZipFile() {
            const rawNames = txtConverted.value;
            if (!rawNames.trim()) return;


            const fileNames = rawNames.split('\n').map(n => n.trim()).filter(n => n.length > 0);
            if (fileNames.length === 0) return;


            let zipParts = [];
            let centralDirectoryParts = [];
            let offset = 0;


            for (let name of fileNames) {
                if (!name.toLowerCase().endsWith('.txt')) {
                    name += '.txt';
                }
                const nameBytes = new TextEncoder().encode(name);
                const nameLength = nameBytes.length;


                const lfh = new ArrayBuffer(30);
                const lfhView = new DataView(lfh);
                lfhView.setUint32(0, 0x04034b50, true); 
                lfhView.setUint16(4, 10, true);         
                lfhView.setUint16(6, 0, true);          
                lfhView.setUint16(8, 0, true);          
                lfhView.setUint16(10, 0x3600, true);    
                lfhView.setUint16(12, 0x54B2, true);    
                lfhView.setUint32(14, 0, true);         
                lfhView.setUint32(18, 0, true);         
                lfhView.setUint32(22, 0, true);         
                lfhView.setUint16(26, nameLength, true);
                lfhView.setUint16(28, 0, true);         


                zipParts.push(lfh, nameBytes);


                const cdfh = new ArrayBuffer(46);
                const cdfhView = new DataView(cdfh);
                cdfhView.setUint32(0, 0x02014b50, true); 
                cdfhView.setUint16(4, 20, true);         
                cdfhView.setUint16(6, 10, true);         
                cdfhView.setUint16(8, 0, true);          
                cdfhView.setUint16(10, 0, true);         
                cdfhView.setUint16(12, 0x3600, true);    
                cdfhView.setUint16(14, 0x54B2, true);    
                cdfhView.setUint32(16, 0, true);         
                cdfhView.setUint32(20, 0, true);         
                cdfhView.setUint32(24, 0, true);         
                cdfhView.setUint16(28, nameLength, true);
                centralDirectoryParts.push(cdfh, nameBytes);
                offset += 30 + nameLength; 
            }


            const cdOffset = offset;
            let cdSize = 0;
            centralDirectoryParts.forEach(p => cdSize += p.byteLength);


            const eocd = new ArrayBuffer(22);
            const eocdView = new DataView(eocd);
            eocdView.setUint32(0, 0x06054b50, true); 
            eocdView.setUint16(4, 0, true);          
            eocdView.setUint16(6, 0, true);          
            eocdView.setUint16(8, fileNames.length, true);  
            eocdView.setUint16(10, fileNames.length, true); 
            eocdView.setUint32(12, cdSize, true);    
            eocdView.setUint32(16, cdOffset, true);  
            eocdView.setUint16(20, 0, true);          


            const finalBlob = new Blob([...zipParts, ...centralDirectoryParts, eocd], { type: 'application/zip' });


            if ('showSaveFilePicker' in window) {
                try {
                    const handle = await window.showSaveFilePicker({
                        suggestedName: 'Converted_Text_Files.zip',
                        types: [{
                            description: 'ZIP Archive',
                            accept: { 'application/zip': ['.zip'] }
                        }]
                    });
                    const writable = await handle.createWritable();
                    await writable.write(finalBlob);
                    await writable.close();
                } catch (err) {
                    console.log('User closed or cancelled save window');
                }
            } else {
                const url = URL.createObjectURL(finalBlob);
                const a = document.createElement('a');
                a.href = url;
                a.download = 'Converted_Text_Files.zip';
                a.click();
                URL.revokeObjectURL(url);
            }
        }
