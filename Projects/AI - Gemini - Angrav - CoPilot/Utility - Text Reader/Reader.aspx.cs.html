using System;
using System.Collections.Generic;
using System.IO;
using System.Text;
using System.Web;
using System.Web.Script.Serialization;
using System.Web.UI;
using System.Web.UI.WebControls;




    public partial class Reader : Page
    {
        private string JsonFilePath
        {
            get { return Server.MapPath("~/Reader.json"); }
        }


        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                BindSavedEntries();
            }
        }


        protected void btnSave_Click(object sender, EventArgs e)
        {
            string voiceName = hfSelectedVoice.Value;
            string textName = txtTextName.Text.Trim();
            string rawContent = txtContent.Text;


            if (string.IsNullOrEmpty(textName))
            {
                textName = "Untitled Reader";
            }


            var entries = LoadEntriesRaw();


            var newEntry = new ReaderEntry
            {
                VoiceName = EncodeBase64(voiceName),
                TextName = EncodeBase64(textName),
                TextContent = EncodeBase64(rawContent)
            };


            entries.Add(newEntry);
            SaveEntriesRaw(entries);


            txtTextName.Text = string.Empty;
            BindSavedEntries();
        }


        protected void btnDelete_Click(object sender, EventArgs e)
        {
            int index;
            if (int.TryParse(hfDeleteIndex.Value, out index))
            {
                var entries = LoadEntriesRaw();
                if (index >= 0 && index < entries.Count)
                {
                    entries.RemoveAt(index);
                    SaveEntriesRaw(entries);
                    BindSavedEntries();
                }
            }
        }


        private void BindSavedEntries()
        {
            var rawEntries = LoadEntriesRaw();
            var decodedList = new List<ReaderEntryDisplay>();


            for (int i = 0; i < rawEntries.Count; i++)
            {
                var item = rawEntries[i];
                decodedList.Add(new ReaderEntryDisplay
                {
                    VoiceName = DecodeBase64(item.VoiceName),
                    TextName = DecodeBase64(item.TextName),
                    TextContent = DecodeBase64(item.TextContent),
                    Index = i
                });
            }


            rptSavedEntries.DataSource = decodedList;
            rptSavedEntries.DataBind();
        }


        protected void rptSavedEntries_ItemDataBound(object sender, RepeaterItemEventArgs e)
        {
            if (e.Item.ItemType != ListItemType.Item && e.Item.ItemType != ListItemType.AlternatingItem)
            {
                return;
            }


            var entry = (ReaderEntryDisplay)e.Item.DataItem;
            if (entry == null)
            {
                return;
            }


            var btnRead = (Button)e.Item.FindControl("btnReadEntry");
            var btnDelete = (Button)e.Item.FindControl("btnDeleteEntry");


            if (btnRead != null)
            {
                btnRead.Text = entry.TextName;
                btnRead.ToolTip = entry.ReadTitle;
                btnRead.Attributes["title"] = entry.ReadTitle;
                btnRead.OnClientClick = entry.ReadCommand;
            }


            if (btnDelete != null)
            {
                btnDelete.Attributes["title"] = "Delete this entry";
                btnDelete.OnClientClick = entry.DeleteCommand;
            }
        }


        private List<ReaderEntry> LoadEntriesRaw()
        {
            if (!File.Exists(JsonFilePath))
            {
                return new List<ReaderEntry>();
            }


            try
            {
                string json = File.ReadAllText(JsonFilePath);
                JavaScriptSerializer js = new JavaScriptSerializer();
                var result = js.Deserialize<List<ReaderEntry>>(json);
                return result ?? new List<ReaderEntry>();
            }
            catch
            {
                return new List<ReaderEntry>();
            }
        }


        private void SaveEntriesRaw(List<ReaderEntry> entries)
        {
            JavaScriptSerializer js = new JavaScriptSerializer();
            string json = js.Serialize(entries);
            File.WriteAllText(JsonFilePath, json);
        }


        private string EncodeBase64(string value)
        {
            if (string.IsNullOrEmpty(value)) return string.Empty;
            byte[] bytes = Encoding.UTF8.GetBytes(value);
            return Convert.ToBase64String(bytes);
        }


        private string DecodeBase64(string base64Value)
        {
            if (string.IsNullOrEmpty(base64Value)) return string.Empty;
            try
            {
                byte[] bytes = Convert.FromBase64String(base64Value);
                return Encoding.UTF8.GetString(bytes);
            }
            catch
            {
                return base64Value; // Fallback if string was unencoded
            }
        }


        public class ReaderEntry
        {
            public string VoiceName { get; set; }
            public string TextName { get; set; }
            public string TextContent { get; set; }
        }


        public class ReaderEntryDisplay
        {
            public string VoiceName { get; set; }
            public string TextName { get; set; }
            public string TextContent { get; set; }
            public int Index { get; set; }


            public string ReadTitle
            {
                get { return "Read by " + VoiceName; }
            }


            public string ReadCommand
            {
                get
                {
                    return "readSavedText(\"" + HttpUtility.JavaScriptStringEncode(TextContent) + "\", \"" + HttpUtility.JavaScriptStringEncode(VoiceName) + "\")";
                }
            }


            public string DeleteCommand
            {
                get { return "deleteEntry(" + Index + ")"; }
            }
        }
