    <script>
        let synth = window.speechSynthesis;
        let voices = [];


        function populateVoices() {
            voices = synth.getVoices();
            const container = document.getElementById('voiceButtonsContainer');
            container.innerHTML = '';


            if (voices.length === 0) {
                container.innerHTML = '<em>No browser voices found.</em>';
                return;
            }


            voices.forEach((voice, index) => {
                let btn = document.createElement('button');
                btn.type = 'button';
                btn.className = 'btn';
                btn.style.backgroundColor = '#6c757d';
                btn.style.borderColor = '#6c757d';
                btn.innerText = voice.name + ' (' + voice.lang + ')';
                
                btn.onclick = function () {
                    document.getElementById('<%= hfSelectedVoice.ClientID %>').value = voice.name;
                    document.getElementById('<%= lblSelectedVoice.ClientID %>').innerText = voice.name;
                    speakText(document.getElementById('<%= txtContent.ClientID %>').value, voice.name);
                };
                
                container.appendChild(btn);
            });


            // Set initial voice selection if empty
            if (!document.getElementById('<%= hfSelectedVoice.ClientID %>').value && voices.length > 0) {
                document.getElementById('<%= hfSelectedVoice.ClientID %>').value = voices[0].name;
                document.getElementById('<%= lblSelectedVoice.ClientID %>').innerText = voices[0].name;
            }
        }


        if (speechSynthesis.onvoiceschanged !== undefined) {
            speechSynthesis.onvoiceschanged = populateVoices;
        }
        window.onload = populateVoices;


        function speakText(text, voiceName) {
            if (!text) return;
            synth.cancel(); // Stop any active speech


            let utterance = new SpeechSynthesisUtterance(text);
            if (voiceName) {
                let foundVoice = voices.find(v => v.name === voiceName);
                if (foundVoice) utterance.voice = foundVoice;
            }
            synth.speak(utterance);
        }


        function stopSpeech() {
            synth.cancel();
        }


        function readSavedText(text, voiceName) {
            document.getElementById('<%= txtContent.ClientID %>').value = text;
            document.getElementById('<%= hfSelectedVoice.ClientID %>').value = voiceName;
            document.getElementById('<%= lblSelectedVoice.ClientID %>').innerText = voiceName;
            speakText(text, voiceName);
        }


        function deleteEntry(index) {
            if (confirm('Delete this saved entry?')) {
                document.getElementById('<%= hfDeleteIndex.ClientID %>').value = index;
                document.getElementById('<%= btnDeleteTrigger.ClientID %>').click();
            }
        }
    


</script>
