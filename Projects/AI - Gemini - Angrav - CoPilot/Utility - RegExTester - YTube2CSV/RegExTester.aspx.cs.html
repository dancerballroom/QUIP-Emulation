using System;
using System.IO; 
using System.Text;
using System.Web;
using System.Web.UI;
using System.Web.UI.WebControls;
using System.Text.RegularExpressions;
using System.Web.Script.Serialization;
using System.Collections.Generic;
using System.Net;


public partial class RegExTester : System.Web.UI.Page
{
protected void Page_Load(object sender, EventArgs e)
{
    RenderSourceHistoryLinks();
    RenderResultHistoryLinks();


    // Move this completely outside the If/Else blocks!
    if (Session["LastFileName"] != null)
    {
        hfLastFileName.Value = Session["LastFileName"].ToString();
    }


    if (!IsPostBack)
    {
        LoadRegExPresets();


        if (Session["RegExOriginal"] != null)
        {
            txtOriginal.Text = Session["RegExOriginal"].ToString();
            Session.Remove("RegExOriginal"); 
        }


        if (Session["RegExResult"] != null)
        {
            txtResult.Text = Session["RegExResult"].ToString();
            Session.Remove("RegExResult"); 
        }
    }
    else
    {
        LoadRegExPresets();
    }
}




    private void RenderSourceHistoryLinks()
    {
        if (phHistoryLinks == null) return;
        phHistoryLinks.Controls.Clear();
        var history = Session["SourceHistory"] as List<string>;


        if (history != null && history.Count > 0)
        {
            for (int i = 0; i < history.Count; i++)
            {
                int index = i;
                LinkButton lnk = new LinkButton();
                lnk.ID = "lnkSrcHistory_" + index;
                lnk.Text = (index + 1).ToString();
                lnk.Style.Add("color", "#007bff");
                lnk.Style.Add("text-decoration", "none");
                
                lnk.Click += (s, ev) => {
                    txtOriginal.Text = history[index];
                };


                phHistoryLinks.Controls.Add(lnk);


                if (index < history.Count - 1)
                {
                    phHistoryLinks.Controls.Add(new Literal { Text = ", " });
                }
            }
        }
    }


    private void RenderResultHistoryLinks()

        if (phResultLinks == null) return;
        phResultLinks.Controls.Clear();
        var history = Session["ResultHistory"] as List<string>;


        if (history != null && history.Count > 0)
        {
            for (int i = 0; i < history.Count; i++)
            {
                int index = i;
                LinkButton lnk = new LinkButton();
                lnk.ID = "lnkResHistory_" + index;
                lnk.Text = (index + 1).ToString();
                lnk.Style.Add("color", "#007bff");
                lnk.Style.Add("text-decoration", "none");
                
                lnk.Click += (s, ev) => {
                    txtResult.Text = history[index];
                };


                phResultLinks.Controls.Add(lnk);


                if (index < history.Count - 1)
                {
                    phResultLinks.Controls.Add(new Literal { Text = ", " });
                }
            }
        }
    }


    protected void btnSaveSource_Click(object sender, EventArgs e)
    {
        if (string.IsNullOrWhiteSpace(txtOriginal.Text)) return;


        var history = Session["SourceHistory"] as List<string> ?? new List<string>();
        history.Add(txtOriginal.Text);
        Session["SourceHistory"] = history;


        // Retain file info if present during an immediate postback state
        if (!string.IsNullOrEmpty(hfLastFileName.Value))
        {
            Session["LastFileName"] = hfLastFileName.Value;
        }


        RenderSourceHistoryLinks();
    }


    protected void btnSaveResult_Click(object sender, EventArgs e)
    {
        if (string.IsNullOrWhiteSpace(txtResult.Text)) return;


        var history = Session["ResultHistory"] as List<string> ?? new List<string>();
        history.Add(txtResult.Text);
        Session["ResultHistory"] = history;


        // Retain file info if present during an immediate postback state
        if (!string.IsNullOrEmpty(hfLastFileName.Value))
        {
            Session["LastFileName"] = hfLastFileName.Value;
        }


        RenderResultHistoryLinks();
    }


    protected void btnClearSession_Click(object sender, EventArgs e)
    {
        Session.Remove("SourceHistory");
        Session.Remove("ResultHistory");
        Session.Remove("LastFileName");
        txtOriginal.Text = "";
        txtResult.Text = "";


        Response.Redirect(Request.Url.AbsoluteUri);
    }


    private void LoadRegExPresets()
    {
        try
        {
            string filePath = Server.MapPath("~/RegExTester.json");
            if (File.Exists(filePath))
            {
                string jsonString = File.ReadAllText(filePath);
                JavaScriptSerializer serializer = new JavaScriptSerializer();
                dynamic presets = serializer.Deserialize<dynamic>(jsonString);


                string currentSelection = ddlPresets.SelectedValue;


                ddlPresets.Items.Clear();
                ddlPresets.Items.Add(new ListItem("-- YouTube Transcript 2 SRT RegEx --", ""));


                foreach (var preset in presets)
                {
                    ListItem item = new ListItem(preset["Name"].ToString(), preset["Name"].ToString());
                    item.Attributes.Add("data-find", preset["Find"].ToString());
                    item.Attributes.Add("data-replace", preset["Replace"].ToString());
                    ddlPresets.Items.Add(item);
                }


                if (!string.IsNullOrEmpty(currentSelection) && ddlPresets.Items.FindByValue(currentSelection) != null)
                {
                    ddlPresets.SelectedValue = currentSelection;
                }
            }
        }
        catch (Exception ex)
        {
            txtResult.Text = "Error loading JSON presets: " + ex.Message;
        }
    }


    protected void btnApply_Click(object sender, EventArgs e)
    {
        string pattern = txtFind.Text;
        string replaceWith = txtReplace.Text;
        string inputData = txtOriginal.Text;


        if (string.IsNullOrWhiteSpace(pattern))
        {
            txtFind.Text = "[\\s\\S]*";
            txtReplace.Text = "$&";
            pattern = txtFind.Text;
            replaceWith = txtReplace.Text;
        }


        try
        {
            Regex regex = new Regex(pattern, RegexOptions.Multiline);
            string finalResultText = "";


            if (ddlPresets.SelectedValue.Contains("Isolate") || replaceWith == "$&")
            {
                MatchCollection matches = regex.Matches(inputData);
                StringBuilder extractedLines = new StringBuilder();


                foreach (Match match in matches)
                {
                    if (match.Success)
                    {
                        extractedLines.AppendLine(match.Value.Trim());
                    }
                }
                finalResultText = extractedLines.ToString();
            }
            else
            {
                finalResultText = regex.Replace(inputData, replaceWith);
            }


            // Persist the custom drop file name across the redirect loop
            if (!string.IsNullOrEmpty(hfLastFileName.Value))
            {
                Session["LastFileName"] = hfLastFileName.Value;
            }


            Session["RegExOriginal"] = inputData;
            Session["RegExResult"] = finalResultText;


            Response.Redirect(Request.Url.AbsoluteUri);
        }
        catch (ArgumentException ex)
        {
            txtResult.Text = "RegEx Parsing Error: " + ex.Message;
        }
    }
