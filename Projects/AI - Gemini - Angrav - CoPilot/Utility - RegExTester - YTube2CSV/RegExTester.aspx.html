        function onPresetChange(ddl) {
            if (!ddl) return;
            var selectedOption = ddl.options[ddl.selectedIndex];
            if (!selectedOption) return;


            var findVal = selectedOption.getAttribute("data-find");
            var replaceVal = selectedOption.getAttribute("data-replace");
            
            var findBox = document.getElementById('<%= txtFind.ClientID %>');
            var replaceBox = document.getElementById('<%= txtReplace.ClientID %>');
            
            if (findBox) findBox.value = findVal || '';
            if (replaceBox) replaceBox.value = replaceVal || '';
        }

        }

        function mergeBoxesColumnMode() {
            var box1 = document.getElementById('<%= txtBox1.ClientID %>');
            var box2 = document.getElementById('<%= txtBox2.ClientID %>');
            var box3 = document.getElementById('<%= txtBox3.ClientID %>');
            
            if (!box1 || !box2 || !box3) return;


            var lines1 = box1.value.replace(/\r\n/g, "\n").split("\n");
            var lines2 = box2.value.replace(/\r\n/g, "\n").split("\n");


            var maxLines = Math.max(lines1.length, lines2.length);
            var mergedLines = [];


            for (var i = 0; i < maxLines; i++) {
                var part1 = lines1[i] !== undefined ? lines1[i] : "";
                var part2 = lines2[i] !== undefined ? lines2[i] : "";


                if (part1 !== "" || part2 !== "" || i < maxLines - 1) {
                    if (part1 !== "" && part2 !== "") {
                        mergedLines.push(part1 + " " + part2);
                    } else {
                        mergedLines.push(part1 + part2);
                    }
                }
            }
            box3.value = mergedLines.join("\n");
            refreshLineNumbers(box3, document.getElementById('lineNumbers_txtBox3'));
        }




async function saveBox3ToFile() {
    var txtBox3 = document.getElementById('<%= txtBox3.ClientID %>');
    if (!txtBox3) return false;


    var currentBox3Text = txtBox3.value || '';
    if (currentBox3Text.trim() === '') return false;


    var saveButton = document.getElementById('btnSaveBox3File');
    var dropZone = document.getElementById('txtFileDropZone');
    var hiddenField = document.getElementById('hfLastFileName');
    var suggestedName = (saveButton && saveButton.getAttribute('data-last-file-name'))
        || (hiddenField && hiddenField.value)
        || (dropZone && (dropZone.getAttribute('data-last-file-name') || dropZone.dataset.lastFileName))
        || (typeof lastDroppedTextFileName !== 'undefined' ? lastDroppedTextFileName : null)
        || 'box3.txt';
    if (!suggestedName) {
        suggestedName = 'box3.txt';
    }
    if (!/\.txt$/i.test(suggestedName)) {
        suggestedName += '.txt';
    }


    var blob = new Blob([currentBox3Text], { type: 'text/plain;charset=utf-8' });


    try {
        if ('showSaveFilePicker' in window) {
            var handle = await window.showSaveFilePicker({
                suggestedName: suggestedName,
                types: [{
                    description: 'Text Files',
                    accept: { 'text/plain': ['.txt'] }
                }]
            });
            var writable = await handle.createWritable();
            await writable.write(blob);
            await writable.close();
        } else {
            var url = window.URL.createObjectURL(blob);
            var link = document.createElement('a');
            link.href = url;
            link.download = suggestedName;
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            window.URL.revokeObjectURL(url);
        }
    } catch (err) {
        if (err && err.name !== 'AbortError') {
            console.error('Unable to save file', err);
        }
    }


    return false;
}


function copyBox3ToClipboard() {
    var txtBox3 = document.getElementById('<%= txtBox3.ClientID %>');
    if (!txtBox3 || txtBox3.value.trim() === '') return false;
    
    // Create a temporary hidden, editable textarea
    var tempTextArea = document.createElement("textarea");
    tempTextArea.value = txtBox3.value;
    
    // Position completely off-screen so the user doesn't see it layout flash
    Object.assign(tempTextArea.style, {
        position: "fixed",
        top: "0",
        left: "-9999px",
        width: "2em",
        height: "2em",
        padding: "0",
        border: "none",
        outline: "none",
        boxShadow: "none",
        background: "transparent"
    });
    
    document.body.appendChild(tempTextArea);
    
    // Explicitly focus and highlight the entire text structure
    tempTextArea.focus();
    tempTextArea.select();
    tempTextArea.setSelectionRange(0, tempTextArea.value.length);
    
    var success = false;
    try {
        // execCommand bypasses the user gesture constraint for macro execution loops
        success = document.execCommand('copy');
    } catch (err) {
        success = false;
    }
    
    // Clean up the DOM element
    document.body.removeChild(tempTextArea);


    // If execCommand falls behind, hit the modern API as a dual fallback layered layer
    if (!success && navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(txtBox3.value).then(function() {
            success = true;
        }).catch(function() {
            success = false;
            console.warn("Clipboard access denied.");
        });
    }


    if (success) {
        var status = document.getElementById("copyStatus");
        if (status) {
            status.innerHTML = "Box 3 Copied!";
            status.style.display = "inline";
            setTimeout(function() { 
                status.style.display = "none"; 
                status.innerHTML = "Copied!"; 
            }, 2000);
        }
    }
    return false;
}

        function toggleEditorFullscreen(editorShellId, button) {
            var shell = document.getElementById(editorShellId);
            if (!shell) return;


            var isFullscreen = shell.classList.toggle('fullscreen');
            if (button) {
                button.textContent = isFullscreen ? '×' : '⬍';
                button.title = isFullscreen ? 'Restore' : 'Full Screen';
                button.classList.toggle('close-btn', isFullscreen);
            }


            if (isFullscreen) {
                var overlayContainer = document.createElement('div');
                overlayContainer.style.position = 'fixed';
                overlayContainer.style.top = '16px';
                overlayContainer.style.right = '16px';
                overlayContainer.style.zIndex = '99999';
                overlayContainer.style.display = 'flex';
                overlayContainer.style.gap = '8px';


                var copyOverlay = document.createElement('button');
                copyOverlay.type = 'button';
                copyOverlay.className = 'editor-toggle close-btn';
                copyOverlay.textContent = '📋';
                copyOverlay.title = 'Copy All';
                copyOverlay.onclick = function () {
                    var targetTextarea = shell.querySelector('textarea');
                    if (targetTextarea) {
                        targetTextarea.select();
                        targetTextarea.setSelectionRange(0, targetTextarea.value.length);
                        document.execCommand('copy');
                    }
                };


                var closeOverlay = document.createElement('button');
                closeOverlay.type = 'button';
                closeOverlay.className = 'editor-toggle close-btn';
                closeOverlay.textContent = '×';
                closeOverlay.title = 'Restore';
                closeOverlay.onclick = function () {
                    toggleEditorFullscreen(editorShellId, button);
                    if (overlayContainer.parentNode) {
                        overlayContainer.parentNode.removeChild(overlayContainer);
                    }
                };


                overlayContainer.appendChild(copyOverlay);
                overlayContainer.appendChild(closeOverlay);
                document.body.appendChild(overlayContainer);
            } else {
                var existingOverlay = document.body.querySelector('div[style*="position: fixed"][style*="top: 16px"][style*="right: 16px"]');
                if (existingOverlay) {
                    existingOverlay.remove();
                }
            }


            document.body.style.overflow = isFullscreen ? 'hidden' : '';
        }


        document.addEventListener('keydown', function (event) {
            if (event.key === 'Escape') {
                var activeShell = document.querySelector('.editor-shell.fullscreen');
                if (activeShell) {
                    toggleEditorFullscreen(activeShell.id, activeShell.querySelector('.editor-toggle'));
                }
            }
        });

        var lastDroppedTextFileName = null;
        var droppedTextFileQueue = [];
        var textFileQueueStorageKey = 'SRT_Text_File_Queue';


        function syncProcessingFileLabel() {
            var processingLabel = document.getElementById('txtProcessingFileLabel');
            if (!processingLabel) return;


            var txtOriginal = document.getElementById('<%= txtOriginal.ClientID %>');
            var sourceText = txtOriginal ? (txtOriginal.value || '').trim() : '';


            if (!sourceText) {
                processingLabel.textContent = 'Text File Processing:';
                return;
            }


            var fileName = localStorage.getItem('SRT_Last_File_Name') || lastDroppedTextFileName;
            processingLabel.textContent = fileName ? ('Text File Processing: ' + fileName) : 'Text File Processing:';
        }


        function persistTextFileQueue() {
            try {
                localStorage.setItem(textFileQueueStorageKey, JSON.stringify(droppedTextFileQueue));
            } catch (err) {
                console.warn('Unable to persist text file queue.', err);
            }
        }


        function restoreTextFileQueue() {
            var rawQueue = localStorage.getItem(textFileQueueStorageKey);
            if (!rawQueue) {
                droppedTextFileQueue = [];
                return;
            }


            try {
                var parsedQueue = JSON.parse(rawQueue);
                if (!Array.isArray(parsedQueue)) {
                    droppedTextFileQueue = [];
                    return;
                }


                droppedTextFileQueue = parsedQueue.filter(function (item) {
                    return item && typeof item.name === 'string' && typeof item.content === 'string';
                });
            } catch (err) {
                droppedTextFileQueue = [];
                console.warn('Unable to restore text file queue.', err);
            }
        }


        function renderTextFileQueue() {
            var queueBox = document.getElementById('txtProcessTextFiles');
            if (!queueBox) return;
            queueBox.value = droppedTextFileQueue.map(function (item) {
                return item.name;
            }).join('\n');
            persistTextFileQueue();
            updateStepUiByQueueState();
        }


        function queueDroppedTextFiles(fileList) {
            var txtFiles = Array.prototype.slice.call(fileList || []).filter(function (file) {
                return file && /\.txt$/i.test(file.name);
            });


            if (!txtFiles.length) {
                alert('Please drop one or more .txt files.');
                return;
            }


            Promise.all(txtFiles.map(function (file) {
                return new Promise(function (resolve, reject) {
                    var reader = new FileReader();
                    reader.onload = function (e) {
                        resolve({
                            name: file.name,
                            content: e.target.result || ''
                        });
                    };
                    reader.onerror = function () {
                        reject(file.name);
                    };
                    reader.readAsText(file);
                });
            })).then(function (loadedFiles) {
                droppedTextFileQueue = droppedTextFileQueue.concat(loadedFiles);
                renderTextFileQueue();
            }).catch(function () {
                alert('Unable to read one or more selected text files.');
            });
        }


        function loadNextQueuedTextFile() {
            if (!droppedTextFileQueue.length) {
                showToast('Please Drag / Drop Files First');
                return false;
            }


            var nextFile = droppedTextFileQueue.shift();
            renderTextFileQueue();


            var txtOriginal = document.getElementById('<%= txtOriginal.ClientID %>');
            var gutter = document.getElementById('lineNumbers_txtOriginal');
            if (txtOriginal) {
                txtOriginal.value = nextFile.content || '';
                if (gutter) {
                    refreshLineNumbers(txtOriginal, gutter);
                }
            }


            lastDroppedTextFileName = nextFile.name;
            localStorage.setItem('SRT_Last_File_Name', nextFile.name);


            var dropZone = document.getElementById('txtFileDropZone');
            if (dropZone) {
                dropZone.setAttribute('data-last-file-name', nextFile.name);
                dropZone.dataset.lastFileName = nextFile.name;
            }


            var hfName = document.getElementById('hfLastFileName');
            if (hfName) {
                hfName.value = nextFile.name;
            }


            var saveButton = document.getElementById('btnSaveBox3File');
            if (saveButton) {
                saveButton.setAttribute('data-last-file-name', nextFile.name);
            }


            syncProcessingFileLabel();


            return false;
        }


        function clearProcessTextFileQueue() {
            droppedTextFileQueue = [];
            renderTextFileQueue();
            return false;
        }


        function attachTextFileDropZone() {
            var dropZone = document.getElementById('txtFileDropZone');
            if (!dropZone) return;


            function preventDefault(e) {
                e.preventDefault();
                e.stopPropagation();
            }


            ['dragenter', 'dragover', 'dragleave', 'drop'].forEach(function (eventName) {
                dropZone.addEventListener(eventName, preventDefault);
            });


            dropZone.addEventListener('dragenter', function () {
                dropZone.classList.add('drag-over');
            });


            dropZone.addEventListener('dragover', function () {
                dropZone.classList.add('drag-over');
            });


            dropZone.addEventListener('dragleave', function (e) {
                if (!dropZone.contains(e.relatedTarget)) {
                    dropZone.classList.remove('drag-over');
                }
            });


            dropZone.addEventListener('drop', function (e) {
                dropZone.classList.remove('drag-over');
                var files = e.dataTransfer && e.dataTransfer.files;
                if (!files || !files.length) return;
                queueDroppedTextFiles(files);
            });
        }


        window.addEventListener('load', function () {

                }
                var restoredSaveButton = document.getElementById('btnSaveBox3File');
                if (restoredSaveButton) {
                    restoredSaveButton.setAttribute('data-last-file-name', restoredFileName);
                }
                lastDroppedTextFileName = restoredFileName;
            }


            attachTextFileDropZone();
            attachLineNumbers(document.getElementById('<%= txtOriginal.ClientID %>'), document.getElementById('lineNumbers_txtOriginal'));
            attachLineNumbers(document.getElementById('<%= txtResult.ClientID %>'), document.getElementById('lineNumbers_txtResult'));

            if (localStorage.getItem("SRT_Macro_Active") === "true") {
                executeNextMacroStep();
            }
        });
