            function buildSvgSearchCandidates(rawQuery) {
                const original = (rawQuery || '').trim();
                const normalized = original
                    .toLowerCase()
                    .replace(/[^a-z0-9\s-]/g, ' ')
                    .replace(/\s+/g, ' ')
                    .trim();


                const words = normalized ? normalized.split(' ').filter(Boolean) : [];
                const candidates = [];


                function addCandidate(value) {
                    const candidate = (value || '').trim();
                    if (!candidate) return;
                    if (!candidates.includes(candidate)) {
                        candidates.push(candidate);
                    }
                }


                addCandidate(original);
                addCandidate(normalized);


                if (words.length > 1) {
                    addCandidate(words.join('-'));
                    addCandidate(words.join('_'));
                }


                for (let end = words.length - 1; end > 0; end--) {
                    const head = words.slice(0, end).join(' ');
                    addCandidate(head);
                    addCandidate(head.replace(/\s+/g, '-'));
                }


                for (let start = 1; start < words.length; start++) {
                    const tail = words.slice(start).join(' ');
                    addCandidate(tail);
                    addCandidate(tail.replace(/\s+/g, '-'));
                }


                words.forEach(word => addCandidate(word));
                return candidates;
            }

            function scoreIconColorfulness(iconName) {

            }


            async function findSvgIconOptionsForCandidates(candidates, maxResults) {
                const unique = new Set();
                const options = [];


                for (const candidate of candidates) {
                    const searchUrl = `https://api.iconify.design/search?query=${encodeURIComponent(candidate)}&limit=8`;
                    const searchResponse = await fetch(searchUrl);
                    if (!searchResponse.ok) {
                        continue;
                    }


                    const searchResult = await searchResponse.json();
                    const icons = Array.isArray(searchResult.icons) ? searchResult.icons : [];
                    const rankedIcons = icons
                        .map(icon => ({ icon, score: scoreIconColorfulness(icon) }))
                        .sort((a, b) => b.score - a.score)
                        .map(entry => entry.icon);


                    for (const icon of rankedIcons) {
                        if (unique.has(icon)) {
                            continue;
                        }


                        unique.add(icon);
                        options.push({ icon: icon, matchedQuery: candidate });
                        if (options.length >= maxResults) {
                            return options;
                        }
                    }
                }


                return options;
            }

            function appendSvgMarkupToCanvas(markup) {
                const svgRoot = parseSvgMarkup(markup);
                if (!svgRoot) {
                    return false;
                }


                const children = Array.from(svgRoot.childNodes)
                    .filter(node => node.nodeType === Node.ELEMENT_NODE)
                    .map(node => document.importNode(node, true));


                if (children.length === 0) {
                    return false;
                }


                children.forEach(node => shapesGroup.appendChild(node));
                return true;
            }

            function getSVGCoordinates(e) {
                const rect = canvas.getBoundingClientRect();
                const viewBox = canvas.viewBox.baseVal;
                const width = viewBox && viewBox.width ? viewBox.width : rect.width;
                const height = viewBox && viewBox.height ? viewBox.height : rect.height;
                const x = ((e.clientX - rect.left) / rect.width) * width;
                const y = ((e.clientY - rect.top) / rect.height) * height;


                return {
                    x: Math.round(x),
                    y: Math.round(y)
                };
            }


            function getEnclosingShapeForPoint(coords) {
                const candidateShapes = Array.from(shapesGroup.children)
                    .filter(el => el.tagName && el.tagName.toLowerCase() !== 'defs' && !el.hasAttribute('data-fill-layer'));


                for (let i = candidateShapes.length - 1; i >= 0; i--) {
                    const shape = candidateShapes[i];
                    const tag = shape.tagName.toLowerCase();
                    if (tag === 'text' || tag === 'line') continue;


                    const d = shape.getAttribute('d') || '';
                    if (tag === 'path' && d && !d.includes('Z')) continue;


                    const bbox = shape.getBBox ? shape.getBBox() : null;
                    if (!bbox) continue;


                    const inside = coords.x >= bbox.x && coords.x <= bbox.x + bbox.width && coords.y >= bbox.y && coords.y <= bbox.y + bbox.height;
                    if (inside) {
                        return shape;
                    }
                }


                return null;
            }


            function createFillShape(coords, fillColor) {
                const width = canvas.viewBox.baseVal.width || drawContainer.clientWidth || 800;
                const height = canvas.viewBox.baseVal.height || drawContainer.clientHeight || 600;
                const group = document.createElementNS('http://www.w3.org/2000/svg', 'g');
                group.setAttribute('data-fill-layer', 'true');


                const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
                rect.setAttribute('x', 0);
                rect.setAttribute('y', 0);
                rect.setAttribute('width', width);
                rect.setAttribute('height', height);
                rect.setAttribute('fill', fillColor);
                rect.setAttribute('stroke', 'none');
                group.appendChild(rect);


                const enclosingShape = getEnclosingShapeForPoint(coords);
                if (enclosingShape) {
                    const clipPath = document.createElementNS('http://www.w3.org/2000/svg', 'clipPath');
                    const clipId = `fillClip-${Date.now()}-${Math.random().toString(16).slice(2)}`;
                    clipPath.setAttribute('id', clipId);


                    const clone = enclosingShape.cloneNode(true);
                    clone.removeAttribute('class');
                    clone.removeAttribute('id');
                    clipPath.appendChild(clone);
                    fillDefs.appendChild(clipPath);
                    rect.setAttribute('clip-path', `url(#${clipId})`);
                }


                return group;
            }

            function rotateShape(elem, degrees) {
                const tag = elem.tagName.toLowerCase();
                if (tag !== 'path' && tag !== 'rect' && tag !== 'circle' && tag !== 'line' && tag !== 'text') {
                    return;
                }


                const currentTransform = elem.getAttribute('transform') || '';
                const rotationMatch = currentTransform.match(/rotate\(([-+]?\d*\.?\d+)/i);
                const currentRotation = rotationMatch ? parseFloat(rotationMatch[1]) : 0;
                const nextRotation = currentRotation + degrees;


                let centerX = 0;
                let centerY = 0;
                const bbox = elem.getBBox ? elem.getBBox() : null;
                if (bbox) {
                    centerX = bbox.x + bbox.width / 2;
                    centerY = bbox.y + bbox.height / 2;
                } else {
                    centerX = parseFloat(elem.getAttribute('cx')) || 0;
                    centerY = parseFloat(elem.getAttribute('cy')) || 0;
                    if (!centerX && !centerY) {
                        centerX = parseFloat(elem.getAttribute('x')) || 0;
                        centerY = parseFloat(elem.getAttribute('y')) || 0;
                    }
                }


                const rotationValue = `${nextRotation} ${centerX} ${centerY}`;


                if (currentTransform) {
                    const withoutRotation = currentTransform.replace(/rotate\([^)]*\)\s*/i, '').trim();
                    elem.setAttribute('transform', withoutRotation ? `rotate(${rotationValue}) ${withoutRotation}` : `rotate(${rotationValue})`);
                } else {
                    elem.setAttribute('transform', `rotate(${rotationValue})`);
                }
            }


            function resizeShape(elem, factor) {
                if (!elem) return;


                const tag = elem.tagName.toLowerCase();
                if (tag === 'defs' || tag === 'g') return;


                const bbox = elem.getBBox ? elem.getBBox() : null;
                const centerX = bbox ? bbox.x + bbox.width / 2 : (parseFloat(elem.getAttribute('cx')) || parseFloat(elem.getAttribute('x')) || 0);
                const centerY = bbox ? bbox.y + bbox.height / 2 : (parseFloat(elem.getAttribute('cy')) || parseFloat(elem.getAttribute('y')) || 0);
                const currentTransform = elem.getAttribute('transform') || '';
                const scaleTransform = `translate(${centerX} ${centerY}) scale(${factor}) translate(${ -centerX } ${ -centerY })`;


                elem.setAttribute('transform', currentTransform ? `${currentTransform} ${scaleTransform}` : scaleTransform);
            }

            // Universal Element Movement Logic
            function moveShape(elem, dx, dy) {
                const tag = elem.tagName.toLowerCase();
                if (tag === 'circle') {
                    elem.setAttribute('cx', parseFloat(elem.getAttribute('cx')) + dx);
                    elem.setAttribute('cy', parseFloat(elem.getAttribute('cy')) + dy);
                } else if (tag === 'line') {
                    elem.setAttribute('x1', parseFloat(elem.getAttribute('x1')) + dx);
                    elem.setAttribute('y1', parseFloat(elem.getAttribute('y1')) + dy);
                    elem.setAttribute('x2', parseFloat(elem.getAttribute('x2')) + dx);
                    elem.setAttribute('y2', parseFloat(elem.getAttribute('y2')) + dy);
                } else if (tag === 'rect' || tag === 'text') {
                    elem.setAttribute('x', parseFloat(elem.getAttribute('x')) + dx);
                    elem.setAttribute('y', parseFloat(elem.getAttribute('y')) + dy);
                } else if (tag === 'path') {
                    const currentTransform = elem.getAttribute('transform') || '';
                    const translateMatch = currentTransform.match(/translate\(([-+]?\d*\.?\d+)\s+([-+]?\d*\.?\d+)/i);


                    if (translateMatch) {
                        const tx = parseFloat(translateMatch[1]) + dx;
                        const ty = parseFloat(translateMatch[2]) + dy;
                        const withoutTranslate = currentTransform.replace(/translate\([^)]*\)\s*/i, '').trim();
                        elem.setAttribute('transform', withoutTranslate ? `translate(${tx} ${ty}) ${withoutTranslate}` : `translate(${tx} ${ty})`);
                    } else {
                        elem.setAttribute('transform', `translate(${dx} ${dy})`);
                    }
                }
            }


            // Shape Creation Factory
            function createShapeElement(tool, x, y) {
                const stroke = strokeColorInput.value;
                const fill = getActiveFill();
                const strokeWidth = strokeWidthInput.value;


                let elem;
                if (tool === 'circle') {
                    elem = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
                    elem.setAttribute('cx', x); elem.setAttribute('cy', y); elem.setAttribute('r', 0);
                } else if (tool === 'line') {
                    elem = document.createElementNS('http://www.w3.org/2000/svg', 'line');
                    elem.setAttribute('x1', x); elem.setAttribute('y1', y);
                    elem.setAttribute('x2', x); elem.setAttribute('y2', y);
                } else if (tool === 'square' || tool === 'rectangle' || tool === 'roundrect') {
                    elem = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
                    elem.setAttribute('x', x); elem.setAttribute('y', y);
                    elem.setAttribute('width', 0); elem.setAttribute('height', 0);
                    if (tool === 'roundrect') { elem.setAttribute('rx', 10); elem.setAttribute('ry', 10); }
                } else if (tool === 'pencil' || tool === 'heart' || tool === 'curve' || tool === 'polygon' || tool.startsWith('arrow')) {
                    elem = document.createElementNS('http://www.w3.org/2000/svg', 'path');
                    elem.setAttribute('d', '');
                }


                if (elem) {
                    elem.setAttribute('stroke', stroke);
                    elem.setAttribute('fill', (tool === 'line' || tool === 'pencil' || tool === 'curve') ? 'none' : fill);
                    elem.setAttribute('stroke-width', strokeWidth);
                    elem.setAttribute('stroke-linecap', 'round');
                    elem.setAttribute('stroke-linejoin', 'round');
                }


                return elem;
            }


            // Shape Resizing / Updating
            function updateShapeElement(elem, tool, x1, y1, x2, y2) {
                const dx = x2 - x1;
                const dy = y2 - y1;


                if (tool === 'line') {
                    elem.setAttribute('x2', x2);
                    elem.setAttribute('y2', y2);
                } else if (tool === 'circle') {
                    elem.setAttribute('r', Math.round(Math.hypot(dx, dy)));
                } else if (tool === 'rectangle' || tool === 'roundrect') {
                    elem.setAttribute('x', Math.min(x1, x2));
                    elem.setAttribute('y', Math.min(y1, y2));
                    elem.setAttribute('width', Math.abs(dx));
                    elem.setAttribute('height', Math.abs(dy));
                } else if (tool === 'square') {
                    const side = Math.max(Math.abs(dx), Math.abs(dy));
                    elem.setAttribute('x', x2 >= x1 ? x1 : x1 - side);
                    elem.setAttribute('y', y2 >= y1 ? y1 : y1 - side);
                    elem.setAttribute('width', side);
                    elem.setAttribute('height', side);
                } else if (tool === 'curve') {
                    const cx = (x1 + x2) / 2;
                    const cy = y1 - (y2 - y1);
                    elem.setAttribute('d', `M ${x1} ${y1} Q ${cx} ${cy} ${x2} ${y2}`);
                } else if (tool === 'polygon') {
                    elem.setAttribute('d', generateRegularPolygonPath(x1, y1, Math.abs(dx), 6));
                } else if (tool === 'heart') {
                    elem.setAttribute('d', generateHeartPath(Math.min(x1, x2), Math.min(y1, y2), Math.abs(dx), Math.abs(dy)));
                } else if (tool.startsWith('arrow')) {
                    const dir = tool.replace('arrow-', '');
                    elem.setAttribute('d', generateArrowPath(Math.min(x1, x2), Math.min(y1, y2), Math.abs(dx), Math.abs(dy), dir));
                }
            }




// Geometry Generators for Complex Paths
            function generateRegularPolygonPath(cx, cy, r, sides) {
                if (r <= 0) return '';
                let pts = [];
                for (let i = 0; i < sides; i++) {
                    const angle = (i * 2 * Math.PI / sides) - (Math.PI / 2);
                    const px = Math.round(cx + r * Math.cos(angle));
                    const py = Math.round(cy + r * Math.sin(angle));
                    pts.push(`${i === 0 ? 'M' : 'L'} ${px} ${py}`);
                }
                return pts.join(' ') + ' Z';
            }


            function generateHeartPath(x, y, w, h) {
                if (w <= 0 || h <= 0) return '';
                const topCurveHeight = h * 0.3;
                return `M ${x + w / 2} ${y + h} ` +
                       `C ${x} ${y + h / 2}, ${x} ${y}, ${x + w / 4} ${y} ` +
                       `C ${x + w / 2} ${y}, ${x + w / 2} ${y + topCurveHeight}, ${x + w / 2} ${y + topCurveHeight} ` +
                       `C ${x + w / 2} ${y + topCurveHeight}, ${x + w / 2} ${y}, ${x + (3 * w) / 4} ${y} ` +
                       `C ${x + w} ${y}, ${x + w} ${y + h / 2}, ${x + w / 2} ${y + h} Z`;
            }




            function generateArrowPath(x, y, w, h, dir) {
                if (w <= 0 || h <= 0) return '';
                switch (dir) {
                    case 'up':
                        return `M ${x + w / 2} ${y} L ${x + w} ${y + h / 2} H ${x + (3 * w) / 4} V ${y + h} H ${x + w / 4} V ${y + h / 2} H ${x} Z`;
                    case 'down':
                        return `M ${x + w / 2} ${y + h} L ${x + w} ${y + h / 2} H ${x + (3 * w) / 4} V ${y} H ${x + w / 4} V ${y + h / 2} H ${x} Z`;
                    case 'left':
                        return `M ${x} ${y + h / 2} L ${x + w / 2} ${y} V ${y + h / 4} H ${x + w} V ${y + (3 * h) / 4} H ${x + w / 2} V ${y + h} Z`;
                    case 'right':
                        return `M ${x + w} ${y + h / 2} L ${x + w / 2} ${y} V ${y + h / 4} H ${x} V ${y + (3 * h) / 4} H ${x + w / 2} V ${y + h} Z`;
                    default:
                        return '';
                }
            }

            function parseSvgMarkup(markup) {
                const trimmed = (markup || '').trim();
                if (!trimmed) {
                    return null;
                }


                const parser = new DOMParser();
                const doc = parser.parseFromString(trimmed, 'image/svg+xml');
                const parserError = doc.querySelector('parsererror');
                if (parserError) {
                    return null;
                }


                const svgRoot = doc.documentElement;
                if (!svgRoot || svgRoot.nodeName.toLowerCase() !== 'svg') {
                    return null;
                }


                return svgRoot;
            }


            function applySvgMarkupToCanvas(markup, options) {
                const settings = options || {};
                const preserveViewport = !!settings.preserveViewport;
                const svgRoot = parseSvgMarkup(markup);
                if (!svgRoot) {
                    shapesGroup.innerHTML = '';
                    return false;
                }


                const children = Array.from(svgRoot.childNodes)
                    .filter(node => node.nodeType === Node.ELEMENT_NODE)
                    .map(node => document.importNode(node, true));


                shapesGroup.replaceChildren(...children);


                if (!preserveViewport) {
