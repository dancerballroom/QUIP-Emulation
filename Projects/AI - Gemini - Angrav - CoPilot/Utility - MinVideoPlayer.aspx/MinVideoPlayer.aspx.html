        let currentFileName = "N/A";

        let toasterTimeout = null;


        // --- IndexedDB Setup for Session Persistence ---
        function openDB() {
            return new Promise((resolve, reject) => {
                const request = indexedDB.open('MinVideoPlayerDB', 1);
                request.onupgradeneeded = (e) => {
                    const db = e.target.result;
                    if (!db.objectStoreNames.contains('sessions')) {
                        db.createObjectStore('sessions');
                    }
                };
                request.onsuccess = (e) => resolve(e.target.result);
                request.onerror = (e) => reject(e.target.error);
            });
        }


        async function saveSessionToDB() {
            try {
                const db = await openDB();
                const transaction = db.transaction('sessions', 'readwrite');
                const store = transaction.objectStore('sessions');
                
                const sessionData = {
                    fileHandle: currentFileHandle,
                    fileName: currentFileName,
                    timestamps: txtTimestamps.value,
                    offset: numOffset.value,
                    currentTime: video.currentTime
                };


                store.put(sessionData, 'lastSession');
                return new Promise((resolve, reject) => {
                    transaction.oncomplete = () => resolve(true);
                    transaction.onerror = () => reject(transaction.error);
                });
            } catch (err) {
                console.error("Save error:", err);
                return false;
            }
        }


        async function loadSessionFromDB() {
            try {
                const db = await openDB();
                const transaction = db.transaction('sessions', 'readonly');
                const store = transaction.objectStore('sessions');
                const request = store.get('lastSession');


                return new Promise((resolve, reject) => {
                    request.onsuccess = () => resolve(request.result);
                    request.onerror = () => reject(request.error);
                });
            } catch (err) {
                console.error("Load error:", err);
                return null;
            }
        }


        // --- Session Button Event Handlers ---
        btnSaveSession.addEventListener('click', async () => {
            const success = await saveSessionToDB();
            showToaster(success ? "Session Saved" : "Save Failed");
        });


        btnLoadSession.addEventListener('click', async () => {
            const session = await loadSessionFromDB();
            if (!session) {
                showToaster("No Saved Session");
                return;
            }


            currentFileName = session.fileName || "N/A";
            currentFileHandle = session.fileHandle || null;
            txtTimestamps.value = session.timestamps || "";
            numOffset.value = session.offset || 0;


            if (currentFileHandle) {
                try {
                    const opts = { mode: 'read' };
                    if ((await currentFileHandle.queryPermission(opts)) !== 'granted') {
                        if ((await currentFileHandle.requestPermission(opts)) !== 'granted') {
                            showToaster("Permission Denied");
                            return;
                        }
                    }


                    const file = await currentFileHandle.getFile();
                    const fileURL = URL.createObjectURL(file);
                    video.src = fileURL;
                    video.currentTime = session.currentTime || 0;
                    video.play().catch(() => {});
                    welcomePrompt.style.display = 'none';
                    showToaster("Session Loaded");
                    closeModal();
                } catch (err) {
                    console.error(err);
                    showToaster("Error Loading File Handle");
                }
            } else {
                showToaster("No File Handle Found");
            }
        });


        // --- Toaster Notification Helper ---
        function showToaster(message) {
            toaster.textContent = message;
            toaster.style.opacity = '1';
            clearTimeout(toasterTimeout);
            toasterTimeout = setTimeout(() => {
                toaster.style.opacity = '0';
            }, 1200);
        }


        function showTimeToast() {
            const currentStr = formatTime(video.currentTime);
            const durationStr = formatTime(video.duration || 0);
            showToaster(`${currentStr} / ${durationStr}`);
        }


        // --- Format Time to HH:MM:SS ---
        function formatTime(seconds) {
            if (isNaN(seconds)) return "00:00:00";
            const h = Math.floor(seconds / 3600);
            const m = Math.floor((seconds % 3600) / 60);
            const s = Math.floor(seconds % 60);
            return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
        }

        // --- Helper: Parse Timestamps with Offset Applied ---
        function getParsedTimestamps() {
            const lines = txtTimestamps.value.split('\n');
            let offset = parseInt(numOffset.value, 10);
            if (isNaN(offset) || offset < 0) offset = 0;
            if (offset > 6) offset = 6;


            let secondsArray = [];
            lines.forEach(line => {
                const trimmed = line.trim();
                const parts = trimmed.split(':').map(Number);
                let baseSeconds = NaN;
                if (parts.length === 3 && !parts.some(isNaN)) {
                    baseSeconds = parts[0] * 3600 + parts[1] * 60 + parts[2];
                } else if (parts.length === 2 && !parts.some(isNaN)) {
                    baseSeconds = parts[0] * 60 + parts[1];
                }


                if (!isNaN(baseSeconds)) {
                    secondsArray.push(baseSeconds + offset);
                }
            });
            // Ensure unique, sorted timestamps to prevent duplicate hopping
            return [...new Set(secondsArray)].sort((a, b) => a - b);
        }


        // --- Execute Jump Logic ---
        function executeJump() {
            if (!video.src) {
                showToaster("No Video Loaded");
                return;
            }


            const timeStr = txtJumpTime.value.trim();
            const parts = timeStr.split(':').map(Number);
            let targetSeconds = NaN;


            if (parts.length === 3 && !parts.some(isNaN)) {
                targetSeconds = parts[0] * 3600 + parts[1] * 60 + parts[2];
            } else if (parts.length === 2 && !parts.some(isNaN)) {
                targetSeconds = parts[0] * 60 + parts[1];
            } else if (parts.length === 1 && !isNaN(parts[0])) {
                targetSeconds = parts[0];
            }


            if (isNaN(targetSeconds)) {
                showToaster("Invalid Time Format");
                return;
            }


            const duration = video.duration || 0;


            if (targetSeconds < 0) {
                video.currentTime = 0;
                showToaster("Beginning of Video");
            } else if (duration > 0 && targetSeconds > duration) {
                video.currentTime = duration;
                showToaster("End of Video");
            } else {
                video.currentTime = targetSeconds;
                showTimeToast();
            }


            jumpModal.style.display = 'none';
        }

            // Skipping via Arrow Keys (when inputs aren't focused)

                const duration = video.duration || 0;


                // 1. Control + Alt + Left / Right Arrow -> Jump through Timestamp List
                if (e.ctrlKey && e.altKey && (e.key === 'ArrowLeft' || e.key === 'ArrowRight')) {
                    e.preventDefault();
                    e.stopImmediatePropagation();
                    const times = getParsedTimestamps();
                    if (times.length > 0) {
                        const currentTime = video.currentTime;
                        let targetTime = null;


                        if (e.key === 'ArrowRight') {
                            let nextTime = times.find(t => t > currentTime + 0.05);
                            if (nextTime !== undefined) targetTime = nextTime;
                        } else {
                            let prevTimes = times.filter(t => t < currentTime - 0.05);
                            if (prevTimes.length > 0) targetTime = prevTimes[prevTimes.length - 1];
                        }


                        if (targetTime !== null) {
                            if (targetTime < 0) {
                                showToaster("Beginning of Video");
                            } else if (duration > 0 && targetTime > duration) {
                                showToaster("End of Video");
                            } else {
                                video.currentTime = targetTime;
                                showTimeToast();
                            }
                        } else {
                            showToaster(e.key === 'ArrowRight' ? "End of Video" : "Beginning of Video");
                        }
                    }
                    return;
                }


                // 2. Control + Shift + Left / Right Arrow -> 1 Second
                if (e.ctrlKey && e.shiftKey && (e.key === 'ArrowLeft' || e.key === 'ArrowRight')) {
                    e.preventDefault();
                    e.stopImmediatePropagation();
                    let target = video.currentTime + (e.key === 'ArrowRight' ? 1 : -1);
                    if (target < 0) {
                        video.currentTime = 0;
                        showToaster("Beginning of Video");

                        showToaster("End of Video");
                    } else {

                        showTimeToast();
                    }
                    return;
                }


                // 3. Control + Left / Right Arrow -> 10 Seconds
                if (e.ctrlKey && !e.shiftKey && !e.altKey && (e.key === 'ArrowLeft' || e.key === 'ArrowRight')) {
                    e.preventDefault();
