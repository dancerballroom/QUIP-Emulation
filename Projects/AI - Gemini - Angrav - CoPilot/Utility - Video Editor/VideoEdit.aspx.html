        function getViewportRange() {
            const duration = video.duration || 0;
            if (!duration) return { start: 0, end: 0 };
            if (waveformZoom <= 1) return { start: 0, end: duration };


            const visibleDuration = Math.max(0.05, duration / waveformZoom);
            const center = (waveViewStartSec + waveViewEndSec) / 2 || video.currentTime || 0;
            let start = center - (visibleDuration / 2);
            let end = center + (visibleDuration / 2);


            if (start < 0) {
                start = 0;
                end = visibleDuration;
            }
            if (end > duration) {
                end = duration;
                start = Math.max(0, end - visibleDuration);
            }


            return { start, end };
        }

        function chooseTickStep(visibleSeconds) {
            const targetTicks = 8;
            const minStep = visibleSeconds / targetTicks;
            const candidates = [0.1, 0.2, 0.5, 1, 2, 5, 10, 15, 30, 60, 120, 300, 600, 900, 1800];


            for (let i = 0; i < candidates.length; i++) {
                if (candidates[i] >= minStep) return candidates[i];
            }
            return 3600;
        }


        function drawTimelineTicks(width, height, viewStart, viewEnd) {

            const visibleSeconds = Math.max(0.05, viewEnd - viewStart);
            const tickStep = chooseTickStep(visibleSeconds);
            const firstTick = Math.ceil(viewStart / tickStep) * tickStep;


            ctx.save();
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.22)';
            ctx.lineWidth = 1;
            ctx.fillStyle = 'rgba(255, 255, 255, 0.78)';
            ctx.font = '11px monospace';


            for (let tick = firstTick; tick <= viewEnd; tick += tickStep) {
                const x = ((tick - viewStart) / visibleSeconds) * width;
                ctx.beginPath();
                ctx.moveTo(x + 0.5, 0);
                ctx.lineTo(x + 0.5, height);
                ctx.stroke();
                ctx.fillText(formatMarkerTime(tick), Math.max(2, x + 3), 11);
            }


            const currentX = ((video.currentTime - viewStart) / visibleSeconds) * width;
            if (currentX >= 0 && currentX <= width) {
                ctx.strokeStyle = 'rgba(255, 64, 64, 0.95)';
                ctx.beginPath();
                ctx.moveTo(currentX + 0.5, 0);
                ctx.lineTo(currentX + 0.5, height);
                ctx.stroke();
            }
            ctx.restore();
        }

        function handleGlobalKeydown(e) {
            const targetEl = e.target;


            if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'z') {
                e.preventDefault();
                if (e.shiftKey) {
                    redo();
                } else {
                    undo();
                }
                return;
            }
            if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'y') {
                e.preventDefault();
                redo();
                return;
            }


            if (isEditableTarget(targetEl)) return;


            const isArrowRight = e.key === 'ArrowRight' || e.code === 'ArrowRight';
            const isArrowLeft = e.key === 'ArrowLeft' || e.code === 'ArrowLeft';


            if (isArrowRight) {
                e.preventDefault();
                e.stopPropagation();
                seekByStep(seekStepSeconds);
            } else if (isArrowLeft) {
                e.preventDefault();
                e.stopPropagation();
                seekByStep(-seekStepSeconds);
            } else if (e.key === '[') {
                markStart();
            } else if (e.key === ']') {
                markEnd();
            }
        }

        async function getDroppedFileSource(dataTransfer) {
            if (!dataTransfer) {
                return { file: null, sourceHandle: null };
            }


            if (dataTransfer.items && dataTransfer.items.length > 0) {
                for (const item of dataTransfer.items) {
                    if (!item || item.kind !== 'file') continue;


                    if (typeof item.getAsFileSystemHandle === 'function') {
                        try {
                            const handle = await item.getAsFileSystemHandle();
                            if (handle && handle.kind === 'file') {
                                const file = await handle.getFile();
                                return { file: file, sourceHandle: handle };
                            }
                        } catch (_) {
                            // Fall back to the legacy file object below.
                        }
                    }
                }
            }


            const file = dataTransfer.files && dataTransfer.files.length > 0 ? dataTransfer.files[0] : null;
            return { file: file, sourceHandle: null };
        }

            try {
                payload = await response.json();
            } catch (_) {
                // Ignore parse details, a non-JSON response is handled below.
            }


            if (!response.ok || !payload || payload.ok !== true) {
                const reason = payload && payload.error ? payload.error : `Server returned ${response.status}`;
                throw new Error(reason);
            }


            if (!Array.isArray(payload.mins) || !Array.isArray(payload.maxs)) {
                throw new Error('Server response did not include waveform peak arrays.');
            }


            if (jobId !== waveformJobId) return false;


            waveformPeakMins = Float32Array.from(payload.mins);
            waveformPeakMaxs = Float32Array.from(payload.maxs);
            waveformPeakCount = Math.min(waveformPeakMins.length, waveformPeakMaxs.length);


            if (waveformPeakCount <= 0) {
                throw new Error('Server returned an empty waveform peak set.');
            }


            waveformDecodeFailed = false;
            setWaveformStatus('Waveform ready (server fallback).');
            drawWaveform();
            setTimeout(() => {
                if (jobId === waveformJobId) clearWaveformStatus();
            }, 1200);
            return true;
        }


        // Time Formatting & Parsing Utilities
        function formatTime(seconds) {
            if (isNaN(seconds) || seconds < 0) seconds = 0;
            const date = new Date(seconds * 1000);
            const hh = String(Math.floor(seconds / 3600)).padStart(2, '0');
            const mm = String(date.getUTCMinutes()).padStart(2, '0');
            const ss = String(date.getUTCSeconds()).padStart(2, '0');
            const mmm = String(date.getUTCMilliseconds()).padStart(3, '0');
            return `${hh}:${mm}:${ss}.${mmm}`;
        }


        function formatMarkerTime(seconds) {
            if (isNaN(seconds) || seconds < 0) seconds = 0;
            const rounded = Math.round(seconds);
            const hh = String(Math.floor(rounded / 3600)).padStart(2, '0');
            const mm = String(Math.floor((rounded % 3600) / 60)).padStart(2, '0');
            const ss = String(rounded % 60).padStart(2, '0');
            return `${hh}:${mm}:${ss}`;
        }


        function parseTime(timeStr) {
            if (!timeStr) return 0;
            const parts = timeStr.trim().split(':');
            let seconds = 0;


            if (parts.length === 3) {
                seconds = (parseFloat(parts[0]) || 0) * 3600 +
                          (parseFloat(parts[1]) || 0) * 60 +
                          (parseFloat(parts[2]) || 0);
            } else if (parts.length === 2) {
                seconds = (parseFloat(parts[0]) || 0) * 60 +
                          (parseFloat(parts[1]) || 0);
            } else if (parts.length === 1) {
                seconds = parseFloat(parts[0]) || 0;
            }
            return Math.max(0, seconds);
        }

                    marker.endTime = currentTime;
                    renderMarkers();

                }

                endTime: currentTime,

            isBuildingEntry = true;

            saveHistoryState();

            }

            renderMarkers();
        }


        function deleteMarker(id) {
            saveHistoryState();
            markers = markers.filter(m => m.id !== id);
            if (activeMarkerId === id) {
                activeMarkerId = markers.length > 0 ? markers[markers.length - 1].id : null;
                isBuildingEntry = false;
            }
            renderMarkers();
        }


        function updateMarkerData(id, field, value) {
            saveHistoryState();
            const marker = markers.find(m => m.id === id);
            if (marker) {
                if (field === 'startTime' || field === 'endTime') {
                    marker[field] = formatMarkerTime(parseTime(value));
                } else {
                    marker[field] = value;
                }
                drawWaveform();
            }
        }

        async function buildWaveformPeaks(rawData, jobId, targetPeakCount = 12000) {
            const length = rawData.length;
            if (!length) {
                waveformPeakMins = null;
                waveformPeakMaxs = null;
                waveformPeakCount = 0;
                return;
            }


            const peakCount = Math.max(512, Math.min(targetPeakCount, length));
            const blockSize = Math.max(1, Math.floor(length / peakCount));
            const mins = new Float32Array(peakCount);
            const maxs = new Float32Array(peakCount);


            const progressBatch = 220;
            for (let i = 0; i < peakCount; i++) {
                if (jobId !== waveformJobId) return;


                const start = i * blockSize;
                const end = i === peakCount - 1 ? length : Math.min(length, start + blockSize);


                let min = 1.0;
                let max = -1.0;
                for (let j = start; j < end; j++) {
                    const datum = rawData[j];
                    if (datum < min) min = datum;
                    if (datum > max) max = datum;
                }


                mins[i] = min;
                maxs[i] = max;


                if (i % progressBatch === 0) {
                    const progress = Math.floor((i / peakCount) * 100);
                    setWaveformStatus(`Building waveform peaks... ${progress}%`);
                    await new Promise(resolve => requestAnimationFrame(resolve));
                }
            }


            setWaveformStatus('Building waveform peaks... 100%');


            waveformPeakMins = mins;
            waveformPeakMaxs = maxs;
            waveformPeakCount = peakCount;
        }


        // Audio Waveform Decoding
        async function extractAudioWaveform(file, jobId) {
            const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            try {
                setWaveformStatus('Reading video audio...');
                const arrayBuffer = await file.arrayBuffer();
                if (jobId !== waveformJobId) return;


                setWaveformStatus('Decoding audio track...');
                const decoded = await audioCtx.decodeAudioData(arrayBuffer);
                if (jobId !== waveformJobId) return;


                audioBuffer = decoded;
                await buildWaveformPeaks(decoded.getChannelData(0), jobId);
                if (jobId !== waveformJobId) return;


                // Release heavy decoded buffer reference after generating peaks.
                audioBuffer = null;
                waveformDecodeFailed = false;
                drawWaveform();
                setWaveformStatus('Waveform ready.');
                setTimeout(() => {
                    if (jobId === waveformJobId) clearWaveformStatus();
                }, 900);
            } catch (err) {
                if (jobId !== waveformJobId) return;
                waveformDecodeFailed = true;
                waveformPeakMins = null;
                waveformPeakMaxs = null;
                waveformPeakCount = 0;
                console.warn('Could not decode audio waveform for this file.', err);


                try {
                    await tryServerWaveformFallback(file || currentVideoFile, jobId);
                } catch (fallbackErr) {
                    if (jobId !== waveformJobId) return;
                    console.warn('Server waveform fallback failed.', fallbackErr);
                    setWaveformStatus('Waveform unavailable for this file (browser + server decode failed).', true);
                    drawWaveform();
                }
            } finally {
                if (audioCtx && typeof audioCtx.close === 'function') {
                    audioCtx.close().catch(() => {});
                }
            }
        }


        function getTimeFromX(x) {
            if (!video.duration) return 0;
            const rect = canvas.getBoundingClientRect();
            const percent = Math.max(0, Math.min(1, (x - rect.left) / rect.width));
            const viewport = getViewportRange();
            return viewport.start + (percent * (viewport.end - viewport.start));
        }


        function drawWaveform() {
            const width = canvas.width = canvas.parentElement.clientWidth;
            const height = canvas.height = canvas.parentElement.clientHeight;
            const viewport = getViewportRange();
            const visibleDuration = Math.max(0.05, viewport.end - viewport.start);


            ctx.clearRect(0, 0, width, height);


            if (!waveformPeakMins || !waveformPeakMaxs || waveformPeakCount === 0) {
                ctx.fillStyle = '#666';
                ctx.font = '13px sans-serif';
                const message = waveformDecodeFailed
                    ? 'Waveform preview unavailable for this file. Timeline + markers remain usable.'
                    : 'Audio waveform will display here once video is loaded.';
                ctx.fillText(message, 20, height / 2);
                drawTimelineTicks(width, height, viewport.start, viewport.end);
                return;
            }


            const totalPeaks = waveformPeakCount;
            const startPeak = Math.max(0, Math.floor((viewport.start / video.duration) * totalPeaks));
            const endPeak = Math.min(totalPeaks, Math.ceil((viewport.end / video.duration) * totalPeaks));
            const samples = width;
            const visiblePeakCount = Math.max(1, endPeak - startPeak);
            const peaksPerPixel = visiblePeakCount / Math.max(1, samples);
            const amp = height / 2;


            ctx.fillStyle = '#007acc';
            for (let i = 0; i < samples; i++) {
                const from = Math.min(totalPeaks - 1, Math.max(0, Math.floor(startPeak + (i * peaksPerPixel))));
                const to = Math.min(totalPeaks - 1, Math.max(from, Math.floor(startPeak + ((i + 1) * peaksPerPixel))));


                let min = 1;
                let max = -1;
                for (let p = from; p <= to; p++) {
                    const peakMin = waveformPeakMins[p];
                    const peakMax = waveformPeakMaxs[p];
                    if (peakMin < min) min = peakMin;
                    if (peakMax > max) max = peakMax;
                }


                ctx.fillRect(i, (1 + min) * amp, 1, Math.max(1, (max - min) * amp));
            }


            drawTimelineTicks(width, height, viewport.start, viewport.end);


            if (video.duration) {
                markers.forEach((m) => {
                    const startSec = parseTime(m.startTime);
                    const endSec = parseTime(m.endTime);


                    if (endSec < viewport.start || startSec > viewport.end) return;


                    const startPx = ((startSec - viewport.start) / visibleDuration) * width;
                    const endPx = ((endSec - viewport.start) / visibleDuration) * width;
                    const isSelected = m.id === activeMarkerId;


                    ctx.fillStyle = isSelected ? 'rgba(0, 122, 204, 0.4)' : 'rgba(40, 167, 69, 0.3)';
                    ctx.fillRect(startPx, 0, Math.max(2, endPx - startPx), height);


                    ctx.fillStyle = isSelected ? '#007acc' : '#28a745';
                    ctx.fillRect(startPx - 2, 0, 4, height);
                    ctx.fillRect(endPx - 2, 0, 4, height);


                    ctx.fillRect(startPx - 5, 0, 10, 8);
                    ctx.fillRect(endPx - 5, 0, 10, 8);
                });
            }


            if (isDragging && dragMode === 'create' && video.duration) {
                const sPx = ((Math.min(dragStartSec, dragCurrentSec) - viewport.start) / visibleDuration) * width;
                const ePx = ((Math.max(dragStartSec, dragCurrentSec) - viewport.start) / visibleDuration) * width;


                ctx.fillStyle = 'rgba(255, 193, 7, 0.4)';
                ctx.fillRect(sPx, 0, ePx - sPx, height);


                ctx.strokeStyle = '#ffc107';
                ctx.lineWidth = 2;
                ctx.strokeRect(sPx, 0, ePx - sPx, height);
            }
        }


        const HANDLE_THRESHOLD_PX = 8;
        const CREATE_MIN_DRAG_PX = 8;
        const CREATE_MIN_DURATION_SEC = 0.2;


        function hitTestMarkerHandle(clickSec) {
            if (!video.duration) return null;
            const width = canvas.clientWidth;
            const viewport = getViewportRange();
            const secondsPerPx = (viewport.end - viewport.start) / Math.max(1, width);
            const thresholdSec = secondsPerPx * HANDLE_THRESHOLD_PX;


            for (let i = markers.length - 1; i >= 0; i--) {
                const m = markers[i];
                const startSec = parseTime(m.startTime);
                const endSec = parseTime(m.endTime);


                if (Math.abs(clickSec - startSec) <= thresholdSec) {
                    return { marker: m, handle: 'start' };
                }
                if (Math.abs(clickSec - endSec) <= thresholdSec) {
                    return { marker: m, handle: 'end' };
                }
            }
            return null;
        }


        waveformContainer.addEventListener('mousedown', (e) => {

            const hit = hitTestMarkerHandle(clickSec);


            if (hit) {
                isDragging = true;

                activeMarkerId = targetMarker.id;

                isDragging = true;

            }

        });


        waveformContainer.addEventListener('mousemove', (e) => {
            if (!video.duration) return;


            dragCurrentClientX = e.clientX;
            const currentSec = getTimeFromX(e.clientX);


            if (!isDragging) {
                const hit = hitTestMarkerHandle(currentSec);
                waveformContainer.style.cursor = hit ? 'ew-resize' : 'text';
                return;
            }


            if (dragMode === 'create') {
                dragCurrentSec = currentSec;
            } else if (dragMode === 'move-start' && targetMarker) {
                targetMarker.startTime = formatMarkerTime(Math.min(currentSec, parseTime(targetMarker.endTime)));
            } else if (dragMode === 'move-end' && targetMarker) {
                targetMarker.endTime = formatMarkerTime(Math.max(currentSec, parseTime(targetMarker.startTime)));
            }


            drawWaveform();
        });


        waveformContainer.addEventListener('wheel', (e) => {
            if (!video.duration || waveformZoom <= 1) return;


            e.preventDefault();


            const viewport = getViewportRange();
            const visibleSpan = Math.max(0.05, viewport.end - viewport.start);
            const dominantDelta = Math.abs(e.deltaX) > Math.abs(e.deltaY) ? e.deltaX : e.deltaY;
            const units = dominantDelta / 120;
            const shiftSeconds = units * visibleSpan * 0.2;


            shiftWaveformViewport(shiftSeconds);
        }, { passive: false });


        window.addEventListener('mouseup', (e) => {
            if (!isDragging) return;


            dragCurrentClientX = typeof e.clientX === 'number' ? e.clientX : dragCurrentClientX;


            if (dragMode === 'create') {
                dragCurrentSec = getTimeFromX(e.clientX);
                const finalStart = Math.min(dragStartSec, dragCurrentSec);
                const finalEnd = Math.max(dragStartSec, dragCurrentSec);
                const dragDistancePx = Math.abs(dragCurrentClientX - dragStartClientX);
                const selectedDuration = finalEnd - finalStart;


                if (dragDistancePx >= CREATE_MIN_DRAG_PX && selectedDuration >= CREATE_MIN_DURATION_SEC) {
                    const newMarker = addMarker(finalStart, finalEnd, true);
                    activeMarkerId = newMarker.id;
                    isBuildingEntry = false;
                } else {
                    undoStack.pop();
                    updateUndoRedoUI();
                    video.currentTime = finalStart;
                    updatePlayheadPosition();
                }
            }


            isDragging = false;
            dragMode = null;
            targetMarker = null;
            renderMarkers();
        });


        // Keyframe Navigation & Global Shortcuts
        const isEditableTarget = (targetEl) => {
            const tagName = targetEl && targetEl.tagName ? targetEl.tagName.toUpperCase() : '';
            return tagName === 'TEXTAREA' || tagName === 'INPUT' || !!(targetEl && targetEl.isContentEditable);
        };

        // Batch File Export (.bat)
        async function exportBat() {
            if (markers.length === 0) {
                alert("Please add at least one marker/selected segment.");
                return;
            }


            const sourceFileName = loadedVideoFileName || `${loadedVideoBaseName}.mp4`;
            const mergedDirectory = "Merged";
            const mergedOutputPath = `${mergedDirectory}\\${sourceFileName}`;


            let totalDuration = video.duration;


            if (!totalDuration || isNaN(totalDuration) || totalDuration === 0) {
                const userVal = prompt("Video duration is unknown (possibly due to page refresh).\nPlease enter total video duration in seconds (e.g. 120) or HH:MM:SS format:");
                if (!userVal) return;
                totalDuration = userVal.includes(':') ? parseTime(userVal) : parseFloat(userVal);
            }


            if (!totalDuration || isNaN(totalDuration)) {
                alert("Invalid total duration specified.");
                return;
            }


            const sortedMarkers = [...markers].map(m => ({
                start: parseTime(m.startTime),
                end: parseTime(m.endTime)
            })).sort((a, b) => a.start - b.start);


            const keepSegments = [];
            let currentPointer = 0;


            sortedMarkers.forEach(m => {
                if (m.start > currentPointer) {
                    keepSegments.push({ start: currentPointer, end: m.start });
                }
                if (m.end > currentPointer) {
                    currentPointer = m.end;
                }
            });


            if (totalDuration > currentPointer) {
                keepSegments.push({ start: currentPointer, end: totalDuration });
            }


            if (keepSegments.length === 0) {
                alert("Your markers cover the entire duration. No outside content to extract.");
                return;
            }


            let batContent = "@echo off\r\nsetlocal EnableExtensions EnableDelayedExpansion\r\nrem FFmpeg Inverted Cut & Merge Script\r\n\r\n";
            batContent += "if exist concat_list.txt del /q concat_list.txt\r\n";
            batContent += "if exist process_failures.log del /q process_failures.log\r\n\r\n";
            batContent += `if not exist "${mergedDirectory}" mkdir "${mergedDirectory}"\r\n\r\n`;


            keepSegments.forEach((seg, idx) => {
                const rawClipName = `keep_part_${idx + 1}_raw.mp4`;
                const clipName = `keep_part_${idx + 1}.mp4`;
                const startStr = formatTime(seg.start);
                const duration = (seg.end - seg.start).toFixed(3);


                batContent += `echo Extracting inverted segment ${idx + 1} of ${keepSegments.length} (${startStr} for ${duration}s)...\r\n`;
                batContent += `ffmpeg -ss ${startStr} -i "${sourceFileName}" -t ${duration} -c copy -y "${rawClipName}"\r\n`;
                batContent += "if errorlevel 1 (\r\n";
                batContent += `  echo ERROR: segment ${idx + 1} extraction failed >> process_failures.log\r\n`;
                batContent += "  goto :FAILED\r\n";
                batContent += ")\r\n";
                batContent += `echo Repairing timestamps for segment ${idx + 1}...\r\n`;
                batContent += `ffmpeg -fflags +genpts+igndts -i "${rawClipName}" -map 0 -c copy -avoid_negative_ts make_zero -movflags +faststart -y "${clipName}"\r\n`;
                batContent += "if errorlevel 1 (\r\n";
                batContent += `  echo Segment ${idx + 1} remux repair failed, retrying with audio re-encode...\r\n`;
                batContent += `  ffmpeg -fflags +genpts -i "${rawClipName}" -map 0 -c:v copy -c:a aac -b:a 192k -movflags +faststart -y "${clipName}"\r\n`;
                batContent += "  if errorlevel 1 (\r\n";
                batContent += `    echo ERROR: segment ${idx + 1} repair failed >> process_failures.log\r\n`;
                batContent += "    goto :FAILED\r\n";
                batContent += "  )\r\n";
                batContent += ")\r\n";
                batContent += `echo file '${clipName}' >> concat_list.txt\r\n\r\n`;
            });


            batContent += `echo Joining all extracted segments into ${mergedOutputPath}...\r\n`;
            batContent += `ffmpeg -f concat -safe 0 -i concat_list.txt -c copy -movflags +faststart -y "${mergedOutputPath}"\r\n`;
            batContent += "if errorlevel 1 (\r\n";
            batContent += "  echo Concat with stream copy failed, retrying with audio re-encode...\r\n";
            batContent += `  ffmpeg -f concat -safe 0 -i concat_list.txt -c:v copy -c:a aac -b:a 192k -movflags +faststart -y "${mergedOutputPath}"\r\n`;
            batContent += "  if errorlevel 1 goto :FAILED\r\n";
            batContent += ")\r\n\r\n";


            batContent += "rem Cleanup temporary files\r\n";
            keepSegments.forEach((_, idx) => {
                batContent += `del "keep_part_${idx + 1}_raw.mp4"\r\n`;
                batContent += `del "keep_part_${idx + 1}.mp4"\r\n`;
            });
            batContent += "del concat_list.txt\r\n\r\n";


            batContent += `echo Finished processing! Final video saved as ${mergedOutputPath}\r\n`;
            batContent += "if exist process_failures.log del /q process_failures.log\r\n";
            batContent += "pause\r\n";
            batContent += "goto :EOF\r\n\r\n";
            batContent += ":FAILED\r\n";
            batContent += "echo.\r\n";
            batContent += "echo Processing failed. See process_failures.log for details.\r\n";
            batContent += "pause\r\n";


            const suggestedFileName = `${loadedVideoBaseName}_Trim.bat`;
            await saveFileWithPicker(suggestedFileName, batContent, "text/plain");
        }


        // Native Save As Dialog with Fallback
        async function saveFileWithPicker(suggestedName, content, mimeType) {
            const dotIndex = suggestedName.lastIndexOf('.');
            const fileExtension = dotIndex >= 0 ? suggestedName.substring(dotIndex).toLowerCase() : '';
            let fileDescription = 'Text File';


            if (fileExtension === '.bat') {
                fileDescription = 'Batch Script File';
            } else if (fileExtension === '.srt') {
                fileDescription = 'SubRip Subtitle File';
            }


            if ('showSaveFilePicker' in window) {
                try {
                    const handle = await window.showSaveFilePicker({
                        suggestedName: suggestedName,
                        startIn: getSavePickerStartIn(),
                        types: [{
                            description: fileDescription,
                            accept: { [mimeType]: fileExtension ? [fileExtension] : ['.txt'] }
                        }]
                    });
                    const writable = await handle.createWritable();
                    await writable.write(content);
                    await writable.close();
                    return;
                } catch (err) {
                    if (err.name === 'AbortError') return; // User cancelled
                    console.warn('File System Access API failed, falling back to direct download.', err);
                }
            }


            downloadFile(suggestedName, content, mimeType);
        }


        // Subtitle Export (.srt)
        async function exportSrt() {
            if (markers.length === 0) {
                alert("No subtitles available to export.");
                return;
            }


            let srtContent = "";
            markers.forEach((m, idx) => {
                const autoNum = idx + 1;
                const startSrt = `${formatMarkerTime(parseTime(m.startTime))},000`;
                const endSrt = `${formatMarkerTime(parseTime(m.endTime))},000`;


                srtContent += `${autoNum}\r\n${startSrt} --> ${endSrt}\r\n${m.text || 'Subtitle Entry'}\r\n\r\n`;
            });


            const suggestedFileName = `${loadedVideoBaseName}_B4_Trim.srt`;
            await saveFileWithPicker(suggestedFileName, srtContent, "text/plain");
        }


        function downloadFile(filename, content, mimeType) {
            const blob = new Blob([content], { type: mimeType });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = filename;
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

            } catch (err) {
                content.textContent = 'Help Under Construction';
            }
        }
