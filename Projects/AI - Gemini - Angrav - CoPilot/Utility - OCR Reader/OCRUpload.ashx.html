<%@ WebHandler Language="C#" Class="OCRUpload" %>


using System;
using System.IO;
using System.Text;
using System.Web;


public class OCRUpload : IHttpHandler
{
    public bool IsReusable { get { return false; } }


    public void ProcessRequest(HttpContext context)
    {
        context.Response.ContentType = "application/json";
        context.Response.ContentEncoding = Encoding.UTF8;


        try
        {
            if (context.Request.Files.Count == 0)
            {
                WriteJson(context, 400, false, null, "No uploaded file was received.");
                return;
            }


            HttpPostedFile file = context.Request.Files[0];
            if (file == null || file.ContentLength <= 0)
            {
                WriteJson(context, 400, false, null, "Uploaded file is empty.");
                return;
            }


            string originalName = Path.GetFileName(file.FileName);
            string ext = Path.GetExtension(originalName);
            if (string.IsNullOrWhiteSpace(ext) || ext.Length > 10)
            {
                ext = ".mp4";
            }


            string tempDir = context.Server.MapPath("~/App_Data/Temp/");
            if (!Directory.Exists(tempDir))
            {
                Directory.CreateDirectory(tempDir);
            }


            string token = "ocrvid_" + Guid.NewGuid().ToString("N") + ext.ToLowerInvariant();
            string savePath = Path.Combine(tempDir, token);
            file.SaveAs(savePath);


            WriteJson(context, 200, true, token, null);
        }
        catch (Exception ex)
        {
            WriteJson(context, 500, false, null, ex.Message);
        }
    }


    private static void WriteJson(HttpContext context, int statusCode, bool ok, string token, string error)
    {
        context.Response.StatusCode = statusCode;
        string json = "{\"ok\":" + (ok ? "true" : "false");


        if (!string.IsNullOrWhiteSpace(token))
        {
            json += ",\"token\":\"" + JsonEscape(token) + "\"";
        }


        if (!string.IsNullOrWhiteSpace(error))
        {
            json += ",\"error\":\"" + JsonEscape(error) + "\"";
        }


        json += "}";
        context.Response.Write(json);
    }


    private static string JsonEscape(string value)
    {
        if (value == null) return string.Empty;
        return value
            .Replace("\\", "\\\\")
            .Replace("\"", "\\\"")
            .Replace("\r", "\\r")
            .Replace("\n", "\\n");
    }
        }
