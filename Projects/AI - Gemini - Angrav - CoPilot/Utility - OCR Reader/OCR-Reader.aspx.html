private static string JsonEscape(string value)
{
    if (value == null) return string.Empty;
    return value
        .Replace("\\", "\\\\")
        .Replace("\"", "\\\"")
        .Replace("\r", "\\r")
        .Replace("\n", "\\n");
}


private static bool ResolveTesseractPaths(out string tesseractExePath, out string tessdataDir)
{
    tesseractExePath = null;
    tessdataDir = null;


    string[] exeCandidates = new string[]
    {
        @"C:\Program Files\Tesseract-OCR\tesseract.exe",
        @"C:\Program Files (x86)\Tesseract-OCR\tesseract.exe"
    };


    for (int i = 0; i < exeCandidates.Length; i++)
    {
        string exe = exeCandidates[i];
        if (!File.Exists(exe)) continue;


        string exeDir = Path.GetDirectoryName(exe) ?? string.Empty;
        string[] tessdataCandidates = new string[]
        {
            Path.Combine(exeDir, "tessdata"),
            @"C:\Program Files\Tesseract-OCR\tessdata",
            @"C:\Program Files (x86)\Tesseract-OCR\tessdata"
        };


        for (int j = 0; j < tessdataCandidates.Length; j++)
        {
            string td = tessdataCandidates[j];
            string engPath = Path.Combine(td, "eng.traineddata");
            if (Directory.Exists(td) && File.Exists(engPath))
            {
                tesseractExePath = exe;
                tessdataDir = td;
                return true;
            }
        }
    }


    return false;
}


private static string BuildOcrResultJson(string text, string enhUrl, string rawUrl, string bwUrl, string invUrl)
{
    return "{"
        + "\"text\":\"" + JsonEscape(text ?? string.Empty) + "\"," 
        + "\"debug\":{"
        + "\"enh\":\"" + JsonEscape(enhUrl ?? string.Empty) + "\"," 
        + "\"raw\":\"" + JsonEscape(rawUrl ?? string.Empty) + "\"," 
        + "\"bw\":\"" + JsonEscape(bwUrl ?? string.Empty) + "\"," 
        + "\"inv\":\"" + JsonEscape(invUrl ?? string.Empty) + "\""
        + "}"
        + "}";
}


private static string SaveDebugImage(string tempDir, string sourcePath, string label, string debugToken)
{
    if (string.IsNullOrWhiteSpace(sourcePath) || !File.Exists(sourcePath))
    {
        return string.Empty;
    }


    string fileName = "ocrdbg_" + label + "_" + debugToken + ".png";
    string outputPath = Path.Combine(tempDir, fileName);
    File.Copy(sourcePath, outputPath, true);
    return fileName;
}


private static void CleanupPreviousDebugFiles(string tempDir)
{
    string latestManifest = Path.Combine(tempDir, "ocrdbg_latest.txt");
    if (!File.Exists(latestManifest)) return;


    string[] oldFiles = File.ReadAllLines(latestManifest);
    for (int i = 0; i < oldFiles.Length; i++)
    {
        string oldName = oldFiles[i] == null ? string.Empty : oldFiles[i].Trim();
        if (oldName.StartsWith("ocrdbg_", StringComparison.OrdinalIgnoreCase))
        {
            string full = Path.Combine(tempDir, oldName);
            if (File.Exists(full)) File.Delete(full);
        }
    }
}


private static string RunTesseractAndReadText(string tesseractPath, string tessdataDir, string imagePath, string outputBasePath, string extraArgs)
{
    string outputTxt = outputBasePath + ".txt";
    string args = string.Format("\"{0}\" \"{1}\" --tessdata-dir \"{2}\" {3}", imagePath, outputBasePath, tessdataDir, extraArgs);


    ProcessStartInfo psi = new ProcessStartInfo
    {
        FileName = tesseractPath,
        Arguments = args,
        UseShellExecute = false,
        CreateNoWindow = true,
        RedirectStandardError = true
    };
    psi.EnvironmentVariables["TESSDATA_PREFIX"] = tessdataDir;


    using (Process proc = Process.Start(psi))
    {
        proc.StandardError.ReadToEnd();
        proc.WaitForExit();
    }


    if (!File.Exists(outputTxt))
    {
        return string.Empty;
    }


    string text = File.ReadAllText(outputTxt);
    File.Delete(outputTxt);
    return string.IsNullOrWhiteSpace(text) ? string.Empty : text.Trim();
}


private static bool IsMeaningfulText(string text)
{
    if (string.IsNullOrWhiteSpace(text)) return false;
    string trimmed = text.Trim();
    if (trimmed.Length < 3) return false;
    return Regex.IsMatch(trimmed, "[A-Za-z0-9]");
}


[WebMethod]
public static string ProcessVideoFrame(string videoPath, double timeInSeconds, int cropX, int cropY, int cropW, int cropH)
{
    // Integrated paths to your local binaries
    string ffmpegPath = @"C:\ffmpeg\bin\ffmpeg.exe"; // Adjust to your actual path or use "ffmpeg.exe" if in System PATH
    string tesseractPath;
    string tessdataDir;


    string tempDir = HttpContext.Current.Server.MapPath("~/App_Data/Temp/");
    if (!Directory.Exists(tempDir)) Directory.CreateDirectory(tempDir);


    string tempImg = Path.Combine(tempDir, "frame_" + Guid.NewGuid().ToString("N") + ".jpg");
    string tempImgRaw = Path.Combine(tempDir, "frame_raw_" + Guid.NewGuid().ToString("N") + ".png");
    string tempImgInv = Path.Combine(tempDir, "frame_inv_" + Guid.NewGuid().ToString("N") + ".png");
    string tempImgBw = Path.Combine(tempDir, "frame_bw_" + Guid.NewGuid().ToString("N") + ".png");
    string tempTxtBase = Path.Combine(tempDir, "ocr_" + Guid.NewGuid().ToString("N"));
    string tempTxtFile = tempTxtBase + ".txt";
    string safeVideoPath = null;
    string debugEnh = string.Empty;
    string debugRaw = string.Empty;
    string debugBw = string.Empty;
    string debugInv = string.Empty;
    string debugToken = Guid.NewGuid().ToString("N");


    // Verify paths exist before executing processes
    if (!File.Exists(ffmpegPath) && ffmpegPath != "ffmpeg.exe")
    {
        return "Server Error: FFmpeg executable not found at " + ffmpegPath;
    }


    if (!ResolveTesseractPaths(out tesseractPath, out tessdataDir))
    {
        return BuildOcrResultJson("Server Error: Could not locate Tesseract with eng.traineddata. Please install English language data.", "", "", "", "");
    }


    try
    {
        CleanupPreviousDebugFiles(tempDir);


        if (string.IsNullOrWhiteSpace(videoPath))
        {
            return BuildOcrResultJson("Error: Missing uploaded video path token.", "", "", "", "");
        }


        string videoToken = Path.GetFileName(videoPath);
        safeVideoPath = Path.Combine(tempDir, videoToken);
        if (!File.Exists(safeVideoPath))
        {
            return BuildOcrResultJson("Error: Uploaded video not found on server. Please reload the video.", "", "", "", "");
        }


        if (cropW < 2 || cropH < 2)
        {
            return BuildOcrResultJson("Error: OCR selection is too small. Please drag a larger area.", "", "", "", "");
        }


        cropX = Math.Max(0, cropX);
        cropY = Math.Max(0, cropY);


        // 1. FFmpeg Frame Extraction with OCR-friendly pre-processing
        string cropFilter = string.Format("crop={0}:{1}:{2}:{3},scale=iw*3:ih*3:flags=lanczos,format=gray,eq=contrast=1.9:brightness=0.06,unsharp=5:5:1.5", cropW, cropH, cropX, cropY);
        string ffmpegArgs = string.Format("-loglevel error -i \"{1}\" -ss {0} -vframes 1 -vf \"{2}\" -y \"{3}\"", 
            timeInSeconds.ToString("0.00", System.Globalization.CultureInfo.InvariantCulture), safeVideoPath, cropFilter, tempImg);
        string ffmpegErrorLogs = "";
        
        ProcessStartInfo ffInfo = new ProcessStartInfo(ffmpegPath, ffmpegArgs)
        {
            UseShellExecute = false,
            CreateNoWindow = true,
            RedirectStandardError = true
        };


        using (Process ffProc = Process.Start(ffInfo))
        {
            ffmpegErrorLogs = ffProc.StandardError.ReadToEnd();
            ffProc.WaitForExit();


            if (ffProc.ExitCode != 0)
            {
                string firstErrorLine = "Unknown FFmpeg error.";
                if (!string.IsNullOrWhiteSpace(ffmpegErrorLogs))
                {
                    using (StringReader sr = new StringReader(ffmpegErrorLogs.Trim()))
                    {
                        string line;
                        while ((line = sr.ReadLine()) != null)
                        {
                            string trimmed = line.Trim();
                            if (!string.IsNullOrWhiteSpace(trimmed))
                            {
                                firstErrorLine = trimmed;
                                break;
                            }
                        }
                    }
                }


                return BuildOcrResultJson("Error: FFmpeg failed to process video frame. " + firstErrorLine, "", "", "", "");
            }
        }


        if (!File.Exists(tempImg))
        {
            return BuildOcrResultJson("Error: FFmpeg failed to process video frame. Output image was not created.", "", "", "", "");
        }


        debugEnh = SaveDebugImage(tempDir, tempImg, "enh", debugToken);


        // 2. Multi-pass Tesseract OCR on enhanced frame
        string bestText = string.Empty;
        string extractedText = RunTesseractAndReadText(tesseractPath, tessdataDir, tempImg, tempTxtBase + "_enh6", "--oem 3 --psm 6 -l eng");
        if (IsMeaningfulText(extractedText)) bestText = extractedText;


        if (!IsMeaningfulText(bestText))
        {
            extractedText = RunTesseractAndReadText(tesseractPath, tessdataDir, tempImg, tempTxtBase + "_enh7", "--oem 3 --psm 7 -l eng");
            if (IsMeaningfulText(extractedText)) bestText = extractedText;
        }


        if (!IsMeaningfulText(bestText))
        {
            extractedText = RunTesseractAndReadText(tesseractPath, tessdataDir, tempImg, tempTxtBase + "_enh11", "--oem 3 --psm 11 -l eng");
            if (IsMeaningfulText(extractedText)) bestText = extractedText;
        }


        if (!IsMeaningfulText(bestText))
        {
            extractedText = RunTesseractAndReadText(tesseractPath, tessdataDir, tempImg, tempTxtBase + "_enh13", "--oem 3 --psm 13 -l eng");
            if (IsMeaningfulText(extractedText)) bestText = extractedText;
        }


        // 3. Fallback on raw crop and inverted crop for difficult contrast/subtitle cases
        if (!IsMeaningfulText(bestText))
        {
            string rawFilter = string.Format("crop={0}:{1}:{2}:{3}", cropW, cropH, cropX, cropY);
            string ffmpegRawArgs = string.Format("-loglevel error -i \"{1}\" -ss {0} -vframes 1 -vf \"{2}\" -y \"{3}\"",
                timeInSeconds.ToString("0.00", System.Globalization.CultureInfo.InvariantCulture), safeVideoPath, rawFilter, tempImgRaw);


            ProcessStartInfo ffRawInfo = new ProcessStartInfo(ffmpegPath, ffmpegRawArgs)
            {
                UseShellExecute = false,
                CreateNoWindow = true,
                RedirectStandardError = true
            };


            using (Process ffRawProc = Process.Start(ffRawInfo))
            {
                ffRawProc.StandardError.ReadToEnd();
                ffRawProc.WaitForExit();
            }


            if (File.Exists(tempImgRaw))
            {
                debugRaw = SaveDebugImage(tempDir, tempImgRaw, "raw", debugToken);
                extractedText = RunTesseractAndReadText(tesseractPath, tessdataDir, tempImgRaw, tempTxtBase + "_raw6", "--oem 3 --psm 6 -l eng");
                if (IsMeaningfulText(extractedText)) bestText = extractedText;
            }


            if (!IsMeaningfulText(bestText) && File.Exists(tempImgRaw))
            {
                string bwFilter = string.Format("crop={0}:{1}:{2}:{3},scale=iw*3:ih*3:flags=lanczos,format=gray,eq=contrast=2.4:brightness=0.08,lut=y='if(gte(val,150),255,0)'", cropW, cropH, cropX, cropY);
                string ffmpegBwArgs = string.Format("-loglevel error -i \"{1}\" -ss {0} -vframes 1 -vf \"{2}\" -y \"{3}\"",
                    timeInSeconds.ToString("0.00", System.Globalization.CultureInfo.InvariantCulture), safeVideoPath, bwFilter, tempImgBw);


                ProcessStartInfo ffBwInfo = new ProcessStartInfo(ffmpegPath, ffmpegBwArgs)
                {
                    UseShellExecute = false,
                    CreateNoWindow = true,
                    RedirectStandardError = true
                };


                using (Process ffBwProc = Process.Start(ffBwInfo))
                {
                    ffBwProc.StandardError.ReadToEnd();
                    ffBwProc.WaitForExit();
                }


                if (File.Exists(tempImgBw))
                {
                    debugBw = SaveDebugImage(tempDir, tempImgBw, "bw", debugToken);
                    extractedText = RunTesseractAndReadText(tesseractPath, tessdataDir, tempImgBw, tempTxtBase + "_bw7", "--oem 3 --psm 7 -l eng");
                    if (IsMeaningfulText(extractedText)) bestText = extractedText;
                }
            }


            if (!IsMeaningfulText(bestText) && File.Exists(tempImgRaw))
            {
                string invFilter = string.Format("crop={0}:{1}:{2}:{3},negate", cropW, cropH, cropX, cropY);
                string ffmpegInvArgs = string.Format("-loglevel error -i \"{1}\" -ss {0} -vframes 1 -vf \"{2}\" -y \"{3}\"",
                    timeInSeconds.ToString("0.00", System.Globalization.CultureInfo.InvariantCulture), safeVideoPath, invFilter, tempImgInv);


                ProcessStartInfo ffInvInfo = new ProcessStartInfo(ffmpegPath, ffmpegInvArgs)
                {
                    UseShellExecute = false,
                    CreateNoWindow = true,
                    RedirectStandardError = true
                };


                using (Process ffInvProc = Process.Start(ffInvInfo))
                {
                    ffInvProc.StandardError.ReadToEnd();
                    ffInvProc.WaitForExit();
                }


                if (File.Exists(tempImgInv))
                {
                    debugInv = SaveDebugImage(tempDir, tempImgInv, "inv", debugToken);
                    extractedText = RunTesseractAndReadText(tesseractPath, tessdataDir, tempImgInv, tempTxtBase + "_inv7", "--oem 3 --psm 7 -l eng");
                    if (IsMeaningfulText(extractedText)) bestText = extractedText;
                }
            }
        }


        if (string.IsNullOrWhiteSpace(debugRaw) && File.Exists(tempImgRaw)) debugRaw = SaveDebugImage(tempDir, tempImgRaw, "raw", debugToken);
        if (string.IsNullOrWhiteSpace(debugBw) && File.Exists(tempImgBw)) debugBw = SaveDebugImage(tempDir, tempImgBw, "bw", debugToken);
        if (string.IsNullOrWhiteSpace(debugInv) && File.Exists(tempImgInv)) debugInv = SaveDebugImage(tempDir, tempImgInv, "inv", debugToken);


        string[] latestFiles = new string[] { debugEnh, debugRaw, debugBw, debugInv };
        File.WriteAllLines(Path.Combine(tempDir, "ocrdbg_latest.txt"), latestFiles);


        string enhUrl = string.IsNullOrWhiteSpace(debugEnh) ? "" : "OCRDebugImage.ashx?f=" + HttpUtility.UrlEncode(debugEnh) + "&t=" + debugToken;
        string rawUrl = string.IsNullOrWhiteSpace(debugRaw) ? "" : "OCRDebugImage.ashx?f=" + HttpUtility.UrlEncode(debugRaw) + "&t=" + debugToken;
        string bwUrl = string.IsNullOrWhiteSpace(debugBw) ? "" : "OCRDebugImage.ashx?f=" + HttpUtility.UrlEncode(debugBw) + "&t=" + debugToken;
        string invUrl = string.IsNullOrWhiteSpace(debugInv) ? "" : "OCRDebugImage.ashx?f=" + HttpUtility.UrlEncode(debugInv) + "&t=" + debugToken;


        if (!IsMeaningfulText(bestText))
        {
            return BuildOcrResultJson("No text detected in selected region.", enhUrl, rawUrl, bwUrl, invUrl);
        }


        return BuildOcrResultJson(bestText, enhUrl, rawUrl, bwUrl, invUrl);
    }
    catch (Exception ex)
    {
        return BuildOcrResultJson("Server Error: " + ex.Message, "", "", "", "");
    }
    finally
    {
        if (File.Exists(tempImg)) File.Delete(tempImg);
        if (File.Exists(tempImgRaw)) File.Delete(tempImgRaw);
        if (File.Exists(tempImgInv)) File.Delete(tempImgInv);
        if (File.Exists(tempImgBw)) File.Delete(tempImgBw);
        if (File.Exists(tempTxtFile)) File.Delete(tempTxtFile);
    }
}

        function formatDurationMs(ms) {
            if (!isFinite(ms) || ms < 0) {
                return 'n/a';
            }


            if (ms < 1000) {
                return Math.round(ms) + ' ms';
            }


            return (ms / 1000).toFixed(1) + ' s';
        }

        function refreshOcrAverageDisplay() {
            if (!ocrDurationsMs.length) {
                updateOcrTimingDisplay('Idle', 'n/a', 'n/a');
                return;
            }


            let total = 0;
            for (let i = 0; i < ocrDurationsMs.length; i++) {
                total += ocrDurationsMs[i];
            }


            const averageMs = total / ocrDurationsMs.length;
            const lastMs = ocrDurationsMs[ocrDurationsMs.length - 1];
            updateOcrTimingDisplay('Idle', formatDurationMs(lastMs), formatDurationMs(averageMs));
        }


        function beginOcrTiming() {
            ocrRunStartedAt = Date.now();


            if (ocrTimingIntervalId) {
                clearInterval(ocrTimingIntervalId);
            }


            updateOcrTimingDisplay('Running... 0.0 s', ocrDurationsMs.length ? formatDurationMs(ocrDurationsMs[ocrDurationsMs.length - 1]) : 'n/a', ocrDurationsMs.length ? formatDurationMs(ocrDurationsMs.reduce(function (sum, value) { return sum + value; }, 0) / ocrDurationsMs.length) : 'n/a');


            ocrTimingIntervalId = window.setInterval(function () {
                const elapsedMs = Date.now() - ocrRunStartedAt;
                const currentNode = document.getElementById('ocrCurrentTiming');
                if (currentNode) {
                    currentNode.innerHTML = '<strong>Current OCR:</strong> Running... ' + formatDurationMs(elapsedMs);
                }
            }, 100);
        }


        function finishOcrTiming() {
            if (!ocrRunStartedAt) {
                refreshOcrAverageDisplay();
                return;
            }


            const elapsedMs = Date.now() - ocrRunStartedAt;
            ocrRunStartedAt = 0;


            if (ocrTimingIntervalId) {
                clearInterval(ocrTimingIntervalId);
                ocrTimingIntervalId = null;
            }


            ocrDurationsMs.push(elapsedMs);
            if (ocrDurationsMs.length > 10) {
                ocrDurationsMs.shift();
            }


            refreshOcrAverageDisplay();
        }


        function formatTimeHMS(totalSeconds) {
            const sec = Math.max(0, Math.floor(totalSeconds || 0));
            const h = Math.floor(sec / 3600);
            const m = Math.floor((sec % 3600) / 60);
            const s = sec % 60;
            const hh = String(h).padStart(2, '0');
            const mm = String(m).padStart(2, '0');
            const ss = String(s).padStart(2, '0');
            return hh + ':' + mm + ':' + ss;
        }


        function appendOcrLine(videoSeconds, text) {
            const box = document.getElementById('txtResult');
            let normalizedText = (text || '').replace(/\r\n/g, '\n');
            // Remove OCR-leading separators like "||" so lines stay "HH:MM:SS text".
            normalizedText = normalizedText.replace(/^\s*\|+\s*/, '');
            const line = formatTimeHMS(videoSeconds) + ' ' + normalizedText;
            if (box.value && box.value.length > 0) {
                box.value += '\n' + line;
            } else {
                box.value = line;
            }
            box.scrollTop = box.scrollHeight;
        }

        function resizeRoiHorizontally(deltaPerSide) {
            const box = document.getElementById('roiBox');
            const video = document.getElementById('videoPlayer');
            const wrapper = document.getElementById('videoWrapper');
            if (!box || !video || !wrapper || box.style.display === 'none' || roi.w <= 0 || roi.h <= 0) {
                return;
            }


            const wrapperRect = wrapper.getBoundingClientRect();
            const videoRect = video.getBoundingClientRect();
            const minLeft = Math.round(videoRect.left - wrapperRect.left);
            const maxRight = Math.round(videoRect.right - wrapperRect.left);
            const minWidth = 12;


            /* let newLeft = roi.x - deltaPerSide; */
            let newLeft = roi.x; 
            let newRight = roi.x + roi.w + deltaPerSide;


            newLeft = Math.max(minLeft, newLeft);
            newRight = Math.min(maxRight, newRight);


            if (newRight - newLeft < minWidth) {
                const center = roi.x + (roi.w / 2);
                newLeft = Math.max(minLeft, Math.round(center - (minWidth / 2)));
                newRight = Math.min(maxRight, newLeft + minWidth);
                newLeft = Math.max(minLeft, newRight - minWidth);
            }


            roi.x = newLeft;
            roi.w = Math.max(minWidth, newRight - newLeft);


            box.style.left = roi.x + 'px';
            box.style.width = roi.w + 'px';
            box.style.display = 'block';


            updateStatus('OCR box width: ' + roi.w + 'px');
        }

            if (!debugPanelVisible) {
                openDebugPanel();
                return;
            }


            debugPanelVisible = false;
            const debugGrid = document.getElementById('debugGrid');
            const btn = document.getElementById('btnToggleDebug');
            debugGrid.style.display = 'none';
            btn.innerText = 'Show OCR debug crops';
        }


        function setDebugMeta(metaId, text, isError) {
            const meta = document.getElementById(metaId);
            if (!meta) return;
            meta.className = isError ? 'debug-url error' : 'debug-url';
            meta.innerText = text || '';
        }


        function setDebugImage(imgId, metaId, src) {
            const img = document.getElementById(imgId);
            if (!img) return;
            if (src && src.length > 0) {
                img.onload = function() {
                    setDebugMeta(metaId, src, false);
                };
                img.onerror = function() {
                    setDebugMeta(metaId, 'Preview failed to load: ' + src, true);
                };
                img.src = src;
                img.style.display = 'block';
            } else {
                img.removeAttribute('src');
                img.onload = null;
                img.onerror = null;
                img.style.display = 'none';
                setDebugMeta(metaId, 'No image generated for this pass.', false);
            }
        }


        function renderDebugCrops(debug) {
            const empty = document.getElementById('debugEmpty');
            const hasAny = debug && (debug.enh || debug.raw || debug.bw || debug.inv);


            setDebugImage('dbgEnh', 'dbgEnhMeta', hasAny ? debug.enh : '');
            setDebugImage('dbgRaw', 'dbgRawMeta', hasAny ? debug.raw : '');
            setDebugImage('dbgBw', 'dbgBwMeta', hasAny ? debug.bw : '');
            setDebugImage('dbgInv', 'dbgInvMeta', hasAny ? debug.inv : '');


            if (hasAny) {
                empty.style.display = 'none';
            } else {
                empty.style.display = 'block';
            }
        }


        function loadLatestDebugCrops() {
            return fetch('OCRDebugList.ashx', { method: 'GET' })
                .then(res => res.json())
                .then(data => {
                    if (data && data.ok && data.debug) {
                        const hasAny = data.debug.enh || data.debug.raw || data.debug.bw || data.debug.inv;
                        if (hasAny) {
                            lastDebugPayload = data.debug;
                            renderDebugCrops(data.debug);
                        }
                    }
                })
                .catch(() => {
                    // Keep silent; panel fallback should not interrupt OCR flow.
                });
        }


        // WebMethod Ajax Invocation
        
    function executeOCR() {
        const video = document.getElementById('videoPlayer');
        if (!video.src) {
            alert('Please load a video first.');
            return;
        }


        if (!isVideoUploaded || !currentVideoPath) {
            alert('Video is not uploaded to the server yet. Please reload the video.');
            return;
        }


        if (!video.videoWidth || !video.videoHeight || !video.clientWidth || !video.clientHeight) {
            alert('Video metadata is not ready yet. Please wait a moment and try again.');
            return;
        }
    
        const wrapperRect = document.getElementById('videoWrapper').getBoundingClientRect();
        const videoRect = video.getBoundingClientRect();
        const videoOffsetX = videoRect.left - wrapperRect.left;
        const videoOffsetY = videoRect.top - wrapperRect.top;


        const scaleX = video.videoWidth / videoRect.width;
        const scaleY = video.videoHeight / videoRect.height;


        let roiXInVideo = roi.w > 0 ? (roi.x - videoOffsetX) : 0;
        let roiYInVideo = roi.h > 0 ? (roi.y - videoOffsetY) : 0;
        let roiWInVideo = roi.w > 0 ? roi.w : videoRect.width;
        let roiHInVideo = roi.h > 0 ? roi.h : videoRect.height;


        roiXInVideo = Math.max(0, Math.min(roiXInVideo, videoRect.width - 2));
        roiYInVideo = Math.max(0, Math.min(roiYInVideo, videoRect.height - 2));
        roiWInVideo = Math.max(2, Math.min(roiWInVideo, videoRect.width - roiXInVideo));
        roiHInVideo = Math.max(2, Math.min(roiHInVideo, videoRect.height - roiYInVideo));


        let cropW = Math.round(roiWInVideo * scaleX);
        let cropH = Math.round(roiHInVideo * scaleY);
        let cropX = Math.round(roiXInVideo * scaleX);
        let cropY = Math.round(roiYInVideo * scaleY);
    
        // Clamp crop rectangle to frame bounds
        cropX = Math.max(0, Math.min(cropX, video.videoWidth - 2));
        cropY = Math.max(0, Math.min(cropY, video.videoHeight - 2));
        cropW = Math.max(2, Math.min(cropW, video.videoWidth - cropX));
        cropH = Math.max(2, Math.min(cropH, video.videoHeight - cropY));
    
        // FFmpeg crop works more reliably with even dimensions
        if (cropW % 2 !== 0) cropW--;
        if (cropH % 2 !== 0) cropH--;
        if (cropW < 2) cropW = 2;
        if (cropH < 2) cropH = 2;
    
        updateStatus("FFmpeg extracting frame & running Tesseract...");
        beginOcrTiming();
        const ocrRequestedAtSeconds = video.currentTime;
    
        const payload = {
            videoPath: currentVideoPath,
            timeInSeconds: video.currentTime,
            cropX: cropX,
            cropY: cropY,
            cropW: cropW,
            cropH: cropH
        };
    
        fetch('OCR-Reader.aspx/ProcessVideoFrame', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json; charset=utf-8' },
            body: JSON.stringify(payload)
        })
        .then(res => res.json())
        .then(data => {
            let responseText = data && data.d ? data.d : '';
            let parsed = null;


            if (typeof responseText === 'string' && responseText.length > 0) {
                try {
                    parsed = JSON.parse(responseText);
                } catch (e) {
                    parsed = null;
                }
            }


            if (parsed && typeof parsed.text === 'string') {
                appendOcrLine(ocrRequestedAtSeconds, parsed.text);
                lastDebugPayload = parsed.debug || null;
                renderDebugCrops(lastDebugPayload);
                if (!debugPanelVisible) {
                    openDebugPanel();
                }
                if (parsed.text.indexOf('Error:') === 0 || parsed.text.indexOf('Server Error:') === 0) {
                    updateStatus(parsed.text);
                } else {
                    updateStatus('OCR processing complete.');
                }
                finishOcrTiming();
            } else {
                appendOcrLine(ocrRequestedAtSeconds, responseText);
                renderDebugCrops(null);
                if (!debugPanelVisible) {
                    openDebugPanel();
                }
                loadLatestDebugCrops();
                if (typeof responseText === 'string' && (responseText.indexOf('Error:') === 0 || responseText.indexOf('Server Error:') === 0)) {
                    updateStatus(responseText);
                } else {
                    updateStatus('OCR processing complete.');
                }
                finishOcrTiming();
            }
        })
        .catch(err => {
            finishOcrTiming();
            updateStatus("Error processing OCR frame.");
            console.error(err);
        });
    }
