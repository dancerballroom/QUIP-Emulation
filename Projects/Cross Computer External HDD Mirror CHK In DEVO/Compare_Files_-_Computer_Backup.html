01_mklink_Generator.bat

@Echo Off
setlocal enableDelayedExpansion

REM mklink Generator 
REM by Lawrence W. Davies
REM
REM Creates Symbolic Links for Jellyfin Home Movie Server Management
REM EXAMPLE:
REM mklink /j "C:\JELLYFIN_HOME\HDD-6\DOCU-SERIES" "K:\DOCU-SERIES"

REM CONSTANTS

REM SET $VIRTUAL-HOME=D:\JELLYFIN_HOME\HDD-1\
REM SET $HDD-DIR-ROOT=E:\

IF "%~1" == "" GOTO :Help

SET $VIRTUAL-HOME=%1
SET $HDD-DIR-ROOT=%2

REM Prep Batch FILE
Echo @Echo Off 
REM > $BATCH-FILE
Echo.

REM # PSEUDOCODE
REM Loop through DIRS -> 
REM     $CMD = $CMD + $DIRS + " "
REM     $CMD = $CMD + $HDD-DIR-ROOT\DIRS
REM End Loop 
pushd !$HDD-DIR-ROOT!
for /d %%a in (*) do (
    SET $CMD=mklink /j "!$VIRTUAL-HOME!%%a" 
    SET $CMD=!$CMD! "!$HDD-DIR-ROOT!%%a"
    Echo !$CMD!    
)
popd
goto :ENDER
:Help
Echo Purpose: Create a mklink List
Echo Recommedation: Pipe list into a batch file and run it.
Echo.
Echo USAGE:  
Echo 1. Create New Dir
Echo 2. mklink-Generator.bat VIRTUAL-HOME DRIVE-LETTER NEW-DIR
Echo.
Echo Example: 
Echo 01_mklink-Generator.bat (NO ARGS) for this HELP listing
Echo 01_mklink-Generator.bat D:\MyVirtualDir\ F:\ (pipe to) BATCH-FILE  
Echo Run BATCH-FILE to create the Jellyfin Virtual Links.
Echo.
pause
:ENDER

02_CreateDB.bat

@Echo Off
setlocal enabledelayedexpansion
REM ********* Series/!CNAME!_SEASONS *************
REM This Batch File:
REM Generate DB_CREATE_!CNAME!.bat

IF "%1"=="" (
    Set /p CNAME="Name of the computer : "
) ELSE (
    Set CNAME=%1
)

IF "%2"=="" (
    Set /p SUBDIR="Virtual name of Sub-directory : "
) ELSE (
    Set SUBDIR=%2
)

IF "%3"=="" (
    Set /p HDD="Hard Drive to Scan (F:, G:, etc.) : "
) ELSE (
    Set HDD=%3
)

Set NewPath=%~d0\
Set NewPath=!NewPath!%~p0\!SUBDIR!

Echo !CNAME!
Echo !SUBDIR!
Echo !NewPath!

pause
dir /B/AD/S !SUBDIR! | findstr /VI "metadata" | findstr /I "Season" > TEMP-ALL.txt
POWERSHELL -Command (get-content TEMP-ALL.txt) -replace '!NewPath!','!HDD!' > !CNAME!_SEASONS.txt
dir /B/AD/S !SUBDIR! | findstr /VI "metadata" | findstr /VI "Series" | findstr /VI "Season" > TEMP-ALL.txt
POWERSHELL -Command (get-content TEMP-ALL.txt) -replace '!NewPath!','!HDD!' > !CNAME!_MOVIES.txt

03_Add_md.bat

@echo off
setLocal EnableDelayedExpansion

set cmds=mkdir 
for /f "tokens=* delims= " %%a in (%1) do (
    rem set /a N+=1
    set dirs=%%a
    set drive=%%a
    set drive=!drive:~0,1!
    set dirs=!dirs:~3!
    REM echo !dirs!
    echo !cmds!^"!drive!\!dirs!^">>%~n1_md.bat
)

04_Add_rd.bat

@echo off
setLocal EnableDelayedExpansion

set cmds=rmdir 
for /f "tokens=* delims= " %%a in (%1) do (
    rem set /a N+=1
    set dirs=%%a
    set drive=%%a
    set drive=!drive:~0,1!
    set dirs=!dirs:~3!
    REM echo !dirs!
    echo !cmds!^^^"!drive!\\!dirs!^^^">>%~n1_rd.bat
)
