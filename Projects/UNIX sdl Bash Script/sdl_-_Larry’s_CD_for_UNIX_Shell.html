#  PseudoCode for (l)arry's (cd)see dee (sdl):
#  ------------------------------------
# if user does not ask for -help then 
#  while leave = no
#    clear screen
#    get the path
#    if number of files/directories > 400
#          give user an option to not view them (Very Slow otherwise)
#    else 
#          create menu 
#          Ask for user to enter the number to change directory to into variable $selection
#          Select Case ($Selection)
#              Case 1
#                  shell the command:  CD ..
#              Case (2 - Max Directories)
#                  shell the command:  CD directoryname
#              Case x
#                  exit this script
#              Case other
#                  try to shell command to the unix prompt
#          End Select
#    end if
#  end while
# else 
#  give user help message
# end if
#**************************************************************************
#  Instructions:
#  
#  To Install this Script Effectively, follow these instructions
#  1.  Save the script to a directory of your choice.
#  2.  Find out where the file is.  Type in 'pwd' at the UNIX prompt.  
#      Let's call this $FILEPATH (for this documentation only).
#  3.  In your root directory edit your .profile with any editor
#      (vi .profile) or (emacs .profile) or (pico .profile) or (vim .profile) etc.
#  4.  Place this line in your .profile:
#                      SYNTAX:  alias 1='. $FILEPATH/sdl'           (Don't forget the dot!!!)
#      HOME DIRECTORY EXAMPLE:  alias 1='. ~/sdl'
#       ABSOLUTE PATH EXAMPLE:  alias 1='. /user/s4/lwdavies/internet/CSCI057/script/sdl'
#  5.  Done!
#  6.  When you type in the number 1 plus ENTER it will now run the script in any directory.
# 
#  7.  NOTE:  Release of liability:  Feel Free to Use, Steal, Copy, Destroy, Mangle, or

#*************************************************************************
#  Limitations:  (4 that I know of)
#  1.  This script can really only handle 99 directories at present.
#  2.  This script can run arbitrary commands but not with "cd" from the Prompt
#  3.  CD .  and  CD .. both operate the  CD .. command
#  4.  Bug:  The Path at the Root has a double slash //.  
#
#  (Future Versions will include fixes for the 99 directory limitation)
#*************************************************************************
if [ "$1" != "-help" ] && [ "$1" != "-h" ] && \
   [ "$1" != "?" ] && [ "$1" != "-?" ] && \
   [ "$1" != "-HELP" ] && [ "$1" != "-H" ]           #  If help wasn't asked for do commands
then
leave=no                                             #  End Condition is set to NO
while [ $leave = no ]                                #  While ( USER doesn't want to leave )  
 do
   clear                                             #  Clear Screen
   pathinfo=`pwd`                                    #  Get Current Working Directory
   echo "--------[ CURRENT DIRECTORY ]--------"
   echo $pathinfo                                    #  Print Current Working Directory
   echo "---------[ SUB DIRECTORIES ]---------"
   count="1"                                         #  Count keeps track of Number of Directories
   totnum=`ls | wc -l`                               #  totnum = total number of Files and Directories

                                                     #  if number of files/directories < 400
                                                     #     Show the Main Menu
   if [ $totnum -lt '400' ]    
      then
      directories=`ls -a -p | grep /`                #  directories = names of all the directories

                                                     #  Create the Main Menu
      for fn in $directories
         do
                  # Using echo  to use Ansi Escape Sequences for highlight
                  #
                  echo  "\033[7m"`expr $count` "\033[0mCD "`basename ./"$fn"`""
                  #
                  count=`expr $count + 1`            #  Increment the Directory Count
         done  # End For 
 
       pn=`expr $count - 1`  
                                                     #  else if number of files/directories > 400
                                                     #     Give an option not to view the directories
   else
      pn=$totnum                                     #  pn stores the number to display at the bottom
                                                     #  ie:  -------------[ 4 ]----------------
      echo "Too many directories/files to display: (over 400)"
      echo "Do you want to display them anyway? (Y/N) "
#   if [ ${1} != "" ]
#   then
#      YN = ${1}
#      shift
#   else
      echo "Choice -> \c"
      read YN
#   fi
      #
      # Using echo  to use Ansi Escape Sequences to highlight the Numbers
      #
      #  NOTE:   echo  " \033[" is the first part of the escape sequence     
      #          echo  " \033[7m"  makes the following text inverted (White)
      #          echo  " \033[0m"  turns the highlight off.
      #          

      echo  "\033[7mb)\033[0m CD Return Back"      #  Display:  CD Return Back
      echo  "\033[7mf)\033[0m CD .."               #            CD..
      echo  "\033[7mh)\033[0m CD $HOME"            #            CD home
                                                     #  if the user wants to see the files anyway
                                                     #  then display them (Takes some time)
      if [ $YN = Y -o $YN = y ]
      then 
          directories=`ls -a -p | grep /`            #  directories = names of all the directories
          for fn in $directories
             do
                  #
                  # Using echo  to use Ansi Escape Sequences to highlight the Numbers
                  #
                  echo  "\033[7m"`expr $count` "\033[0mCD "`basename ./"$fn"`""
                  count=`expr $count + 1`            
                                                     #  Increment the Directory Count 
             done
             pn=`expr $count - 1`                    #  pn stores the number to display at the bottom
      fi  # End If       
   fi  # End If
   # Using echo  to use Ansi Escape Sequences for highlighting
   echo  "\033[7mx)\033[0mExit                           "
   echo "-------------[ $pn ]----------------"       #  ie:  -------------[ 4 ]----------------
   echo "Please enter your selection. "

#   if [ ${1} != "" ]
#   then
#      YN = ${1}
#      shift
#   else
             echo "Choice -> \c"
             read selection                         #  SELECT CASE
#   fi

   case $selection in
     1)                                              #  CD ..
    cd ..
    ;;
     f)
        cd ..
        ;;

     h)
        cd $HOME
        ;;

     b)
        cd $OLDPWD
        ;;

     x)
        currentdir=`pwd`                             #  EXIT the SCRIPT
        cd $currentdir
        leave=yes
    ;;

     *)
          chars=`expr $selection : '.*'`             #
                                                     #  chars = the number of characters entered by the user
                                                     #  This script only handles 98 directories.  ( 2 - 99 )
                                                     #  If the number of directories exceeds this, this script will
                                                     #  blow up.
                                                     #  Only the following 4 commands are excluded from the 
                                                     #  cd command:  "ls" "ps" "vi" "ll"
                                                     #  if you need more, just add to the lines below
                                                     #  && [ "$selection" != "??" ]
                                                     #
          if [ $chars -lt 3 ] && [ "$selection" != "ls" ] && [ "$selection" != "ps" ] \
                              && [ "$selection" != "vi" ] && [ "$selection" != "ll" ]
            then
            count="1"                                #  Start Directory count with Number 2
            NAME=`pwd`                               #  NAME = Current working Directory
                                                     #  Compare the $selection with the
                                                     #  Number associated with the directory
                                                     #  If they are equal operate the CD command
            # Check all directories
            for fn in $directories
                do
                      if [ $count -eq $selection ]
                      then
                             cd "$NAME/"$fn""        #  Main function of the Program
                      fi
                      count=`expr $count + 1`        #  Increment Count to next directory number

                done
                count="1"
          else
                                                     #  If the length of the entry is 2 or greater, as in
                                                     #  pwd, cat, emacs, vim, sed, grep, etc.
                                                     #  try to run the command
                                                     #  This is where a 3 digit directory would blow up.
            
            $selection                               #  Secondary function of the Program

            #
            # Using echo  to use Ansi Escape Sequences for highlight
            #
            echo  "\033[7mPress the Return Key to Continue \033[0m"
                                                     # Pause before clearing the screen
            read pauser
          fi  # End If
        ;;            
   esac  # End Select
 done  # End While
#  Help was asked for
else
    cat <<++
