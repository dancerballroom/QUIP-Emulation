@Echo Off
SETLOCAL enabledelayedexpansion
REM Show HELP if NOT MP4 DRAG/DROPPED or ARG1 missing
IF "%~1" == "" (
     CALL :Help
     EXIT
)

REM INITIALIZE math VARIABLES
REM             count=How many times looped
set /A count=!count!+1
REM             elapsed = TIME Elapsed between start and end
set /A elapsed=0

REM             TRUE = SHOW DEBUGGING 
set /A TRUE=1
REM             FALSE = HIDE DEBUGGING
set /A FALSE=0
REM SHOW/HIDE DEBUGGING
set /A DEBUG=!FALSE!
REM set /A DEBUG=!TRUE!

                REM Start Video to Sync Subtitles
Start /b "" "%~d1%~p1%~n1%~x1"

REM INITIALIZE string VARIABLES

                REM FIRST_TIME = INITIAL START_TIME 
SET FIRST_TIME=%TIME%
if !DEBUG! EQU !TRUE! ECHO FIRST_TIME String: !FIRST_TIME!
REM EXAMPLE:  FIRST_TIME String:  8:21:03.26

               REM Convert String to Number
CALL :CALC_FIRST_TIME !FIRST_TIME! firsttime
if !DEBUG! EQU !TRUE! Echo firsttime Number: !firsttime!
REM EXAMPLE:   firsttime Number: 3006426

REM LOAD FILE
SET File="%~d1%~p1%~n1.txt"
CALL :LOAD_TEXT output line_count !File!
SET SRT_NAME="%~d1%~p1%~n1-!firsttime!.srt"
copy "%~d1%~p1%~n1%~x1" "%~d1%~p1%~n1-!firsttime!%~x1"


REM CRLF=CARRIAGE RETURN
REM *** BELOW this line SET CRLF Leave 2 Lines Blank ***
SET CRLF=^


REM *** ABOVE this line SET CRLF Leave 2 Lines Blank ***

                REM The Great Loop (BEGIN)
                REM Do... Loop until Subtitles END
:BEGIN
REM Looping START_TIME string
REM START_TIME changed from STARTTIME
set START_TIME=%TIME%
CALL :CALC_START_TIME !START_TIME! starttime

REM Only at the Beginning, HIT ENTER TWICE
REM Once to Load Subtitle Text File
if !count! EQU 1 (
    SET /P wait="........[Hit ONCE to LOAD]..........."
)

REM Hit ENTER to MARK Subtitle TIMES
REM ROUND ROBIN START=END and Calculate NEW END 
if !count! GEQ 2 (
      SET /P wait="........[Hit Enter Once to MARK].............."
)
REM Looping END_TIME string
REM END_TIME changed from ENDTIME
CALL :LINE_NUMBER 73
set END_TIME=%TIME%
CALL :CALC_END_TIME !END_TIME! endtime
CALL :CALC_SRT_START_TIME srtstarttime 
CALL :CALC_SRT_END_TIME srtendtime
CALL :CALC_ELAPSED_TIME srtstarttime srtendtime elapsedtime
CALL :LINE_NUMBER 79

CALL :NUM2STR_START_TIME starttime START_TIME_STR
CALL :NUM2STR_END_TIME endtime END_TIME_STR
CALL :NUM2STR_ELAPSED_TIME elapsedtime ELAPSED_TIME_STR
CALL :NUM2STR_SRT_START_TIME srtstarttime SRT_START_TIME_STR
CALL :NUM2STR_SRT_END_TIME srtendtime SRT_END_TIME_STR

if !DEBUG! EQU !TRUE! (
    ECHO START_TIME String: !START_TIME!  COUNT: !COUNT!
    ECHO END_TIME String: !END_TIME!
    ECHO starttime Number: !starttime!  to STR: !START_TIME_STR!
    ECHO endtime Number: !endtime!  to STR: !END_TIME_STR!
    ECHO elapsedtime: !elapsedtime! to STR: !ELAPSED_TIME_STR!
    ECHO srtstarttime: !srtstarttime! to STR: !SRT_START_TIME_STR!
    ECHO srtendtime: !srtendtime! to STR: !SRT_END_TIME_STR!
    CALL :LINE_NUMBER 95
)
if !count! EQU 1 (
    SET File="%~d1%~p1%~n1.txt"
    CALL :LOAD_TEXT output line_count !File!
    SET SRT_NAME="%~d1%~p1%~n1-!firsttime!.srt"
    copy "%~d1%~p1%~n1%~x1" "%~d1%~p1%~n1-!firsttime!%~x1"
)

if !count! GEQ 2 (
    SET /A SUBTITLE_COUNT=!SUBTITLE_COUNT! + 1
    ECHO -------------------------
    ECHO !SUBTITLE_COUNT!  of !line_count!
    ECHO !SRT_START_TIME_STR! --^> !SRT_END_TIME_STR!
    CALL :SHOW_LINE output SUBTITLE_COUNT SRT_NAME line_count
    ECHO -------------------------
    REM MAIN ENGINE OF THIS Program
    ECHO !SUBTITLE_COUNT! >> !SRT_NAME!
    ECHO !SRT_START_TIME_STR! --^> !SRT_END_TIME_STR! >> !SRT_NAME!
    CALL :WRITE_LINE output SUBTITLE_COUNT SRT_NAME line_count
    ECHO. >> !SRT_NAME!
)


set /A count=!count!+1
     If !SUBTITLE_COUNT! EQU !line_count! (
          Echo End of Program
          Pause
          Exit
     )
     REM DEBUG
     REM If !SUBTITLE_COUNT! EQU 10 (
          REM Echo End of Program
          REM Pause
          REM Exit
     REM )

goto :BEGIN

REM *************************************************
REM *******        CALC NUMBER MATH            ******
REM *************************************************

REM Load the Subtitle Lines from the TEXT FILE
:LOAD_TEXT
     echo !File!
     set /a line_count=0
     set /a lines=0
     for /F "tokens=*" %%a in ('Type !File!') do (
         Set /a line_count=!line_count!+1
         Set output[!line_count!]=%%a
     )
     if !DEBUG! EQU !TRUE! Echo Total Lines:      !line_count!
Exit /b

:SHOW_LINE
REM Echo Line Number: !SUBTITLE_COUNT! of !line_count!
for %%i in (!SUBTITLE_COUNT!) do (
  set element=!output[%%i]!
  echo !element!
)
Exit /b

:WRITE_LINE
REM Echo Line Number: !SUBTITLE_COUNT! of !line_count!
for %%i in (!SUBTITLE_COUNT!) do (
  set element=!output[%%i]!
  echo !element! >> !SRT_NAME!
)
Exit /b

:CALC_SRT_START_TIME 
    CALL :LINE_NUMBER 167
    SET /A LAG=0
    if !count! GEQ 2 SET /A LAG=30
    REM if !count! GTR 1 (
    REM      SET /A LAG=10
    REM )
    REM if !count! EQU 1 (
         REM SET /A LAG=0
    REM )
    set /A srtstarttime=(!starttime! - !firsttime! - !LAG!)
Exit /b

:CALC_SRT_END_TIME
CALL :LINE_NUMBER 180
    SET /A JMP=0
    if !count! GTR 2 (
         SET /A JMP=60
    )
    set /A srtendtime=(!endtime! - !firsttime! + !JMP!)
Exit /b

:CALC_ELAPSED_TIME
CALL :LINE_NUMBER 189
REM    set /A elapsedtime=(!endtime! - !starttime!)
    set /A elapsedtime=(!srtendtime! - !srtstarttime!)
Exit /b

:CALC_START_TIME
CALL :LINE_NUMBER 195
REM MATHEFY start TIME
for /F "tokens=1-4 delims=:.," %%a in ("!START_TIME!") do (
    set /A "starttime=(((%%a*60)+1%%b %% 100)*60+1%%c %% 100)*100+1%%d %% 1000"
) 
Exit /b

REM end=NUMBER
:CALC_END_TIME
CALL :LINE_NUMBER 204

REM MATHEFY end TIME
REM if !DEBUG! EQU !TRUE! powershell -Command "Write-Host '1.. ' -NoNewline"
for /F "tokens=1-4 delims=:.," %%a in ("!END_TIME!") do (
    set /A "endtime=(((%%a*60)+1%%b %% 100)*60+1%%c %% 100)*100+1%%d %% 1000"
)
Exit /b

:CALC_FIRST_TIME
CALL :LINE_NUMBER 214
REM MATHEFY firsttime TIME
for /F "tokens=1-4 delims=:.," %%a in ("!FIRST_TIME!") do (
    set /A "firsttime=(((%%a*60)+1%%b %% 100)*60+1%%c %% 100)*100+1%%d %% 1000"
) 
Exit /b

REM *************************************************
REM *******        CALC STRING CONVERSION      ******
REM *************************************************
:NUM2STR_START_TIME 
CALL :LINE_NUMBER 225
set /A hh=starttime/(60*60*100), rest=starttime%%(60*60*100), mm=rest/(60*100), rest%%=60*100, ss=rest/100, cc=rest%%1000
if !hh! lss 10 set hh=0!hh!
if !mm! lss 10 set mm=0!mm!
if !ss! lss 10 set ss=0!ss!
if !cc! lss 10 set cc=0!cc!
set START_TIME_STR=!hh!:!mm!:!ss!,!cc!
Exit /b

:NUM2STR_END_TIME
CALL :LINE_NUMBER 235
set /A hh=endtime/(60*60*100), rest=endtime%%(60*60*100), mm=rest/(60*100), rest%%=60*100, ss=rest/100, cc=rest%%1000
if !hh! lss 10 set hh=0!hh!
if !mm! lss 10 set mm=0!mm!
if !ss! lss 10 set ss=0!ss!
if !cc! lss 10 set cc=0!cc!
set END_TIME_STR=!hh!:!mm!:!ss!,!cc!
Exit /b

:NUM2STR_ELAPSED_TIME
CALL :LINE_NUMBER 245
set /A hh=elapsedtime/(60*60*100), rest=elapsedtime%%(60*60*100), mm=rest/(60*100), rest%%=60*100, ss=rest/100, cc=rest%%1000
if !hh! lss 10 set hh=0!hh!
if !mm! lss 10 set mm=0!mm!
if !ss! lss 10 set ss=0!ss!
if !cc! lss 10 set cc=0!cc!
set ELAPSED_TIME_STR=!hh!:!mm!:!ss!,!cc!
Exit /b

:NUM2STR_SRT_START_TIME 
CALL :LINE_NUMBER 255
REM srtstarttime SRT_START_TIME_STR
set /A hh=srtstarttime/(60*60*100), rest=srtstarttime%%(60*60*100), mm=rest/(60*100), rest%%=60*100, ss=rest/100, cc=rest%%1000
if !hh! lss 10 set hh=0!hh!
if !mm! lss 10 set mm=0!mm!
if !ss! lss 10 set ss=0!ss!
if !cc! lss 10 set cc=0!cc!
set SRT_START_TIME_STR=!hh!:!mm!:!ss!,!cc!
Exit /b

:NUM2STR_SRT_END_TIME 
CALL :LINE_NUMBER 266
REM srtendtime SRT_END_TIME_STR
set /A hh=srtendtime/(60*60*100), rest=srtendtime%%(60*60*100), mm=rest/(60*100), rest%%=60*100, ss=rest/100, cc=rest%%1000
if !hh! lss 10 set hh=0!hh!
if !mm! lss 10 set mm=0!mm!
if !ss! lss 10 set ss=0!ss!
if !cc! lss 10 set cc=0!cc!
set SRT_END_TIME_STR=!hh!:!mm!:!ss!,!cc!
Exit /b

:LINE_NUMBER
     if !DEBUG! EQU !TRUE! powershell -Command "Write-Host '%1.. ' -NoNewline"
Exit /b

:Help
CALL :LINE_NUMBER 281
Echo Purpose: Quickly Generate .SRT (Subtitle) file for Video
Echo USAGE:  
Echo Drag/Drop VIDEO_NAME.MP4 that has corresponding VIDEO_NAME.TXT
Echo 1. VIDEO_NAME.MP4 Plays
Echo 2. Press ENTER Twice to Start Generating 
Echo    * VIDEO_NAME_TIME-STAMP.SRT
Echo    * Copys VIDEO_NAME.MP4 to VIDEO_NAME_TIME-STAMP.mp4
Echo.
Echo 3. As Video Plays 
Echo    * PRESS ENTER each time you hear each line of TEXT 
Echo.
Echo Example: 
Echo    *  TXT2SRT-TAP.bat (NO ARGS) for this HELP listing
Echo    *  TXT2SRT-TAP.bat VIDEO_NAME.MP4  
Echo           Copies VIDEO_NAME.MP4 to VIDEO_NAME_TIME-STAMP.MP4
Echo           Creates VIDEO_NAME_TIME-STAMP.SRT
Echo    *  The program terminates when the last lyric/TEXT/LINE is Reached.
Echo.
Echo Produces:
Echo    *  Playable VIDEO_NAME_TIME-STAMP.MP4 and VIDEO_NAME_TIME-STAMP.SRT
Echo    *  The final SRT will need adjustment using SUBTITLE EDIT
pause
Exit /b
