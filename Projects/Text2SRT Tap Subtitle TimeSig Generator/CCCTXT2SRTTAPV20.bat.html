CCC_TXT2SRT_TAP_V20.bat

Program:  Create Subtitle “File.srt”
Requires:
  • “File.mp4” Movie
  • “File.txt”  FLAT FILE Lyrics without Time Signatures
  • CCC_TXT2SRT_TAP_V20.bat
Procedure:
  • DRAG/DROP “File.mp4” on to CCC_TXT2SRT_TAP_V20.bat
  • After Program starts Playing “File.mp4”, Manually click back to CMD Window to Give Focus
  • FIRST TIME: Click ONCE at beginning of Singing or Talking
  • FOR-EACH New Set of 2 LINES: Click ONCE at the end of Singing or Talking.
  • At End of Song, Click ONCE to establish the LAST Termination Time
  • Click ONCE to END Program
  • OPEN “File.mp4 with ”Subtitle Edit” and make Synchronization adjustment.
  • DONE!

CCC_TXT2SRT_TAP_V20.bat

@Echo Off
SETLOCAL enabledelayedexpansion

REM Show HELP if NOT MP4 DRAG/DROPPED or ARG1 missing
IF "%~1" == "" (
     CALL :Help
     EXIT
)

REM #            TRUE = SHOW DEBUGGING 
set /A TRUE=1
REM #            FALSE = HIDE DEBUGGING
set /A FALSE=0
REM # SHOW/HIDE DEBUGGING
set /A DEBUG=!FALSE!

SET FIRST_TIME=%TIME%

REM # LOAD Subtitle TEXT-ONLY-FILE
SET File="%~d1%~p1%~n1.txt"
CALL :LOAD_TEXT output line_count !File!
SET /A TEST_LIMIT=(!line_count!/2)+2

REM # *** BELOW this line SET CRLF Leave 2 Lines Blank ***
SET CRLF=^


REM # *** ABOVE this line SET CRLF Leave 2 Lines Blank ***
set /A count=!count!+1  
REM Loop Counter

CALL :CALC_FIRST_TIME !FIRST_TIME! firsttime
CALL :NUM2STR_FIRST_TIME firsttime FIRST_TIME_STR
Echo TIME: !FIRST_TIME!
Echo firsttime Number: !firsttime!
Echo firsttime String: !FIRST_TIME_STR!

Start /b "" "%~d1%~p1%~n1%~x1"

REM # COPY NAME-SESSION.MP4
REM # OUTPUT-FILE: NAME-SESSION.SRT
SET SRT_NAME="%~d1%~p1%~n1-!firsttime!.srt"
copy "%~d1%~p1%~n1%~x1" "%~d1%~p1%~n1-!firsttime!%~x1"

:BEGIN
if !DEBUG! EQU !TRUE! Echo BEGIN

REM # Load LINES 2 at a Time
if !count! GEQ 2 (
    SET /A MODULUS="!count! %% 2"
    REM # PROCESS after LINE1 Start and LINE2 End Processed
    if !MODULUS! == 0 (    
        SET NEXT_TIME=%TIME%
        CALL :CALC_NEXT_TIME !NEXT_TIME! nexttime
        CALL :NUM2STR_NEXT_TIME nexttime NEXT_TIME_STR
        CALL :CALC_SRT_NEXT_TIME nexttime firsttime srtnexttime
        CALL :NUM2STR_SRT_DIFFERENCE_TIME srtnexttime SRT_DIFFERENCE_TIME_STR
        SET /A DUAL_COUNT=!DUAL_COUNT! + 1
        SET /A Minus_One=!DUAL_COUNT!
        set SUBT[!Minus_One!][1]=!SRT_DIFFERENCE_TIME_STR!
        Set /A LINE_1=!SUBTITLE_COUNT! + 1
        for %%i in (!LINE_1!) do (
            set element=!output[%%i]!
            set SUBT[!DUAL_COUNT!][2]=!element!
        )
        Set /A LINE_2=!SUBTITLE_COUNT! + 2
        for %%i in (!LINE_2!) do (
            set element=!output[%%i]!
            set SUBT[!DUAL_COUNT!][3]=!element!
        )
        
        for /l %%n in (!DUAL_COUNT!,1,!DUAL_COUNT!) do (
             REM echo SUBT %%n: !SUBT[%%n][1]!
             echo ********* %%n of !TEST_LIMIT! **********
             echo SUBT %%n: !SUBT[%%n][2]!
             echo SUBT %%n: !SUBT[%%n][3]!
        )
        SET /A SUBTITLE_COUNT=!SUBTITLE_COUNT! + 1
        SET /a SUBTITLE_NEXT=!SUBTITLE_COUNT! + 1
        SET /P wait="........[ READ Next 2 Lines ].........."
        echo ********* CURRENT ABOVE **********
        SET /A SUBTITLE_COUNT=!SUBTITLE_COUNT! + 1
    )
)
SET /A count=!count! + 1
if !DUAL_COUNT! EQU !TEST_LIMIT! (
        set /A TEST_LIMIT_MINUS_ONE = !TEST_LIMIT! - 1
        for /l %%m in (1,1,!TEST_LIMIT_MINUS_ONE!) do (
             Set /A Next_N = %%m + 1
             if %%m lss !TEST_LIMIT_MINUS_ONE! (Echo %%m)
             if %%m lss !TEST_LIMIT_MINUS_ONE! (Echo %%m >> !SRT_NAME!)
             for /l %%n in (!Next_N!,1,!Next_N!) do (
                if %%n lss !TEST_LIMIT! (
                    echo !SUBT[%%m][1]! --^> !SUBT[%%n][1]!
                    echo !SUBT[%%m][1]! --^> !SUBT[%%n][1]! >> !SRT_NAME!
                )
            )
            if %%m lss !TEST_LIMIT_MINUS_ONE! (echo !SUBT[%%m][2]!)
            if %%m lss !TEST_LIMIT_MINUS_ONE! (echo !SUBT[%%m][2]! >> !SRT_NAME!)
            if %%m lss !TEST_LIMIT_MINUS_ONE! (echo !SUBT[%%m][3]!)
            if %%m lss !TEST_LIMIT_MINUS_ONE! (echo !SUBT[%%m][3]! >> !SRT_NAME!)
            echo.
            echo. >> !SRT_NAME!
        )
        Echo End of Program...
    pause
    SET /P wait="........[ EXITING PROGRAM ].........."
    Exit
)

REM # END-MAIN-LOOP
GOTO :BEGIN

REM # FUNCTIONS SECTION
:CALC_FIRST_TIME
if !DEBUG! EQU !TRUE! ECHO CALC_FIRST_TIME
for /F "tokens=1-4 delims=:.," %%a in ("!FIRST_TIME!") do (
    set /A "firsttime=(((%%a*60)+1%%b %% 100)*60+1%%c %% 100)*100+1%%d %% 100"
) 
Exit /b

:CALC_START_TIME
if !DEBUG! EQU !TRUE! ECHO CALC_START_TIME
for /F "tokens=1-4 delims=:.," %%a in ("!START_TIME!") do (
    set /A "starttime=(((%%a*60)+1%%b %% 100)*60+1%%c %% 100)*100+1%%d %% 100"
    REM if !starttime! GEQ 235 (set /A starttime=!starttime! - 235)
) 
Exit /b

:CALC_NEXT_TIME
if !DEBUG! EQU !TRUE! ECHO CALC_NEXT_TIME
for /F "tokens=1-4 delims=:.," %%a in ("!NEXT_TIME!") do (
    set /A "nexttime=(((%%a*60)+1%%b %% 100)*60+1%%c %% 100)*100+1%%d %% 100"
    REM if !nexttime! GEQ 235 (set /A nexttime=!nexttime! - 235)
) 
Exit /b

:LOAD_TEXT
if !DEBUG! EQU !TRUE! ECHO LOAD_TEXT
     echo !File!
     set /a line_count=0
     set /a lines=0
     for /F "tokens=*" %%a in ('Type !File!') do (
         Set /a line_count=!line_count!+1
         Set output[!line_count!]=%%a
     )
     Echo Total Lines (line_count):      !line_count!
Exit /b

:SHOW_LINE
if !DEBUG! EQU !TRUE! ECHO SHOW_LINE
for %%i in (!SUBTITLE_COUNT!) do (
    set element=!output[%%i]!
    echo !element!
    set /a SECOND=!SUBTITLE_COUNT!+1
    for %%j in (!SECOND!) do (
      set element2=!output[%%j]!
    )
    echo !element2!
)
Exit /b

:NUM2STR_FIRST_TIME 
if !DEBUG! EQU !TRUE! ECHO NUM2STR_FIRST_TIME
set /A hh=firsttime/(60*60*100), rest=firsttime%%(60*60*100), mm=rest/(60*100), rest%%=60*100, ss=rest/100, cc=rest%%100
if !hh! lss 10 set hh=0!hh!
if !mm! lss 10 set mm=0!mm!
if !ss! lss 10 set ss=0!ss!
if !cc! lss 10 set cc=0!cc!
set FIRST_TIME_STR=!hh!:!mm!:!ss!,!cc!
Exit /b

:NUM2STR_START_TIME 
if !DEBUG! EQU !TRUE! ECHO NUM2STR_START_TIME 
set /A hh=starttime/(60*60*100), rest=starttime%%(60*60*100), mm=rest/(60*100), rest%%=60*100, ss=rest/100, cc=rest%%100
if !hh! lss 10 set hh=0!hh!
if !mm! lss 10 set mm=0!mm!
if !ss! lss 10 set ss=0!ss!
if !cc! lss 10 set cc=0!cc!
set START_TIME_STR=!hh!:!mm!:!ss!,!cc!
Exit /b

:NUM2STR_NEXT_TIME
if !DEBUG! EQU !TRUE! ECHO NUM2STR_NEXT_TIME
set /A hh=nexttime/(60*60*100), rest=nexttime%%(60*60*100), mm=rest/(60*100), rest%%=60*100, ss=rest/100, cc=rest%%100
if !hh! lss 10 set hh=0!hh!
if !mm! lss 10 set mm=0!mm!
if !ss! lss 10 set ss=0!ss!
if !cc! lss 10 set cc=0!cc!
set NEXT_TIME_STR=!hh!:!mm!:!ss!,!cc!
Exit /b

:CALC_SRT_START_TIME 
if !DEBUG! EQU !TRUE! ECHO CALC_SRT_START_TIME 
    set /A srtstarttime=(!starttime! - !firsttime!) 
Exit /b

:CALC_SRT_NEXT_TIME
if !DEBUG! EQU !TRUE! ECHO CALC_SRT_NEXT_TIME 
    set /A srtnexttime=(!nexttime! - !firsttime!) 
Exit /b

:NUM2STR_SRT_START_TIME 
if !DEBUG! EQU !TRUE! ECHO NUM2STR_SRT_START_TIME
set /A hh=srtstarttime/(60*60*100), rest=srtstarttime%%(60*60*100), mm=rest/(60*100), rest%%=60*100, ss=rest/100, cc=rest%%100
if !hh! lss 10 set hh=0!hh!
if !mm! lss 10 set mm=0!mm!
if !ss! lss 10 set ss=0!ss!
if !cc! lss 10 set cc=0!cc!
set SRT_START_TIME_STR=!hh!:!mm!:!ss!,!cc!
Exit /b

:NUM2STR_SRT_DIFFERENCE_TIME 
if !DEBUG! EQU !TRUE! ECHO NUM2STR_SRT_DIFFERENCE_TIME 
set /A hh=srtnexttime/(60*60*100), rest=srtnexttime%%(60*60*100), mm=rest/(60*100), rest%%=60*100, ss=rest/100, cc=rest%%100
if !hh! lss 10 set hh=0!hh!
if !mm! lss 10 set mm=0!mm!
if !ss! lss 10 set ss=0!ss!
if !cc! lss 10 set cc=0!cc!
set SRT_DIFFERENCE_TIME_STR=!hh!:!mm!:!ss!,!cc!
Exit /b

:Help
if !DEBUG! EQU !TRUE! ECHO HELP
Echo # Purpose: Quickly Generate .SRT (Subtitle) file for Video
Echo # USAGE:  
Echo Drag/Drop VIDEO_NAME.MP4 that has corresponding VIDEO_NAME.TXT
Echo 1. VIDEO_NAME.MP4 Plays
Echo 2. Press ENTER ONCE to Start Generating 
Echo    * VIDEO_NAME_TIME-STAMP.SRT
Echo    * Copys VIDEO_NAME.MP4 to VIDEO_NAME_TIME-STAMP.mp4
Echo.
Echo 3. As Video Plays 
Echo    * PRESS ENTER each time you hear each PAIR of LINES of TEXT 
Echo.
Echo Example: 
Echo    *  TXT2SRT-TAP.bat (NO ARGS) for this HELP listing
Echo    *  TXT2SRT-TAP.bat VIDEO_NAME.MP4  
Echo           Copies VIDEO_NAME.MP4 to VIDEO_NAME_TIME-STAMP.MP4
Echo           Creates VIDEO_NAME_TIME-STAMP.SRT
Echo    *  The program terminates when the BLANK line of lyric/TEXT/LINES is Reached.
Echo.
Echo # Produces:
Echo    *  Playable VIDEO_NAME_TIME-SESSION.MP4 and VIDEO_NAME_TIME-SESSION.SRT
Echo #  *  Required: Needs adjustment using SUBTITLE EDIT
pause
Exit /b

Example RUN:
