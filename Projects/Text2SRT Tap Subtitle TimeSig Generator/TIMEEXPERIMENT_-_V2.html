TIME_EXPERIMENT.bat

@Echo Off
SETLOCAL enabledelayedexpansion

set STARTTIME=%TIME%
Echo Play a VID 

REM To Be Added
REM Start /b "" "%~d1%~p1%~n1%~x1"

REM Create Time Running Differentiation
Pause

set ENDTIME=%TIME%

REM Function(al) Heaven
Call :CALC_START_TIME STARTTIME start
Call :CALC_END_TIME ENDTIME end  

Call :FORMAT_START_TIME STARTTIME ENDTIME hh mm ss cc start STARTTIME_FORMATTED
Call :FORMAT_END_TIME STARTTIME ENDTIME ehh emm ess ecc end ENDTIME_FORMATTED

Call :CALC_ELAPSED_TIME start end elapsed
CALL :FORMAT_ELAPSED_TIME elapsed
echo Last Statement
Echo !STARTTIME_FORMATTED! --^> !ENDTIME_FORMATTED!
Pause
Exit /b

:CALC_START_TIME
REM MATHEFY TIME
for /F "tokens=1-4 delims=:.," %%a in ("!STARTTIME!") do (
    set /A "start=(((%%a*60)+1%%b %% 100)*60+1%%c %% 100)*100+1%%d %% 100"
) 
Echo Start: !start!
Pause
Exit /b

:CALC_END_TIME
REM MATHEFY TIME
for /F "tokens=1-4 delims=:.," %%a in ("!ENDTIME!") do (
    set /A "end=(((%%a*60)+1%%b %% 100)*60+1%%c %% 100)*100+1%%d %% 100"
) 
Echo End: !end!
Pause
Exit /b

:FORMAT_START_TIME 

REM Calculate and RETURN FUnction Values HH,MM,SS, and CC
set /A hh=start/(60*60*100), rest=start%%(60*60*100), mm=rest/(60*100), rest%%=60*100, ss=rest/100, cc=rest%%100
if !hh! lss 10 set hh=0!hh!
if !mm! lss 10 set mm=0!mm!
if !ss! lss 10 set ss=0!ss!
if !cc! lss 10 set cc=0!cc!
set STARTTIME_FORMATTED=!hh!:!mm!:!ss!,!cc!0
echo START(Formatted): !STARTTIME_FORMATTED!
pause

Exit /b

:FORMAT_END_TIME 
set /A ehh=end/(60*60*100), rest=end%%(60*60*100), emm=rest/(60*100), rest%%=60*100, ess=rest/100, ecc=rest%%100
REM Calculate and RETURN FUnction Values HH,MM,SS, and CC
if !ehh! lss 10 set ehh=0!ehh!
if !emm! lss 10 set emm=0!emm!
if !ess! lss 10 set ess=0!ess!
if !ecc! lss 10 set ecc=0!ecc!
set ENDTIME_FORMATTED=!ehh!:!emm!:!ess!,!ecc!0

echo END(Formatted): !ENDTIME_FORMATTED!
pause

Exit /b

:CALC_ELAPSED_TIME
set /A elapsed=!end! - !start! 
echo Start  : !start!
echo End    : !end!
echo Elapsed: !elapsed!
pause
Exit /b

:FORMAT_ELAPSED_TIME
set /A ethh=!elapsed!/(60*60*100), rest=!elapsed!%%(60*60*100), etmm=rest/(60*100), rest%%=60*100, etss=rest/100, etcc=rest%%100
if !ethh! lss 10 set ethh=0!ethh!
if !etmm! lss 10 set etmm=0!etmm!
if !etss! lss 10 set etss=0!etss!
if !etcc! lss 10 set etcc=0!etcc!
set ELAPSED_TIME=!ethh!:!etmm!:!etss!,!etcc!0
echo ELAPSED_TIME(Formatted) = !ELAPSED_TIME!
pause

Exit /b
