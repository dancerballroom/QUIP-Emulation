dircmp.bat version 2.0

@echo off
SET ARG0=%~1
SET ARG1=%~2

Echo Drives to Compare for Missing Mirrored Backup Files
IF Defined ARG0 Echo.&powershell write-host -back Black -fore White  First  DIR:%ARG0%
IF Defined ARG1 powershell write-host -back Black -fore White  Second DIR:%ARG1%&Echo.

if Not Defined ARG0 GOTO :Usage
if Not Defined ARG1 GOTO :Usage

pause
echo.&powershell write-host -back Black -fore White Processing...&Echo.
PushD %ARG0%
Echo Checking Drive for any files not mirrored in 
powershell write-host -back Black -fore Green %ARG1% 
Echo.
powershell write-host -back Black -fore White Missing Files
for %%x in (*.*) do (
    if NOT exist "%ARG1%\%%x" (
        powershell write-host -back Black -fore Cyan "%ARG0%\%%x"
    )
)
Echo.
PopD
powershell write-host -back Black -fore Blue 1/2 Way There...&Echo.
pause
Echo.&Echo Checking Drive for any files not mirrored in 
powershell write-host -back Black -fore Green %ARG0% 

PushD %ARG1%
Echo.
powershell write-host -back Black -fore White Missing Files
for %%x in (*.*) do (
    if NOT exist "%ARG0%\%%x" (        
        powershell write-host -back Black -fore Yellow "%ARG1%\%%x"
    )
)
Echo.
PopD
powershell write-host -back Black -fore Green DONE&Echo.
pause

goto end

:Usage
echo.&Echo Usage: 
powershell write-host -back Black -fore Cyan %0 dir1 dir2
powershell write-host -back Black -fore White [ where dir1 and dir2 are full paths ]
echo.
:end
