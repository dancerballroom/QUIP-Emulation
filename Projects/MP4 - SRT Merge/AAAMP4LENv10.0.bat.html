AAA_MP4_LEN_v10.0.bat

Purpose: Merge SRT with corrected TIME-STAMPS to compliment Merged MP4
# This works almost perfectly
# Requires some post RUN adjustments with SUBTITLE EDIT

AAA_MP4_LEN_v10.0.bat

@Echo Off
setlocal enableDelayedExpansion

REM Requires VIDEO.mp4
REM          VIDEO.srt
REM (Processes all VIDEO.srt in Current Directory)

REM Create Output SUB Directory "DONE_SRT"
if NOT exist "DONE_SRT" (
   echo DONE_SRT DNE
   mkdir DONE_SRT
) else (
   echo Oh, yeah, DONE_SRT Exists
)

set count=0
Set TotSecs=0
Set TotOverflow=0
for %%F in (*.mp4) do (
   for /f "tokens=2 delims==" %%a in (
      'ffprobe "%%F" -show_entries format^=duration -sexagesimal -v quiet -of compact'
    ) do (
      set /a count=!count! + 1
      echo Count: !count!

      REM Filename.srt
      echo %%~nF.srt

      REM The IF Statement that does all the work
      if !Count! equ 1 (
           type "%~d0%~p0%%~nF.srt" > "%~d0%~p0DONE_SRT\FINAL.srt"
           Set /a TotSecs=0
      ) else (
           REM Delete Tempfile
           del /q "%~d0%~p0DONE_SRT\TEMP.srt"
           REM Change Start time according to Placement in Merged MP4 in Tempfile 
           ffmpeg -itsoffset !TotSecs!.!TotOverflow_milisecond! -i "%~d0%~p0%%~nF.srt" -c copy "%~d0%~p0DONE_SRT\TEMP.srt"
           REM Merge Tempfile to END of FINAL.srt
           type "%~d0%~p0DONE_SRT\TEMP.srt" >> "%~d0%~p0DONE_SRT\FINAL.srt"   
      )

      REM Calculate offset of next song in FINAL.srt
      Set TIMESIG=0%%a

      REM Chap off the Milliseconds .MMMMMM after the decimal point
      Set /a TIMESIG_milisecond=!TIMESIG:~-6!

      REM Chop off the Time HH:MM:SS before the decimal point
      Set TIMESIG=!TIMESIG:~0,-7!

      REM Calculate SECONDS
      for /F "tokens=1-3 delims=:" %%a in ("!TIMESIG!") do (
            set /A "Sec=((1%%a-100)*60+1%%b-100)*60+1%%c-100"
      )
      REM Debug
      echo !TIMESIG!
      echo Cur Secs=!Sec!
      REM Calculate Total Seconds (Cumulator)
      Set /a TotSecs=!TotSecs!+!Sec!
      REM Calculate Milliseconds (Cumulator)
      Set /a TotOverflow=!TotOverflow!+!TIMESIG_milisecond!

      REM Manually add Milliseconds, if CARRY then add 1 second to TotSecs
      Set /a TotOverflow_milisecond=!TotOverflow_milisecond:~-6!
      if !TotOverflow! gtr 999999 (
           echo TotOverflow = YES
           set /a TotSecs = !TotSecs! + 1
      )
      
      REM Debug
      Echo Tot Secs : !TotSecs!
      Echo Seconds: !Sec!.!TIMESIG_milisecond!
      
      REM !TotSecs! is correct
      REM !TotOverflow_milisecond! becomes .5 = Not correct but close
      Echo Total Seconds: !TotSecs!.!TotOverflow_milisecond!

      echo ==========
    )
)
pause
