@echo off

:: Uses %1 and %2
Set CurFile=%~n1%~x1
Set OutFile=%~n2%~x2

:: Build=function Copy2Clipboard using JavaScript
for /f "Tokens=* Delims=" %%x in (atom-copy-2-clipboard-function.txt) do (set Build=!Build!%%x)

:: Huge Find - Replace \xc2\xa0 with &nbsp;  - Output to %2 File
:: This works
SET CMD1="(gc !CurFile!) -replace '\xc2\xa0','&nbsp;'"
Powershell -Command %CMD1% > !OutFile!

Step 2:  Identify ID (Success)

Continuing from Above

REM Keep track of the line numbers
SET /a count=1
for /f "tokens=*" %%s in (!OutFile2!) do (
    IF !count! GEQ 400 (
        REM echo B4: %%s
        REM pause
        echo."%%s" | findstr /C:"<pre" 1>nul
        if errorlevel 1 (
            REM Keep this line here
        ) ELSE (
            set INPUT-STRING=%%s

            Set DIV=!INPUT-STRING:~9,11!

            REM DIV now has the ID of the current PRE
            echo This is DIV ID at Line !count!: !DIV!
            echo. 
        ) 
    )
    SET /a count+=1
)
pause
