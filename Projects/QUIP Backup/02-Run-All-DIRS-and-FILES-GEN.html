02-Run-All-DIRS-and-FILES-GEN.bat

@Echo Off
SETLOCAL ENABLEDELAYEDEXPANSION
set dir-tree=
(set \n=^
%=Do not remove this line=%
)

for /r "%~d0%~p0" %%a in (.) do (
    set curdir_long=%%a

    REM Chop off last character
    SET curdir=^"!curdir_long:^\.=!^"
    pushd %%a

    echo !curdir!
    if exist run-file.txt (
        set dir-tree=!dir-tree!!^n!curdir!
        echo Directory: !curdir!
        echo run-FILE!    
        Call 000_HTML_FILES_GEN_main.bat
    ) 
    if exist run-dir.txt (
        set dir-tree=!dir-tree!!^n!curdir!
        echo Directory: !curdir!
        echo run DIR!
        Call 000_HTML_DIRS_GEN_main.bat
    ) 
    popd 
)
:ender
pause
