SETLOCAL enabledelayedexpansion

:: HTML
SET START-HTML-SCRIPT=^<SCRIPT^>
SET END-HTML-SCRIPT=^</SCRIPT^>

SET BUILD=!START-HTML-SCRIPT!
SET BUILD=!BUILD! function CopyToClipboard(containerid) {
SET BUILD=!BUILD!   if (document.selection) {
SET BUILD=!BUILD!     var range = document.body.createTextRange();
SET BUILD=!BUILD!     range.moveToElementText(document.getElementById(containerid));
SET BUILD=!BUILD!     range.select().createTextRange();
SET BUILD=!BUILD!     document.execCommand("""copy""");
SET BUILD=!BUILD!   } else if (window.getSelection) {
SET BUILD=!BUILD!     var range = document.createRange();
SET BUILD=!BUILD!     range.selectNode(document.getElementById(containerid));
SET BUILD=!BUILD!     window.getSelection().addRange(range);
SET BUILD=!BUILD!     document.execCommand("""copy""");
SET BUILD=!BUILD!   }
SET BUILD=!BUILD! }
SET BUILD=!BUILD! !END-HTML-SCRIPT!
cls

echo Test-ONE-LINER.bat
echo.
pushd "%~d0%~p0"

ONE-LINER.bat

@Echo Off
setlocal enableDelayedExpansion

SET CUR-DIR=%~d1%~p1
SET PSScript=%CUR-DIR%\ONE-LINER.ps1
SET PowerShellDir=C:\Windows\System32\WindowsPowerShell\v1.0

SET Append-Filename=%~n1%~x1            
REM # html
SET Append-Text=%~n2%~x2
REM # text to add
echo ONE-LINER.bat
echo.
echo Filename: !Append-Filename!
echo Text:     !Append-Text!
pause
pushd "%CUR-DIR%"

CALL Powershell -Command "& '%PSScript%' '%Append-Filename%' '%Append-Text%'"
echo Bye from ONE-LINER.bat
:: # Program Exits Here
:: # instead of returning to Caller
pause

ONE-LINER.ps1

Set-Alias rh Read-Host
Set-Alias ds Write-Host

$Append_Filename=$args[0]
$Append_Text=$args[1]

ds "ONE-LINER.ps1"
ds
ds "This is the Append_Filename:   $Append_Filename"
ds "This is the Append_Text:       $Append_Text"

Write $Append_Text | out-file -append $Append_Filename -encoding ASCII

rh "Press any key...   Yada, yada, yada."
