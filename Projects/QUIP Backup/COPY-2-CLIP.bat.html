Success with a couple of QUIP PRO QUOS when using this program:

  • This program as-is currently only handles file names without spaces:  
    • “My Cool Code.html“" change to "My-Cool-Code.html" or something similar before processing.
    • Afterwards you can change the name back.
  • There are going to be errors that need to be fixed in a text editor
    • keep and verify with the original looks the same.
    • and that the buttons work.
  • Unicode and Special Character codes may be corrupted/corrupting to final output.

Trial and Error

  • This Program processes most QUIP html files, but does show 1 or more errors that can be ignored in the CMD window.
  • Some files are processed 100% correct
  • Some files are mostly correct but have.
    • Missing ID’s
    • Incorrect ID’s
      • Duplicated from Earlier
      • Appended with miscellaneous HTML
  • Some files are missing BUTTON code.
  • Some FEW files don’t work at all.

COPY-2-CLIP.bat

@echo off
SETLOCAL enabledelayedexpansion

Set CurFile=%~n1%~x1
Set OutFile=Temp.html
Set OutFile2=Temp2.html
Set OutFile3=Temp3.html
Set OutFile4=DONE\%~n1%~x1
Set DIV=

:: # Output Directory, Create if does not exist
if exist .\DONE (
  echo Subdirectory DONE Exists
) else (
  mkdir .\DONE
)

:: # Get JavaScript from external file
for /f "Tokens=* Delims=" %%x in (atom-copy-2-clipboard-function.txt) do (set Build=!Build!%%x)

:: # Replace Tabs resulting from QUIPs Unicode Encoding in HTML
SET CMD1="(gc !CurFile!) -replace '\xc2\xa0','&nbsp;'"
Powershell -Command %CMD1% > !OutFile!

:: # Replace ampersand code with AND
:: # Not sure if this really is a good idea since it screws with code.
SET CMD2="(gc !OutFile!) -replace '&amp;',' and '"
Powershell -Command %CMD2% > !OutFile2!

:: # Insert JavaScript 
:: # SS=Search String
SET SS=^</HEAD^>

:: # Three Quotes for each Quote to Pass args to Powershell replace function
SET Build=!Build:^"=^"^"^"!

:: # DS=Destination String
SET DS=!Build!^</HEAD^>

:: # gc is an alias in PowerShell for Get-Content
SET CMD3="(gc !OutFile2!) -replace '%SS%','%DS%'"
Powershell -Command %CMD3% > !OutFile3!

:: # Loop through HTML File
SET /a count=1
copy /y NUL !OutFile4! >NUL
for /f "tokens=*" %%s in (!OutFile3!) do (
        SET /a count+=1
        echo."%%s" | findstr /C:"<pre" 1>nul
        if errorlevel 1 (
            :: # Match not found
            echo %%s >> !OutFile4!
        ) ELSE (
            :: # Match found
            echo %%s >> !OutFile4!
            set INPUT-STRING=%%s
            
            :: # Isolate the DIV ID for the current Code Block
            :: # Used to be: Set DIV=!INPUT-STRING:~9,11!
            CALL :SetDIV DIV "%%s" 
            echo !DIV!
                        
            :: # Prepare your Button
            set HTML-Button=^<button id="button!count!" onClick="CopyToClipboard('!DIV!')"^>Click to copy^<^/button^>

            :: # Write the Button on the Next Line of HTML
            echo !HTML-Button! >> !OutFile4!
        )
)
goto :ENDER

REM # Function SetDIV - extract and return DIV string
REM # Merged from Isolate-ID.bat
:SetDIV
set LINE="%~2"
set STR=
set /a cnt = 0
set ID=
set Toggle=OFF
set /a ID_LEN=0

:: # Find the String Length
FOR /L %%N in (1,1,1024) DO (
    SET STR=!STR!!LINE:~%%N,1!
    if !LINE! == "!STR!" (
        set /a cnt=%%N
        goto :exit-for
    )
)
:exit-for

:: # Find the first and second single quote
:: # Toggle ID Capture ON and OFF
:: # Assign String to ID and Length to ID_LEN 
FOR /L %%C in (1,1,!cnt!) DO (
    REM # Find the Single Quote and SET TOGGLE ON/OFF
    if "!LINE:~%%C,1!" == "'" (
        if "!Toggle!" == "ON" (
            Set Toggle=OFF
        ) ELSE (
            Set Toggle=ON
        )
    )

    REM # Capture ID and ID_LEN
    if "!Toggle!" == "ON" (
        if "!ID_LEN!" == "0" (
            REM # Initial Condition Increment
            set /a ID_LEN=!ID_LEN! + 1
        ) ELSE (
            set /a ID_LEN=!ID_LEN! + 1
            SET ID=!ID!!LINE:~%%C,1!
        )
    )
)
:: # Initial Condition Adjustment
:: # NOTE: The single quote doesn't add to the String Length.
set /a ID_LEN=!ID_LEN! - 1

:: Return the ID String
set "%~1=!ID!"
Exit /B 0

:ENDER

:: Clean Up
del Temp.html
del Temp2.html
del Temp3.html

<SCRIPT>
function CopyToClipboard(containerid) {
  if (document.selection) {
    var range = document.body.createTextRange();
    range.moveToElementText(document.getElementById(containerid));
    range.select().createTextRange();
    document.execCommand("copy");
  } else if (window.getSelection) {
    var range = document.createRange();
    range.selectNode(document.getElementById(containerid));
    window.getSelection().removeAllRanges();
    window.getSelection().addRange(range);
    document.execCommand("copy");
  }
}
</SCRIPT>
# END of FILE
