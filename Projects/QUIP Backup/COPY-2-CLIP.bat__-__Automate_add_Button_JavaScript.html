Copy2Clipboard - Atoms of Code (The Most Basic Example)

  • The DIV ID
  • The Button
  • The JavaScript

1.  The DIV ID

<div id="div1">Text To Copy </div>

2.  The Button

<button id="button1" onclick="CopyToClipboard('div1')">
   Click to copy
</button>

3.  The JavaScript

atom-copy-2-clipboard-function.txt 
<SCRIPT>
function CopyToClipboard(containerid) {
  if (document.selection) {
    var range = document.body.createTextRange();
    range.moveToElementText(document.getElementById(containerid));
    range.select().createTextRange();
    document.execCommand("copy");
  } else if (window.getSelection) {
    var range = document.createRange();
    range.selectNode(document.getElementById(containerid));
    window.getSelection().removeAllRanges();
    window.getSelection().addRange(range);
    document.execCommand("copy");
  }
}
</SCRIPT>

COPY-2-CLIP.bat

(to Insert into QUIP Downloaded HTML Code Blocks) 

@echo off
SETLOCAL enabledelayedexpansion

:: Define Filenames
Set CurFile=%~n1%~x1
Set OutFile=Temp.html
Set OutFile2=Temp2.html
Set OutFile3=DONE\%~n1%~x1

:: Output Directory, Create if does not exist
if exist .\DONE (
  echo Subdirectory DONE Exists
) else (
  mkdir .\DONE
)

:: Get JavaScript from external file
for /f "Tokens=* Delims=" %%x in (atom-copy-2-clipboard-function.txt) do (set Build=!Build!%%x)

:: Replace Tabs resulting from QUIPs Unicode Encoding in HTML
SET CMD1="(gc !CurFile!) -replace '\xc2\xa0','&nbsp;'"
:: Write Results to Temp1.html
Powershell -Command %CMD1% > !OutFile!

:: Insert JavaScript 
:: SS=Search String
SET SS=^</HEAD^>

:: Three Quotes for each Quote to Pass args to Powershells replace function
SET Build=!Build:^"=^"^"^"!

:: DS=Destination String
SET DS=!Build!^</HEAD^>

:: Note: gc is an alias in PowerShell for Get-Content
:: Write Results to Temp2.html 
SET CMD2="(gc !OutFile!) -replace '%SS%','%DS%'"
Powershell -Command %CMD2% > !OutFile2!

:: Loop through HTML File
SET /a count=1

:: Write Results to Subdirectory .\DONE\%1
copy /y NUL !OutFile3! >NUL
for /f "tokens=*" %%s in (!OutFile2!) do (
        SET /a count+=1
        echo."%%s" | findstr /C:"<pre" 1>nul
        if errorlevel 1 (
            :: Match not found
            :: Write Line
            echo %%s >> !OutFile3!
        ) ELSE (
            :: Match found
            :: Write Line
            echo %%s >> !OutFile3!
            
            :: Copy the String in order to Isolate the DIV ID
            set INPUT-STRING=%%s
            
            :: Isolate the DIV ID for the current Code Block - QUIP ID ONLY
            Set DIV=!INPUT-STRING:~9,11!
            
            :: Prepare your Button
            set HTML-Button=^<button id="button!count!" onClick="CopyToClipboard('!DIV!')"^>Click to copy^<^/button^>

            :: Write the Button on the Next / New Line of HTML
            echo !HTML-Button! >> !OutFile3!
        )
)
:ENDER

:: Clean Up
del Temp.html
del Temp2.html
