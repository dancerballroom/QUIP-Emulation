SRT_SPLIT_20_2_LRC_GEQ_100.bat

Purpose:  

Chop up SRT Subtitles into 20 minute segments (with 10 second overlap) and Convert the SRT Subtitles (for Video) into LRC lyrics (for Audio)

For Books that get split in more than 100 Segments by  AAA-MP3-SPLIT-20.bat

Related Script

Chop up MP3 Audiobook into 20 minute segments for playing on Cell Phone
A Compliment of:  AAA-MP3-SPLIT-20.bat
Process Part of:  YOUTUBE-2-AUDIOBOOK.bat

Output Directories:    DONE-SRT\  and  DONE-LRC\

SRT_SPLIT_20_2_LRC_GEQ_100.bat

@Echo Off
setlocal enableDelayedExpansion

REM Storage Directories
If Exist *.%ext% (If Not Exist "DONE-SRT\" MD DONE-SRT)
If Exist *.%ext% (If Not Exist "DONE-LRC\" MD DONE-LRC)

REM Initialize Variables
SET START_HOUR=0
SET START_MIN=0
SET START_SEC=0

SET S_MIN=0
SET S_SEC=10

SET END_HOUR=0
SET END_MIN=20
SET END_SEC=10

SET COUNT=0

set start=0
set step=1
set end=200

REM NOW for some Batch MATH
for /L %%i in (!start!, !step!, !end!) do (
  set /a COUNT+=1

  REM Part No.
  if !COUNT! lss 10 (set CNT=00!COUNT!) ELSE (set CNT=0!COUNT!)
  if !COUNT! geq 100 (set CNT=!COUNT!)
  
  REM ADD Padding, EXAMPLE:   1:0:10 becomes 01:00:10 
  if !START_HOUR! lss 10 (set S_HR=0!START_HOUR!) ELSE (set S_HR=!START_HOUR!)
  if !END_HOUR! lss 10 (set E_HR=0!END_HOUR!) ELSE (set E_HR=!END_HOUR!)
  if !START_MIN! lss 10 (set S_MIN=0!START_MIN!) ELSE (set S_MIN=!START_MIN!)
  if !END_MIN! lss 10 (set E_MIN=0!END_MIN!) ELSE (set E_MIN=!END_MIN!)
  if !START_SEC! lss 10 (set S_SEC=0!START_SEC!) ELSE (set S_SEC=!START_SEC!)
  if !END_SEC! lss 10 (set E_SEC=0!END_SEC!) ELSE (set E_SEC=!END_SEC!)
  
  REM CREATE TIME SIGNATURE to Process 20 Minute Parts
  SET START_TIME_SIG=!S_HR!:!S_MIN!:!S_SEC!
  SET END_TIME_SIG=!E_HR!:!E_MIN!:!E_SEC!

  REM The ENGINE that does a lot of the magic (FFMPEG)
  ffmpeg -i "%~n1.srt" -ss !START_TIME_SIG! -to !END_TIME_SIG! -c copy "DONE-SRT\!CNT! - %~n1.srt"
  ffmpeg -i "DONE-SRT\!CNT! - %~n1.srt" "DONE-LRC\!CNT! - %~n1.lrc"

  REM This REMd SECTION Not Necessary 
  REM since FFMPEG errors out at NO_MORE_SUBS
  REM START SECTION
  REM set length=0
  REM for %%a in ("DONE-LRC\!CNT! - %~n1.lrc") do (
  REM    set file=%%~na
  REM    set length=%%~za
  REM )
  REM echo !file! length: !length!
  REM if !length! EQU 0 (EXIT)
  REM END SECTION
  
  REM Next PART
  CALL :SET_START_MINUTE !START_MIN! !START_HOUR!
  CALL :SET_END_MINUTE !END_MIN! !END_HOUR!
)
ECHO DEBUG SECTION 
ECHO Sleep 5 and EXIT
CALL :SLEEP 5
PAUSE
EXIT 0

REM FUNCTION SLEEP (seconds)
:SLEEP
  Powershell -Command "& 'C:\POWER-SHELL-SCRIPTS\SLEEP.ps1' '%1'"
EXIT /B

REM FUNCTION SET_START_HOUR (HOUR++) 
:SET_START_HOUR
  set /a START_HOUR+=1
EXIT /B 0

REM FUNCTION SET_START_MINUTE (MINUTE++)
:SET_START_MINUTE
  set /a START_MIN=%1+20
  if %1 EQU 40 ( 
    SET /a START_MIN=00
    CALL :SET_START_HOUR !START_HOUR!    
  )
EXIT /B

REM FUNCTION SET_END_HOUR (HOUR++)
:SET_END_HOUR
  set /a END_HOUR+=1
EXIT /B

REM FUNCTION SET_END_MINUTE (MINUTE++)
:SET_END_MINUTE
  set /a END_MIN=%1+20
  if %1 EQU 40 ( 
    SET /a END_MIN=00
    CALL :SET_END_HOUR !END_HOUR!    
  )
EXIT /B
