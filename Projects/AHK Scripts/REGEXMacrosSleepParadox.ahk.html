REGEX_Intermediate_1()
{
    ;; Used by TextPad++ CONVERT from TXT to CSV for Bookmarks 1st Half
    ;Send "(^\d{1,3})\.\s{0,3}\b(\d{4,8})\b\s{0,3}([a-zA-Z0-9\s\-\’\,]*)$"
    ;1st  "(^\d{1,3})\.\s{0,3}\b

    ;; VERIFIED WORKS
    ;; Send "({^}\d{{}1,3{}})\.\s{{}0`,3{}}\b"
    ; OK   (^\d{1,3})\.\s{0,3}\b

    ;; Used by TextPad++ CONVERT from TXT to CSV for Bookmarks 2nd Half
    ;                         (\d{4,8})\b\s{0,3}([a-zA-Z0-9\s\-\’\,]*)$"

    ;; VERIFIED WORKS
    ;; Send "(\d{{}4`,8{}})\b\s{{}0,3{}}([a-zA-Z0-9\s\-\’\,]*)$"
    ;; OK                        (\d{4,8})\b\s{0,3}([a-zA-Z0-9\s\-\’\,]*)$

    ; ALL TOGETHER NOW
    Send "({^}\d{{}1,3{}})\.\s{{}0`,3{}}\b(\d{{}4`,8{}})\b\s{{}0,3{}}([a-zA-Z0-9\s\-\’\,]*)${Tab}\1;\2;\3"
    ; RETURN followed by \1;\2;\3
}

REGEX_Intermediate_2()
{
    ; B4
    ; "(^\d{1,3});(\d{4,8});([a-zA-Z0-9\s\-\’\,]*)$"
    ; RETURN followed by \1;\2;\2c;\3
    Send "({^}\d{{}1,3{}})`;(\d{{}4,8{}})`;([a-zA-Z0-9\s\-\’\,]*)${Tab}\1;\2;\2c;\3"
}

REGEX_FinalFormat()
{
    Send "000c{Tab}999"
    ; RETURN followed by 999

}

REGEX_FirstLineFormat()
{
    Send "Number;Start time in milliseconds;End time in milliseconds;Text`r`n"
}

REGEX_NewTextFile()
{
    ;CNTRL + SHIFT + F10
    Send "+{F10}"
    ;W = NEW
    Send "w"
    ;T = TEXT FILE
    Send "t"
    SendInput "^v"
    Send "{Enter}"

    Sleep 1000
    ;CNTRL + SHIFT + F10
    Send "+{F10}"
    ;C = Copy
    Send "C"
    ;Paste
    SendInput "^v"
}
