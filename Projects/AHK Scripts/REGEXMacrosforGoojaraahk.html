REGEX_Season_Info_1()
{
    ;FIND: ^(\d\d)\r\nSeason (.*)\r\n(.*)
    ;REPLACE: SHOW_NAME - S0\2E\1 - \3

    Send "{^}(\d\d)\r\nSeason (.*)\r\n(.*){Tab}SHOW_NAME - S0\2E\1 ^\- \3"
}

REGEX_Yank_Title_2()
{
    ;FIND: \A^(.*)\r\n(.*)\r\n(.*)
    ;REPLACE: \3\r\n\r\n

    Send "\A{^}(.*)\r\n(.*)\r\n(.*){Tab}\3\r\n\r\n"
}

REGEX_First_Line_3()
{

}
