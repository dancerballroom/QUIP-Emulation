REGEX Macros.ahk

#Requires AutoHotkey v2.0
; Control + Shift + E
^+e::REGEX_Intermediate_1()
; Control + Shift + R
^+r::REGEX_Intermediate_2()
; Control + Shift + T
^+t::REGEX_FinalFormat()

; First Run - Establish Macro
MsgBox "Setting CTL + SHFT + E, `n`nCSV Intermediate Format for Bookmarks`n`n(Control+E. Intermediate Comma-Delimited)`n`n(Control+R, Duplicate TimeCode)`n`n(Control+T, Final Replacement)`n`n`nNote: None`n`n"

; # Exit Script - First Run
Exit

REGEX_Intermediate_1()
{
    ;; Used by TextPad++ CONVERT from TXT to CSV for Bookmarks 1st Half
    ;Send "(^\d{1,3})\.\s{0,3}\b(\d{4,8})\b\s{0,3}([a-zA-Z0-9\s\-\’\,]*)$"
    ;1st  "(^\d{1,3})\.\s{0,3}\b

    ;; VERIFIED WORKS
    ;; Send "({^}\d{{}1,3{}})\.\s{{}0`,3{}}\b"
    ; OK   (^\d{1,3})\.\s{0,3}\b

    ;; Used by TextPad++ CONVERT from TXT to CSV for Bookmarks 2nd Half
    ;                         (\d{4,8})\b\s{0,3}([a-zA-Z0-9\s\-\’\,]*)$"

    ;; VERIFIED WORKS
    ;; Send "(\d{{}4`,8{}})\b\s{{}0,3{}}([a-zA-Z0-9\s\-\’\,]*)$"
    ;; OK                        (\d{4,8})\b\s{0,3}([a-zA-Z0-9\s\-\’\,]*)$

    ; ALL TOGETHER NOW
    Send "({^}\d{{}1,3{}})\.\s{{}0`,3{}}\b(\d{{}4`,8{}})\b\s{{}0,3{}}([a-zA-Z0-9\s\-\’\,]*)$"
}

REGEX_Intermediate_2()
{
    ; B4
    ; "(^\d{1,3});(\d{4,8});([a-zA-Z0-9\s\-\’\,]*)$"
    Send "({^}\d{{}1,3{}})`;(\d{{}4,8{}})`;([a-zA-Z0-9\s\-\’\,]*)$"
}

REGEX_FinalFormat()
{
    Send "000c"
}
