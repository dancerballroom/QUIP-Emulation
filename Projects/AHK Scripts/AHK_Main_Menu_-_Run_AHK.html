#Requires AutoHotkey v2.0
ahkDoc := "https://www.autohotkey.com/docs/v2/lib/"   ; URL for AHK v2 function index
gui1   := Gui('+AlwaysOnTop', '[MENU]'), gui1.SetFont('s10') ; Set up GUI

; General Documentation AHK v2.0
gui1.AddLink(, '<a href="' ahkDoc 'GuiControls.htm#Button">Button with event</a> | '  ; Add URLs

; Define Buttons

gui1.OnEvent('Escape', escapeGUI)             ; Define event for GUI escape
showGUI                                       ; Show the GUI

F3::showGui                                   ; F3 = Show the GUI

showGUI() {
     ;ed.Focus      
     ; Focus on the edit control
    gui1.Show('x1000 y150')
}

activated01(btn01, info) {      ; btn01 Actions
    ; Send Close Command to any previously running Macros
    Send "^j"
    ; General Font Format
    btn_Norm()    
    ; Specific Font Format
    btn01.SetFont("cBlack s10  Italic Bold  Q4", "Ariel Black")
    
    ; Run YouTube Set of Macros
    Run 'YouTube - PotPlayer - Playlist - Download - Helper.ahk'
}
