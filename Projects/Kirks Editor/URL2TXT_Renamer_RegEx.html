Where all the RegEx Action Happens!

function convertFileNames() {
            const rawText = txtCurrent.value;
            if (!rawText.trim()) return;


            const lines = rawText.split(/\r?\n/);
            const processedLines = lines.map(line => {
                let s = line;


                // Strip out a leading number index wrapped in parentheses, like "(1) " or "( 20 )"
                s = s.replace(/^\(\s*\d+\s*\)\s*/g, '');


                // Remove the word "YouTube" completely, ignoring uppercase/lowercase differences
                s = s.replace(/YouTube/gi, '');


                // Remove the word "Manhwa" completely, ignoring uppercase/lowercase differences
                s = s.replace(/Manhwa/gi, '');


                // Remove the word "Recap" completely, ignoring uppercase/lowercase differences
                s = s.replace(/Recap/gi, '');


                // Convert any pound/hashtag sign (#) to the literal word "Number"
                s = s.replace(/#/g, 'Number');


                // Convert any ampersand symbol (&) to the literal word "And"
                s = s.replace(/&/g, 'And');


                // Replace any long chain of three or more consecutive periods (...) with a single hyphen
                s = s.replace(/\.{3,}/g, '-');


                // Strip out everything except normal letters, numbers, spaces, periods, hyphens, and parentheses
                s = s.replace(/[^\w\s\.\-\(\)]/g, '');


                // Capitalize the first letter of every single word and lowercase the rest (Title Case formatting)
                s = s.replace(/\b([a-zA-Z])([a-zA-Z0-9]*)\b/g, function(match, first, rest) {
                    return first.toUpperCase() + rest.toLowerCase();
                });


                // Strip out exclamation points entirely
                s = s.replace(/!/g, '');


                // Insert a clean space before the word "And" if it is stuck to the end of another word
                s = s.replace(/([^\s])(And)\b/g, '$1 $2');


                // Insert a clean space before the word "Number" if it is stuck to the end of another word
                s = s.replace(/([^\s])(Number)\b/g, '$1 $2');


                // Trim away any leftover trailing periods or hyphens from the end of the text string
                s = s.replace(/[\.\-]+$/g, '');
                s = s.trim();


                // Strip off a trailing hyphen if it has whitespace before it at the end of the line
                s = s.replace(/\s+\-+$/g, '');


                // Double check and clean up any trailing periods or hyphens left over after the previous space adjustments
                s = s.replace(/[\.\-]+$/g, '');
                s = s.trim();


                return s;
            });


            txtConverted.value = processedLines.join('\n');
        }
