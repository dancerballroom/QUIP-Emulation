03_Merge-MP4-DIR.bat

@Echo Off
setlocal enableDelayedExpansion

REM Create Temp DIR
md temp

REM Copy and Convert avoiding Time Stamps Merge Failure
for %%f in (*.mp4); do ffmpeg -r 25 -i "%%f" -c copy "temp\%%f"

REM Curdir is Temp
cd Temp

REM Create List of all MP4 in videos.txt
for %%f in (*.mp4); do echo file '%%f' >> videos.txt

cd ..
pushd temp

REM Merge List
ffmpeg -f concat -safe 0 -i videos.txt -c copy All-Joined.mp4
REM move All-Joined.mp4 ..
REM del /Q *.*

REM rmdir temp
set /p JoinedName="Name of All-Joined.mp4 (no ext) -> "
ren All-Joined.mp4 !JoinedName!.mp4

REM DONE
popd

04_Add_Bookmarks.bat

@Echo Off
setlocal enableDelayedExpansion

REM # Set Up File Variables 
SET META_FILE=MetaData.txt
SET MP4_FILE="%~n1%~x1"
SET MP4_NEW_FILE="%~n1_.mp4"
SET CSV_FILE=%~n1.csv

REM Work Around - For Loop Space Limitations
ren "!CSV_FILE!" Bookmarks.csv
Echo META_FILE = !META_FILE!
Echo MP4_FILE = !MP4_FILE!
Echo MP4_NEW_FILE = !MP4_NEW_FILE!
Echo CSV_FILE = !CSV_FILE!
Echo.
Echo Current Directory
Echo.
cd
Echo.

REM Extract MOVIE metadata previously existing 
ffmpeg -i !MP4_FILE! -f ffmetadata !META_FILE!
(Echo.)>>!META_FILE!

set /A i=0
echo.

REM Extract CSV DB info line by line
for /F "tokens=1,* delims=," %%a in (Bookmarks.csv) do (
    set /A i+=1
    set array[!i!]=%%a
)
echo Check

set /A n=!i!

REM Extract Line Info Delimited by Semi-Colons
for /L %%i in (2,1,!n!) do ( 
    for /f "tokens=1,2,3,4 delims=;" %%a in ("!array[%%i]!") do (
        set /A FIRST_SEMI=%%a
        set /A SECOND_SEMI=%%b
        set /A THIRD_SEMI=%%c
        set FOURTH_SEMI=%%d

        set /A SROUND=!SECOND_SEMI! + 0 2> nul
        set /A EROUND=!THIRD_SEMI! + 0 2> nul

        echo [CHAPTER]>>!META_FILE!
        echo TIMEBASE=1/1000>>!META_FILE!
        echo START=!SROUND!>>!META_FILE!
        echo END=!EROUND!>>!META_FILE!
        echo TITLE=!FOURTH_SEMI!>>!META_FILE!
        (echo.)>>!META_FILE!
    )
)

REM Rename the CSV DBFile back to its original name spaces and all
ren Bookmarks.csv "!CSV_FILE!"

REM Add New Chapter Files to Video
ffmpeg -i !MP4_FILE! -i MetaData.txt -map_metadata 1 !MP4_NEW_FILE!

05_Extract_Chapters.bat

@ECHO OFF
SETLOCAL ENABLEDELAYEDEXPANSION

IF "%~1" == "" (
     CALL :HELP
        EXIT
)
Set FILENAME="%~d1%~p1%~n1.mp4"
Set CHAPTERS="%~d1%~p1%~n1-Chapters.txt"

REM             TRUE = SHOW DEBUGGING 
set /A TRUE=1
REM             FALSE = HIDE DEBUGGING
set /A FALSE=0
REM SHOW/HIDE DEBUGGING
set /A DEBUG=!FALSE!

CALL :RUN_FFPROBE
exit

:RUN_FFPROBE

for /f "delims=" %%i in ('ffprobe -v quiet -print_format ini -show_chapters !FILENAME! ^| findstr /C:"title" /C:"start_time"') do (
    set my_variable=%%i
    REM echo !my_variable!
    REM echo.
    for /F "tokens=1,2 delims==^=" %%a in ("!my_variable!") do (
        REM echo Key: %%a
        REM echo Value: %%b
        if "%%a" EQU "title" (
            REM Echo %%b
            SET CHAPTER[!chapter_count!][2]= %%b
        )
        if "%%a" EQU "start_time" (
            set /a chapter_count+=1
            SET SPC=
            if !chapter_count! LEQ 9 (
                SET SPC=0
            )
            set /a Time_S=%%b 2> nul
            set /a Time_S=!Time_S! * 1000
            
            CALL :MilliSec2TimeSig !Time_S! Timex TimexC
            REM powershell -command "Write-Host -NoNewLine '!SPC!!chapter_count!. !Timex! '"
            REM Echo !SPC!!chapter_count!. !Timex!
            SET CHAPTER[!chapter_count!][1]=!SPC!!chapter_count!. !Timex!
        )
    )
)
type NUL > !CHAPTERS!
FOR /L %%c IN (1,1,!chapter_count!) DO (
    Echo !CHAPTER[%%c][1]! !CHAPTER[%%c][2]! >> !CHAPTERS!
    Echo !CHAPTER[%%c][1]! !CHAPTER[%%c][2]!
)
REM Echo !CHAPTERS!
type !CHAPTERS! | clip
pause
Exit /b

:HELP
    ECHO ==============================================================  
    ECHO ==== MP4 File not Specified
    ECHO ==== Extract_Chapters.bat "Filename.mp4"
    ECHO ====   Drag/Drop "Filename.mp4" on Extract_Chapters.bat
    ECHO ==============================================================  
pause
Exit /b

REM Function MilliSec2TimeSig
REM INPUT:  %1 (Milli-Seconds)
REM          CALL :MilliSec2TimeSig !Milli-Seconds! Timex TimexC
REM OUTPUT: Timex (Time-Signature: 12:34:56.789)
REM OUTPUT: TimexC (Time-Signature: 12:34:56,789)
:MilliSec2TimeSig
    Set /A MS_TOT=%1
    REM Echo MS_TOT: !MS_TOT!
    Set MS_EXT=!MS_TOT:~-3,3!
    if !DEBUG! EQU !TRUE! Echo Mili-Seconds Extracted: !MS_EXT!
    Set /A Rem_Minus_MS=!MS_TOT!-MS_EXT
    if !DEBUG! EQU !TRUE! Echo Total Minus MS: !Rem_Minus_MS!
    Set /A HH_EXT=(!Rem_Minus_MS!/(60*60*1000))
    if !DEBUG! EQU !TRUE! Echo Hours Extracted: !HH_EXT!
    Set /A Rem_Minus_HH=(!Rem_Minus_MS!%%(60*60*1000))
    if !DEBUG! EQU !TRUE! Echo Total Minus HH: !Rem_Minus_HH!
    Set /A MM_EXT=(!Rem_Minus_HH!/(60*1000))
    if !DEBUG! EQU !TRUE! Echo Minutes Extracted: !MM_EXT!
    Set /A Rem_Minus_MM=(!Rem_Minus_HH!%%(60*1000))
    if !DEBUG! EQU !TRUE! Echo Total Seconds SS: !Rem_Minus_MM!
    Set /A SS_EXT=(!Rem_Minus_MM!/1000)
    if !DEBUG! EQU !TRUE! Echo Seconds Extracted: !SS_EXT!

    if !HH_EXT! lss 10 set HH_EXT=0!HH_EXT!
    if !MM_EXT! lss 10 set MM_EXT=0!MM_EXT!
    if !SS_EXT! lss 10 set SS_EXT=0!SS_EXT!
    REM if !MS_EXT! lss 10 set MS_EXT=00!MS_EXT!
    REM MOD ERROR Results in .0 instead of .000
    REM ALSO Above 00!MS_EXT! Results in .000000 instead of .000

    if !DEBUG! EQU !TRUE! Echo Time Extracted: !HH_EXT!:!MM_EXT!:!SS_EXT!,!MS_EXT!
    Set Timex=!HH_EXT!:!MM_EXT!:!SS_EXT!
    Set TimexC=!HH_EXT!:!MM_EXT!:!SS_EXT!
Exit /b

Add_Bookmarks_CSV_Generator (Only Used by the 01_-_Entry Point...bat)

@echo off
setlocal EnableDelayedExpansion
IF "%~1" == "" CALL :Help Add_Bookmarks_CSV_Generator.bat
IF "%~2" == "" CALL :Help Add_Bookmarks_CSV_Generator.bat
:BEGIN

CALL :Create_Commands %1 HHMMSS MMM
  Echo MilliSeconds: !MMM!
  Echo Sexagessimal: !HHMMSS!
  Echo Test1: %~n1
  Set "addfile=%~n1"
CALL :Format_Save !MMM! "%addfile%" %2

Exit /b

:Create_Commands
  SET Video_HHMMSS=ffprobe -v error
  SET Video_HHMMSS=!Video_HHMMSS! -show_entries format=duration
  SET Video_HHMMSS=!Video_HHMMSS! -of default=noprint_wrappers=1:nokey=1
  SET Video_HHMMSS=!Video_HHMMSS! -i 
  SET Video_HHMMSS=!Video_HHMMSS! "%~n1.mp4"

  SET Video_MMM=ffprobe -v error
  SET Video_MMM=!Video_MMM! -show_entries format=duration
  SET Video_MMM=!Video_MMM! -of default=noprint_wrappers=1:nokey=1
  SET Video_MMM=!Video_MMM! -sexagesimal
  SET Video_MMM=!Video_MMM! -i 
  SET Video_MMM=!Video_MMM! "%~n1.mp4"

  echo.

FOR /F "tokens=* USEBACKQ" %%F IN (`!Video_HHMMSS!`) DO (
  SET MMM=%%F
  SET MMM=!MMM:~0,-3!
)

FOR /F "tokens=* USEBACKQ" %%F IN (`!Video_MMM!`) DO (
  SET HHMMSS=%%F
  SET HHMMSS=!HHMMSS:~0,-3!
)
Exit /b

:Format_Save 
set "datafile=running_total_ps.txt"
set "addfile=%~2"
Echo TEST2: %addfile%
REM pause
:: Initialize or Read Running Total
if not exist "%datafile%" (
    echo %1 > "%datafile%"
    echo Number;Start time in milliseconds;End time in milliseconds;Text>"Running_Total.txt"
    echo 0;000000;000999;[SHIFT COLUMN UP HERE]>>"Running_Total.txt"
    echo Current running total: %1
    Call :Add_Running_Total_To_File %1 "%addfile%" %3
    echo.
    REM pause
    Exit /b
)
set /P "current_total_ps=" < "%datafile%"

:: Prompt for input
set "new_value_ps=%1"

:: Perform calculation using PowerShell
for /F "usebackq" %%i in (`powershell.exe -Command "$result = [System.Decimal]::Parse('%current_total_ps%') + [System.Decimal]::Parse('%new_value_ps%'); $result.ToString()"`) do (
    set "new_total_ps=%%i"
)

:: Store the new total
echo %new_total_ps% > "%datafile%"

Call :Add_Running_Total_To_File %new_total_ps% "%addfile%" %3
echo Current running total: %new_total_ps%
echo.
REM pause
REM SET MMM=!MMM_INT!!MMM_FRA!
Exit /b

:Add_Running_Total_To_File
SET TOT=%~1
SET NinerZero=!TOT:~0,-4!000
SET Niner=!TOT:~0,-4!999
SET addfile=%~2
Echo TEST3: %addfile%
REM pause
echo %3;%NinerZero%;%Niner%;%addfile% >> "Running_Total.txt"

Exit /b

:Help
Echo %~1 Filename Count
Pause
Exit /b
