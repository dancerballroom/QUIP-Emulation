Extract_Chapter.bat

@Echo Off
setlocal enabledelayedexpansion

Set /a Chapter=0
for /f "delims=" %%a in (Temp.txt) do (
    Set /a Chapter+=1
    set output=%%a
    set prefix=Chapter !Chapter! 
    set output=!output:~28,-4!
    echo !prefix! - !output!
)

6.  Tie it all together with Drag-Drop

Extract_Chapter.bat

@Echo Off
setlocal enabledelayedexpansion

ffprobe -v quiet -print_format json -show_chapters %1 | findstr title > Temp.txt

Set /a Chapter=0
for /f "delims=" %%a in (Temp.txt) do (
    Set /a Chapter+=1
    set output=%%a
    set prefix=Chapter !Chapter! 
    set output=!output:~28,-4!
    echo !prefix! - !output!
)

del /q Temp.txt
pause
