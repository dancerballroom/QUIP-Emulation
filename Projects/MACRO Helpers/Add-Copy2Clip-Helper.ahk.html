Add-Copy2Clip-Helper.ahk

Purpose:  Quick Code Insertion into QUIP via AutoHotKey Shortcuts

Important Lessons learned along the way:

  • AHK: Double Quotes (“”) you can send using the back-tick(`) below the tilde (~) key
    • `"Hello World`"
  • AHK:  Curly-Brackets ({}) are a special case
    • You can send an Open curly brace using {{} 2-open-1-closed
    • a Closed curly brace using {}}  1-open-2-closed
  • NotePad++:  CASE matters
    • ^f  =  Control + F  = Find
    • ^F = Control + Alt + F  =  Find In Files
  • AHK:  The Variable “A_Clipboard“ without quotes stores the variable obtained by the Windows Clipboard

#Requires AutoHotkey v2.0
;CONTROL plus
;1:  Find slash head HTML
;2.  Paste in Copy2Clip function
;4:  Find pre
;5:  Find Next pre
;6:  Copy ID, Next Line, Paste in Copy to Clip <Button Code>

^1::FindHead()
^2::InsFunCopy2Clip()
^4::FindPre()
^5::FindNextPre()
^6::InsBTNCopy2Clip()


; First Run - Establish Macro
msg:="Cntrl+`n1 = Find Head`n2 = Insert fn Copy2Clip`n4 = Find pre`n5 = Find Next pre`n6 = Insert Button Copy2Clip"
MsgBox msg

Exit

FindHead()
{
    ;MsgBox "Find Head"
    ;Control f
        Send "^f"
    ;</HEAD>
        Send "</HEAD>"
    ;6 Tabs
        Loop 6
        { 
            Send "{tab}"
        }
    ;Enter
    Send "{enter}"
    Send "{esc}"    
}

InsFunCopy2Clip()
{
    ;MsgBox "Insert Function Copy2Clip"
    ;Send "<SCRIPT>`"{{}HELLO{}}`"</SCRIPT>"
    Send "<SCRIPT>function CopyToClipboard(containerid) {{}  if (document.selection) {{}    var range = document.body.createTextRange();    range.moveToElementText(document.getElementById(containerid));    range.select().createTextRange();    document.execCommand(`"copy`");  {}} else if (window.getSelection) {{}    var range = document.createRange();    range.selectNode(document.getElementById(containerid));    window.getSelection().removeAllRanges();    window.getSelection().addRange(range);    document.execCommand(`"copy`");  {}}{}}</SCRIPT>"
}

FindPre()
{
    ;MsgBox "Find PRE"
    ;Control f
        Send "^f"
    ;</HEAD>
        Send "<PRE"
    ;6 Tabs
        Loop 6
        { 
            Send "{tab}"
        }
    ;Enter
    Send "{enter}"
    Send "{esc}"
    ; Right arrow 4 times
    Loop 4
    { 
        Send "{right}"
    }
    
}

FindNextPre()
{
    ;MsgBox "Find Next Pre"
    Send "{F3}"
}

InsBTNCopy2Clip()
{
    ;MsgBox "Ins BTN Copy2Clip"
    ; Control+Shift+Right Arrow Selects Text Delimited by special chars.
    Send "{down}"
    Send "<button id=`"button1`" onclick=`"CopyToClipboard('" . A_Clipboard . "')`">Click to copy</button>"
}
