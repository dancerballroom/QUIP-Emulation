Prepopulate “Find What” Box in NotePad++

  • Edit the config.xml File for Permanent Keywords
  • Completely close Notepad++.
  • Navigate to your settings directory (usually C:\Users\%username%\AppData\Roaming\Notepad++).
  • Open config.xml in a basic text editor like standard Windows Notepad.
  • Find the <FindHistory> section and manually add your custom strings using <Find name="YourSearchTerm"/>.
  • Save the file and restart Notepad++.
<?xml version="1.0" encoding="UTF-8" ?>
<NotepadPlus>
    <FindHistory nbMaxFindHistoryPath="10" nbMaxFindHistoryFilter="10" 
nbMaxFindHistoryFind="10" nbMaxFindHistoryReplace="10" matchWord="no" matchCase="no" 
wrap="yes" directionDown="yes" fifRecuisive="yes" fifInHiddenFolder="no" 
fifProjectPanel1="no" fifProjectPanel2="no" fifProjectPanel3="no" 
fifFilterFollowsDoc="no" fifFolderFollowsDoc="no" searchMode="2" transparencyMode="1" 
transparency="150" dotMatchesNewline="no" isSearch2ButtonsMode="no" 
regexBackward4PowerUser="no" bookmarkLine="no" purge="no">
        <Filter name="" />
        <Find name="^(\sminute)" />
        <Find name="^(\sminutes)" />
        <Find name="^(\ssecond)" />
        <Find name="^(\sseconds)" />
        <Find name="^(\sminute, \d*)" />
        <Find name="^(\sminutes, \d*)" />
        <Find name="^(\shour, \d*)" />
        <Find name="^(\shours, \d*)" />
    </FindHistory>
