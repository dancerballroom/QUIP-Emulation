Self-Calling HTML JavaScript URL Key-Value Parser

Purpose:  A Self-Calling HTML file that parses variables sent by the GET Method

Goal:  Clicking on Season X of Show Y, the page will load the selected IFRAME:  

Note:  The images of Kermit, Ms. Piggy, Scooter,  Gonzo, etc. are based on Key-Value Pairs called from the same web page:  “Page=Season-X“ where X=1 through 5, or Nothing, which defaults to Season-1.

[ Part of the Online Home Video HTML Project.]  

Output:

CASE 0 (No Value-Pairs)

CASE 1 (Page=Season-1)

CASE 2 (Page=Season-2)

CASE 3 (Page=Season-3)

CASE 4 (Page=Season-4)

CASE 5 (Page=Season-5)

HTML-JavaScript for the Above Example OUTPUT

<HTML>

<SCRIPT>
function Load_Page() {
    // Get the URL
    let url = window.location.search;
    // Initialize
    let base = url.split("?").shift();
    // SPLIT on ? 
    let querystring = url.split("?").pop();
    // SPLIT on &
    let params = querystring.split("&");
    // Array of pairs
    let pairs = {};

    // Tip-Toe through the Value Pairs
    for (let i = 0; i < params.length ; i++) {
      let key = params[i].split("=").shift();
      let value = params[i].split("=").pop();
              // The Default Case
              if (key == "" && value == "") {
                  key = "Page";
                  value = "Season-1";
              }
          // Print out the key-value pairs
          document.write("<CENTER><H2>");
          document.write(key);
          document.write(" = ");
          document.write(value);
          document.write("</H2></CENTER>");
          pairs[key] = value;
    }
    
    // Assign the values to output and retrieve their values
    output = [{url, base, querystring, params, pairs}];
    document.write("<CENTER><H4>");
    document.write("url:          " + output[0].url + "<BR>");
    document.write("params:       " + output[0].params + "<BR>");
    document.write("querystring:  " + output[0].querystring + "<BR>");
    document.write("pairs key:    " + output[0].pairs["Page"] + "<BR>");
    document.write("Image3:    " + output[0].pairs["Page"] + ".jpg<BR>");    
    document.write("</H4></CENTER>");
}

// Ok, here is the magic
// We assign the image based on the value pair "Page=Season-X"
function Bob2IMG(){
    document.getElementById('Bob').src=output[0].pairs["Page"] + ".jpg";
}

</SCRIPT>
</HEAD>

<BODY>
