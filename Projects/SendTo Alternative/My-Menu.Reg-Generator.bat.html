REM Parse
SET DDRIVE=%~d1
SET PPATH=%~p1
SET FFOLDER=%~n1

Echo Drive : %DDRIVE%
Echo Path : %PPATH%
Echo.
Echo The First Line in the REG FILE is based on:
Echo Folder: %FFOLDER%
Echo.
Echo ==========[ My-Menu.REG ]===========
Echo Windows Registry Editor Version 5.00
Echo.
Echo [HKEY_CLASSES_ROOT\*\shell\%FFOLDER%]
Echo "Icon"="shell32.dll,43"
Echo "Subcommands"=""
Echo.
Echo ====================================
Echo.
pause

Directory Lines

Pseudo-CODE

    $Path = HKEY_CLASSES_ROOT\*\shell\%FFOLDER%

    # This only handles a second-level directory structure
    For Each DIRECTORY-NAME
          $Current_Name = PARSE_NAME($Current_Line)
          
          # Each Directory is defined with 3-lines
          $Current_Path = "[" + $Path + "\shell\menu" + i + "]"
          $MUIVerb = "MUIVerb"=$Current_Name
          $SubCommands = "SubCommands"=""
          
          # Example of Directory Entry
          #
          # [HKEY_CLASSES_ROOT\*\shell\My-Menu\shell\menu1]
          # "MUIVerb"="Conversion"
          # "SubCommands"=""
    Next For

MY-MENU-TREE\shell\TRIM\shell\20-40
MY-MENU-TREE\shell\TRIM\shell\50-XX

3.  Extract Path from CURRENT-DIR-NAME

MY-MENU-TREE\shell\TRIM\shell
MY-MENU-TREE\shell\TRIM\shell

5. Replace SUFFIX Directory-Name with Menu-NUMBER

[HKEY_CLASSES_ROOT\*\shell\My-Menu-Tree\shell\menu3]
