@Echo Off
setlocal enableDelayedExpansion
SET /A TEST_CNT=0
CALL :VALIDATE_DIRECTORIES !TEST_CNT!
If !TEST_CNT! EQU 6 (
    Echo ALL TESTS Passed
    Echo SEC-START: !OPA!
    Echo MIN-SEC-END: !PA_HIGH!:!PA_LOW!
    REM *********** RUN CODE HERE **********
)
Pause
Exit /B

:VALIDATE_DIRECTORIES
FOR /D %%I IN ("%CD%") DO SET PA=%%~nxI
SET /A PA_HIGH=!PA:~0,2!
SET /A PA_LOW=!PA:~2,2!

IF !PA_LOW! GTR 0 (
    Echo Test 1 of 6: !PA_LOW! is VALIDATED - GTR than 0
    SET /A TEST_CNT+=1
) 

IF !PA_HIGH! GTR 0 (
    Echo Test 2 of 6: !PA_HIGH! is VALIDATED - GTR than 0
    SET /A TEST_CNT+=1
) 

IF !PA_LOW! LSS 60 (
    Echo Test 3 of 6: !PA_LOW! is VALIDATED - LSS than 60
    SET /A TEST_CNT+=1
) 

IF !PA_HIGH! LSS 60 (
    Echo Test 4 of 6: !PA_HIGH! is VALIDATED - LSS than 60
    SET /A TEST_CNT+=1
) 

pushd ..

FOR /D %%I IN ("%CD%") DO SET OPA=%%~nxI

IF !OPA! GTR 0 (
    Echo Test 5 of 6: !OPA! is VALIDATED - GTR than 0
    SET /A TEST_CNT+=1
) 

IF !OPA! LSS 60 (
    Echo Test 6 of 6: !OPA! is VALIDATED - LSS than 60
    SET /A TEST_CNT+=1
) 

IF !TEST_CNT! LSS 6 (
    Echo INVALID   :  Only !TEST_CNT! out of 6 Test Passed
    SET TEST=!FALSE!         
)
popd

Exit /B
