JPG-2-TXT.bat

@Echo Off
setlocal enableDelayedExpansion
SET JPG=%~n1%~x1
SET OUTPUT=!JPG:~0,-4!.txt
SET SNIPPET=!JPG:~0,-4!-Snippet.html
SET BUILDER=temp.txt
SET NAME=!JPG:~0,-4!

PUSHD %~d0%~p0
REM # Get Dimensions of JPG
Set Irfanview=^"C:\Program Files (x86)\IrfanView\i_view32.exe^"
Set Temp1=Temp1.txt
Set Temp2=Temp2.txt
!Irfanview! !JPG! /info=!Temp1!
type !Temp1! | find /N "dimensions" > !Temp2!
for /F "tokens=4" %%a in (!Temp2!) do set /a w=%%a
for /F "tokens=6" %%a in (!Temp2!) do set /a h=%%a
REM # echo Width = !w!
REM # echo Height = !h!
del !Temp1!
del !Temp2!

REM # Base64 convert JPG to TXT
SET CMD=[Convert]::ToBase64String((Get-Content -Path 
SET CMD=!CMD!.\!JPG! -Encoding Byte))
SET CMD=!CMD! ^> !OUTPUT!

echo !CMD!
Powershell -Command !CMD!
Powershell -Command Start-Sleep 5

type JPG-2-TXT-TOP.txt > !SNIPPET!
echo               case "!NAME!" : >> !SNIPPET!

REM # temp.txt will contain the following 3 lines
echo                  data +=^">!BUILDER!
more !OUTPUT!>>!BUILDER!
echo \"">>!BUILDER!

REM # temp-join-newfile.ps1 concatenates temp.txt in output file newfile.txt
PowerShell -Command ".\temp-join-newfile.ps1"

REM # Add Correctly Formatted BASE64 IMG into HTML file
type newfile.txt >> !SNIPPET!

type JPG-2-TXT-MID.txt >> !SNIPPET!

REM # Original Values
REM # SET HalfScr=50

REM # The next line works if % is pt in TOP.txt
REM # echo        ^<SCRIPT^>RetrieveIMG('!NAME!', !w!, !h!);^</SCRIPT^> >> !SNIPPET!
echo ^<H2^>!HalfScr! Percent Screen Width and Height^</H2^> >> !SNIPPET!
echo        ^<SCRIPT^>RetrieveIMG('!NAME!', !HalfScr!, !HalfScr!);^</SCRIPT^> >> !SNIPPET!
echo ^<HR^> >> !SNIPPET!
echo ^<H2^>!QuartScr! Percent Screen Width and Height^</H2^> >> !SNIPPET!
echo        ^<SCRIPT^>RetrieveIMG('!NAME!', !QuartScr!, !QuartScr!);^</SCRIPT^> >> !SNIPPET!
echo ^<HR^> >> !SNIPPET!

echo        ^<SCRIPT^>RetrieveIMG('!NAME!', !TenthScr!, !TenthScr!);^</SCRIPT^> >> !SNIPPET!
type JPG-2-TXT-BTM.txt >> !SNIPPET!

echo Created:  !SNIPPET!

temp-join-newfile.ps1 - Concatenating CODE + BASE64-data + CODE

$fileName = "temp.txt"
# Get content of file
$content = Get-Content -Path $fileName -Raw -Encoding UTF8

# remove carriage return and newline characters
$newcontent = $content.Replace("`r","").Replace("`n","")

# display the changed text
Write-Host "Content: $newcontent"

# write the modified text to a new file
$newfilename = "newfile.txt"
Out-File -FilePath "$newfilename" -force -InputObject $newcontent
