Mid-Show-Seasons-Gen

000-mid-show-seasons-gen.bat

@Echo Off
SETLOCAL ENABLEDELAYEDEXPANSION
set main=main-middle-series-seasons-6td.txt

set /a count=0
echo ^<tr^> > !main! 
for /d %%D in (*) do (
    set curdir=%%D

(for /F "delims=" %%a in ('findstr /I /L "<plot> <data>" "tvshow.nfo"') do (
   set "line=%%a"
   set "line=!line:*<plot>=!"
   set "line=!line:*<data>=!"
   for /F "delims=<" %%b in ("!line!") do echo %%b
)) > result.txt

    set /a count+=1
    Echo ^<td^> >> !main!
    Echo ^<div class=^"photo-container^"^> >> !main! 
    Echo ^<A HREF=^"%%D\Index.html^"  target=^"_top^"^>^<IMG SRC=^"%%D\Folder.jpg^" class=^"card^"^> >> !main!
    Echo ^<span^>%%D  !curseriesdescription!^<BR^> >> !main!
    
(for /F "delims=" %%a in ('findstr /I /L "<plot> <data>" "tvshow.nfo"') do (
   set "line=%%a"
   set "line=!line:*<plot>=!"
   set "line=!line:*<data>=!"
   for /F "delims=<" %%b in ("!line!") do echo %%b
)) >> !main!
    
    Echo ^</span^> >> !main!
    
    Echo ^<CENTER^>^<H3 class=^"name^"^>%%D^</H3^>^</A^>^</CENTER^>^</TD^>^</DIV^> >> !main!
    REM Echo. >> !main!

    IF "!count!"=="6" (
        Echo ^</tr^> >> !main!
        Echo ^<tr^> >> !main!
        set /a count=0
    )
)
:ender
Echo ^</tr^> >> !main!

type main-top-series-seasons-6td.txt > main.html
type main-middle-series-seasons-6td.txt >> main.html
type main-bottom-series-seasons-6td.txt >> main.html
