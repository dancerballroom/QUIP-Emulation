000_CheckURLs.bat

REG-EX from Quote to End-Of-Line

"[^\r\n]*

000_CheckURLs.bat

@echo off
setlocal ENABLEDELAYEDEXPANSION

REM Count the number of lines in the list
for /f %%a in ('findstr /R /N ^"^^" 000-list.txt | find /C ^"^:^"') do (Set /a lines=%%a)

echo There are !lines! movies left
Set /a Count=!lines!
for /L %%b in (1,1,!lines!) Do ( 

Set /a Count=!Count!-1
Echo There are !Count! movies left
rem fetch the first URL
set /p URL=<000-list.txt

rem Open the browser with that URL
start "" "!URL!"

rem Remove the URL from the front of the file ...
more +1 000-list.txt | findstr /r /v "^$" > 000-tmp_list.txt

rem ... and put it to the end
echo.!URL!>>000-tmp_list.txt
del 000-list.txt

ren 000-tmp_list.txt 000-list.txt

)
pause
endlocal
