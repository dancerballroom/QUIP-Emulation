JPG-2-TXT.bat

Created to Embed Images in HTML to use on drv.tw (Google DRIVE) when IMG tags just aren’t making the GRADE.

JPG-2-TXT.bat

@Echo Off
setlocal enableDelayedExpansion
SET JPG=%~n1%~x1
SET OUTPUT=!JPG:~0,-4!.txt
SET SNIPPET=!JPG:~0,-4!-Snippet.txt

REM Get Dimensions of JPG
Set Irfanview=^"C:\Program Files (x86)\IrfanView\i_view32.exe^"
Set Temp1=Temp1.txt
Set Temp2=Temp2.txt
!Irfanview! !JPG! /info=!Temp1!
type !Temp1! | find /N "dimensions" > !Temp2!
for /F "tokens=4" %%a in (!Temp2!) do set w=%%a
for /F "tokens=6" %%a in (!Temp2!) do set h=%%a
del !Temp1!
del !Temp2!
REM echo Width = !w!
REM echo Height = !h!

REM Base64 convert JPG to TXT
SET CMD=[Convert]::ToBase64String((Get-Content -Path 
SET CMD=!CMD!.\!JPG! -Encoding Byte))
SET CMD=!CMD! ^>^> !OUTPUT!

echo !CMD!
cd
Powershell -Command !CMD!
REM dir /b !JPG!*.*
pause
