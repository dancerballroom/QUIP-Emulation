Extract Line from Text File

@echo off
setlocal enabledelayedexpansion

set "file=your_text_file.txt"
set "line_number=3" 

for /f "tokens=1* delims=:" %%a in ('findstr /n "^" "%file%"') do (
    if %%a equ %line_number% (
        set "extracted_line=%%b"
        goto :break
   )
)

:break
echo %extracted_line%
