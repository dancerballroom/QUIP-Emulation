Procedure

  • Create DIR
  • CONTROL + C
  • CONTROL + V for as many DIRS as you Want
  • Drop the Original dir onto this Batchfile

DDD_Copy_DIR_Man.bat

@Echo Off

REM 1. Create dir
REM 2. CONTROL + C
REM 3. CONTROL + V for as many DIRS as you Want
REM 4. Drop the Original dir onto this Batchfile

echo B4:
echo.
dir /b /ad
echo.
echo Next: Run RMDIR and REN cmd
echo.
echo rmdir "%~n1 - Copy"
echo.
pause
rmdir "%~n1 - Copy"
ren "%~n1" "%~n1 - Copy (1)"
dir /b /ad
echo.
echo Next: Rename Dirs - Zap Copy and Parens
echo.
pause
Powershell -Command "Get-ChildItem -Directory | Rename-Item -ErrorAction SilentlyContinue -NewName {$_.Name -replace '\(|\)|Copy |\-\s', ''}"
dir /b /ad
echo.
echo Next: Rename Dirs - Add Leading Zeroes
echo.
pause
Echo Final Output
echo.
Powershell -Command "Get-ChildItem -Directory | Rename-Item -ErrorAction SilentlyContinue -NewName {$_.Name -replace '\s(\d)$', ' 0$1'}"
dir /b /ad
echo.
pause
