@echo off&mode 71,30&title PokeBatch2&setlocal enabledelayedexpansion&Pushd "%~dp0"
:Presets
if %number_of_processors% GTR 8 (set NOP=4) else (set/a NOP=%number_of_processors%/2)
set "ln=echo ÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄÄ"

set "t1n=&timeout 1 /NOBREAK >nul"
if not exist invisible.vbs (
    call :Createinvisiblevbs
) else (
    for /f "tokens=3" %%a in ('dir /A-D ^| find "invisible.vbs"') do (
        if not %%a==80 (
            call :Createinvisiblevbs
        )
    )
)
::this needs to be here:,10,10,10 because skip=0 gives errors
for /f "skip=2 tokens=1 delims=[]:" %%a in ('find /N /I ")<%%oldname%%.txt" "%~nx0"') do set varcount=%%a
for /f "skip=3 tokens=1 delims=[]:" %%a in ('find /N /I ":Load-Menu" "%~nx0"') do set/a varcount=!varcount!-(%%a+11)&goto Instructions
