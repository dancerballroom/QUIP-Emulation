Purpose:  Create Batchfile Create-Dirlink.ini for use by Create-Dirlink.bat
REF:  Create-Dirlink.bat

Create-Dirlink-INI-GEN.bat

@echo off
:Try_Again_myCMD
    set /p myCMD="Command (mklink /j): "
    if not defined myCMD goto :Try_Again_myCMD
    echo %myCMD%
:Try_Again_vd
    set /p vd="Virtual Directory (C:\DIR): "
    if not defined vd goto :Try_Again_vd
    echo %vd%
:Try_Again_td
    set /p td="Target Directory (F:): "
    if not defined td goto :Try_Again_td
    echo %td%
:Try_Again_tdd
    set /p tdd="Target Description (First HDD): "
    if not defined tdd goto :Try_Again_tdd
    echo %tdd%
:Try_Again_sbfn
    set /p sbfn="Batch File name to Create (C:\DIR\Mklink.bat): "
    if not defined sbfn goto :Try_Again_sbfn
    echo %sbfn%
    set INI_FILE=Create-Dirlink.ini
:Try_Again_Write_To_File
    set /p WTF="Write File Confirmation (Y): "
    if not defined WTF goto :Try_Again_Write_To_File
    
Echo Writing File %INI_FILE%...

Echo myCMD="%myCMD% ">> %INI_FILE%
Echo virtual_directory="%vd%\">> %INI_FILE%
Echo target_directory="%td%\">> %INI_FILE%
Echo target_description=REM %tdd%>> %INI_FILE%
Echo create_batchfile_name="%sbfn%">> %INI_FILE%

Echo INI_FILE
Echo ===========
type %INI_FILE%
Echo ===========

pause
