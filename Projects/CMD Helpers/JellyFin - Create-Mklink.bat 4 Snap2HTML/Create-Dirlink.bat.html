Create-Dirlink.bat

Purpose:  To Facilitate the Creation of Virtual Directories
Special Discovery:  The use of an INI (initialization) file

The INI File:  Create-Dirlink.ini

myCMD="mklink /j "
virtual_directory="C:\JELLYFIN_HOME\HDD-1\"
target_directory="F:\"
target_description=REM First HDD
create_batchfile_name="C:\JELLYFIN_HOME\Mklink.bat"

The Main Program Batchfile:  Create-Dirlink.bat

@Echo Off
SETLOCAL enabledelayedexpansion

for /f "tokens=1,2 delims==" %%a in (Create-Dirlink.ini) do (
    if %%a==myCMD set myCMD=%%~b
    if %%a==virtual_directory set vir_dir=%%~b
    if %%a==target_directory set tar_dir=%%~b
    if %%a==target_description set tar_desc=%%~b
    if %%a==create_batchfile_name set create_bat_name=%%~b
)

Echo @Echo Off > %create_bat_name%
Echo %tar_desc% >> %create_bat_name%

for /f "tokens=1,2 delims==" %%c in ('dir /ad /b "%tar_dir%"') do (
    Set /p Linker="+ %%c (y n x): "
    IF "!Linker!"=="y" (
        echo %myCMD% "%vir_dir%%%c" "%tar_dir%%%c" >> %create_bat_name%
    ) ELSE IF "!Linker!"=="n" (
        echo SKIPPED: %%c
    ) ELSE IF "!Linker!"=="x" (
        exit
    )
)

pause
