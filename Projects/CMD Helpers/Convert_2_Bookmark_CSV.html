Convert_2_Bookmark_CSV.bat

REM Convert_2_Bookmark_CSV.bat
@echo off
setlocal enabledelayedexpansion


:: Output header line
> output.txt echo Number;Start time in milliseconds;End time in milliseconds;Text


set "counter=0"


for /f "usebackq tokens=1* delims= " %%A in ("input.txt") do (
    set /a counter+=1
    
    :: Extract time token (%%A) and description (%%B)
    set "timeStr=%%A"
    set "desc=%%B"
    
    :: Parse timestamps formatted as M:SS or H:MM:SS
    call :ParseTime "!timeStr!" totalSeconds
    
    :: Calculate start and end milliseconds
    set /a startMs=totalSeconds * 1000
    set /a endMs=startMs + 999
    
    :: Append formatted line to output.txt
    >> output.txt echo !counter!;!startMs!;!endMs!;!desc!
)


echo Conversion complete! Output saved to output.txt.
pause
exit /b


:: --- Helper Function to Parse Time Strings ---
:ParseTime
set "t=%~1"
set "h=0"
set "m=0"
set "s=0"


:: Check how many colons exist in the string
for /f "tokens=1,2,3 delims=:" %%X in ("!t!") do (
    if "%%Z" neq "" (
        :: Formatted as H:MM:SS
        set "h=%%X"
        set "m=%%Y"
        set "s=%%Z"
    ) else if "%%Y" neq "" (
        :: Formatted as M:SS
        set "m=%%X"
        set "s=%%Y"
    ) else (
        :: Formatted as SS
        set "s=%%X"
    )
)


:: Strip leading zeroes to prevent batch octal interpretation
if "!h:~0,1!"=="0" if "!h!" neq "0" set "h=!h:~1!"
if "!m:~0,1!"=="0" if "!m!" neq "0" set "m=!m:~1!"
if "!s:~0,1!"=="0" if "!s!" neq "0" set "s=!s:~1!"


:: Convert everything to total seconds: (H * 3600) + (M * 60) + S
set /a totalSec=(h * 3600) + (m * 60) + s
set "%~2=!totalSec!"
        exit /b
