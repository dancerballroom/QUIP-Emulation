Kill OBS_ANSI_V5.bat

Features:

  • UP ARROW = ADD 5 Minutes
  • DOWN ARROW = SUBTRACT 5 Minutes
  • X = Exit Program
  • Large Font
  • Count Down
Testing edit
@Echo Off
setlocal enableDelayedExpansion
CALL :setESC
REM *******************************
REM *    OPEN THIS BATCH FILE    **
REM *    B4 RECORDING            **
REM *******************************

REM 1. ASK USER for Seconds to Run Recorder
Echo Enter Number of Seconds till OBS termination
Echo.
Echo  60= 1 min          3600 = 1 hr   5400 = 1.5 hrs 
Echo  6300= 1 hr 45 min  7200= 2 hrs   9000 = 2.5 hrs  
Echo  9900 = 2 hrs 45 min
Echo.
set /p secWait="Seconds til OBS termination (00000) : "
Echo.
Echo Keep Window Open.  Process obs64.exe will terminate after !secDuration! Seconds
Echo Slow Tripple Press   X=Exit   UP=ADD 5 MINS   DOWN= SUBTRACT 5 MINS
Echo.
set "ABS_TO=%ESC%[32;1H"

:WAITING
REM ECHO Extract Length of secWait
set /a length=0

:loop
    set "testchar=!secWait:~%length%,1!"
    set "PREV_length=!lengthSTR!"
    if defined testchar (
        set /a length+=1
        goto loop
    )
    REM echo Length is !length!
    set /a "lengthSTR=!length!-1"   
    set /A "secWait=!secWait!-1"
    for /L %%i in (0,1,!lengthSTR!) do (
         set "char=!secWait:~%%i,1!"
         REM echo Character at index %%i is !char!
    )
    
:: Check for key press via PowerShell (non-blocking)
for /f "tokens=*" %%K in ('powershell -Command "if ([Console]::KeyAvailable) { $key = [Console]::ReadKey($true); echo $key.Key } else { echo None }"') do (
    set "last_key=%%K"
)

:: Handle arrow keys if detected
REM if "%last_key%"=="UpArrow" echo YOU PRESSED UP!
REM if "%last_key%"=="DownArrow" echo YOU PRESSED DOWN!

:: Logic based on virtual key codes: 
    REM echo "!last_key!"
    if "!last_key!"=="UpArrow" (
        REM Add 1 Minute
        set /A "secWait=secWait+300"
        set /A "Press=1"
    ) else if "!last_key!"=="DownArrow" (
        REM Subtract 1 Minute
        set /A "secWait=secWait-300"
        set /A "Press=1"
    ) else if "!last_key!"=="X" (
        Echo End Of Program
        timeout /t 5 /nobreak 
        exit /b
    ) else (
        set /A "Press=0"
    )    
    REM Clear NUMBER Up To 6 Places: 123456 
    REM    (MAX: 999999 or 11:13:46:39 in DD:HH:MM:SS) 
    if "!Press!"=="1" (
            CALL :setPOS 10
            CALL :BLANKO        
            CALL :setPOS 20
            CALL :BLANKO        
            CALL :setPOS 30
            CALL :BLANKO        
            CALL :setPOS 40
            CALL :BLANKO        
            CALL :setPOS 50
            CALL :BLANKO        
            CALL :setPOS 60
            CALL :BLANKO    
            set /A "Press=0"
    )
    REM HERE IS THE DISPLAY ACTION
    for /L %%i in (0,1,!lengthSTR!) do (
        set "char=!secWait:~%%i,1!"
        set /A "num=(%%i + 1) * 10"
        set /A "num2=!num!+10"
        
        CALL :setPOS !num!
        CALL :!char! 2>nul
        
        if !num! EQU 9 do (
            CALL :setPOS !num2!
            CALL :BLANKO            
        )
        if !num! EQU 99 do (
            CALL :setPOS !num2!
            CALL :BLANKO            
        )
        if !num! EQU 999 do (
            CALL :setPOS !num2!
            CALL :BLANKO            
        )
        if !num! EQU 9999 do (
            CALL :setPOS !num2!
            CALL :BLANKO            
        )
    )    
    REM Echo !ABS_TO!!secWait!
    timeout /t 1 /nobreak > nul
if !secWait! == 1 (Echo OBS Shutting Down) ELSE (GOTO :WAITING)

REM 3. TURN OFF Recording
REM NOTE: ADDED CNTRL+ALT+PGUP/PGDN (Start/Stop Record)
Echo Press CNTRL + SHIFT + PGDN
powershell -c "$wshell = New-Object -ComObject wscript.shell; $wshell.SendKeys('^+{PGDN}')"

REM 4. WAIT 5 SECONDS to let Turn Off Recording to Process
timeout /t 5 /nobreak 

REM 5. SHUT DOWN OBS Program
taskkill /f /im obs64.exe

REM END OF PROGRAM

REM pushd "C:\Program Files\obs-studio\bin\64bit\"
