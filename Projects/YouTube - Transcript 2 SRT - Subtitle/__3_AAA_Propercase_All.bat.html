@echo off
setlocal disabledelayedexpansion

:: Using powershell -encodedCommand or passing via standard input prevents CMD from ruining the $_. variables
powershell -NoProfile -Command "& {Get-ChildItem -File | Rename-Item -NewName { (Get-Culture).TextInfo.ToTitleCase($_.BaseName.ToLower()) + $_.Extension }}"

echo File renaming completed!
pause
