@echo off
setlocal disabledelayedexpansion

:: Strips out all exclamation marks and converts the remaining text to Title Case
powershell -NoProfile -Command "& {Get-ChildItem -File | Rename-Item -NewName { (Get-Culture).TextInfo.ToTitleCase($_.BaseName.Replace('!', '').ToLower()) + $_.Extension }}"

echo File renaming completed!
