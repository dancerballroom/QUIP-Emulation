@echo off
Title Search Replace Batchfile
Setlocal enabledelayedexpansion
set /p SR_File="File or File type -> "

set /p Pattern="String to Replace: -> "
set /p Replace="New String:        -> "

Echo File Type         :  !SR_File!
Echo Pattern to Replace:  !Pattern!
Echo Replacement String:  !Replace!
Echo Press Enter to Commit
pause

For %%a in (!SR_File!) Do (
    Set "File=%%~a"
    Ren "%%a" "!File:%Pattern%=%Replace%!"
    echo "%%a" "!File:%Pattern%=%Replace%!"
)
pause
Exit 0
