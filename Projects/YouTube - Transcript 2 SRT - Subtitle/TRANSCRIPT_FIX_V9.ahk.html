TRANSCRIPT_FIX_V9.ahkk

Main Features:
1.  Gets YouTube Transcripts (Chrome: CNTRL+SHIFT+L)
2.  Turns Formatted Text into SRT (SubtitleEdit: CNTRL+SHIFT+M)
3.  Creates Empty Text File from Name Pasted in Windows Clipboard (Windows Explorer: CNTRL+SHIFT+O)
4.  Handles Formatting TimeSignature and Transcript before Number 2 above
    a. Yank Unformatted Time Signature to New.txt (CNTRL+SHIFT+E)
    b. Format Time Signature in New.txt (CNTRL+SHIFT+T)
    c. Remove Previous Time Signature from Transcript (CNTRL+SHIFT+I) if prepped 
    d. Merges Formatted Time Signature with Transcript (CNTRL+SHIFT+Y) CHAPTER Titles Removed
5.  Integrated Transient Toast Library for AHK from GitHub

Code:

ADD_NewText()
{
	;CNTRL + SHIFT + F10
	Send "+{F10}"
	;W = NEW
	Send "w"
	;T = TEXT FILE
	Send "t"

}

SELECT_TIMESIG()
{
	;Control + Shift + E
	;MARK and Copy TimeSig
	;^\d*:\d*:\d.|^\d*:\d.
	;CNTRL + M brings up MARK TAB
	;Type expression and Press Enter to MARK
	;Settings > Shortcut Mapper
	;CNTRL+SHIFT+C = COPY MARKED TEXT
	SendInput "^m"
	Sleep 500
	Send "{^}\d*:\d*:\d.{|}{^}\d*:\d."
	Sleep 500
	SendInput "{Enter}"
	Sleep 500
	; ALT g
	Send("!g")	
	Sleep 500
	; TAB 4
	Send "{TAB 4}"
	;Enter
	Sleep 500
	SendInput "{Enter}"
	Sleep 500
	;ESC
	Send "{ESC}"
	;CNTRL+N
	Send "^n"
	;CNTRL+V
	Send "^v"
	Toastify.Start("dark", Toastify.ALIGN.BOTTOM_RIGHT)
	Toastify.Show("NEXT: CNTRL+SHFT+T ", "Control+X = Exit", [ {text: "NotePad++", onClick: (*) => Run("NotePad++.exe")} ])

}

	; x:xx to 00:0x:00
	;     Replace: ^(\d:\d.) with 00:0\1

FIX_Transcript()
{
  ;Control + Shift + I
  ; Given Transcript has extra info: hours, minutes, seconds
  ; and Following has been processed
  ;       extracted/deleted: ^Chapter.*\r\n  
  ;       extracted: ^\d*:\d*:\d.|^\d*:\d.
  ;       deleted: ^\d*:\d*:\d*|^\d*:\d*
  ;       All Lines formatted 1Space + Transcript without TimeSigs
  ; This Macro Runs:  Delete Following
  ;                1  (^\shours, \d*)  
  ;                2  (^\shour, \d*)  
  ;                3  (^\sminutes, \d*)
  ;                4  (^\sminute, \d*)
  ;                5  (^\sseconds)
  ;                6  (^\ssecond)  
  ;                7  (^\sminutes)
  ;                8  (^\sminute)
  	SendInput "^h"
	Sleep 500
	
  ; Control PGDN (hours)
  Send "^{PGDN}"
  ; ALT A
  Send("!a")
  Sleep 500

  ; Control PGDN (hour)
  Send "^{PGDN}"
  ; ALT A
  Send("!a")
  Sleep 500
  
  ; Control PGDN (minutes)
  Send "^{PGDN}"
  ; ALT A
  Send("!a")
  Sleep 500

  ; Control PGDN (minute)
  Send "^{PGDN}"
  ; ALT A
  Send("!a")
  Sleep 500

  ; Control PGDN (seconds)
  Send "^{PGDN}"
  ; ALT A
  Send("!a")
  Sleep 500
  
  ; Control PGDN (seconds)
  Send "^{PGDN}"
  ; ALT A
  Send("!a")
  Sleep 500

  ; Control PGDN (minutes)
  Send "^{PGDN}"
  ; ALT A
  Send("!a")
  Sleep 500
  
  ; Control PGDN (minute)
  Send "^{PGDN}"

  ; ALT A
  Send("!a")
  Sleep 500

  SendInput "{ESC}"
  Sleep 500
  
  ; Control END
  Send "^{END}"
  Sleep 500
  
  ; Control PGDN
  Send "^{PGDN}"
  Toastify.Start("dark", Toastify.ALIGN.BOTTOM_RIGHT)
  Toastify.Show("NEXT: CNTRL+SHFT+Y", "Control+X = Exit", [ {text: "NotePad++", onClick: (*) => Run("NotePad++.exe")} ])
}

INTEGRATE_TIMESIG()
{
    ;Control + Shift + Y
	;Given Tab2, TimeSig Format: XX:XX:XX
	;Given Tab1, Transcript No TimeSig
	;Given Tab1, # of Tab2 Lines = # of Tab1 Lines	
	;Start Tab2: Select and Copy
	Sleep 500
    Send "^{HOME}"
	Sleep 500
	Send "^!{END}"
	Sleep 500
	Send "^!{RIGHT}"
	Sleep 500
	Send "^c"
	
	;Change to Tab1
	Sleep 500
	Send "^+{TAB}"	

	;Tab1 Insert Space All Lines
	Sleep 500
    Send "^{HOME}"
	Sleep 500
	Send "^!{END}"
	Sleep 500
	Send "{SPACE}"
	Sleep 500
	Send "{LEFT}"

	;Insert Tab2 TimeSig Into Tab1
	Sleep 500
    Send "^{HOME}"
	Sleep 500
	Send "^!{END}"
	Sleep 500
	Send "^v"
	Toastify.Start("dark", Toastify.ALIGN.BOTTOM_RIGHT)
	Toastify.Show("NEXT: Save CNTRL S Close CNTRL+SHFT+W and Exit ALT+F4", "Control+X = Exit", [ {text: "NotePad++", onClick: (*) => Run("NotePad++.exe")} ])
}
