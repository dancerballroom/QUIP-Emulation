REM # INPUT:  %1 (Time-Signature: 12:34:56,789)
REM # OUTPUT: MilliSec (MilliSeconds: 123456789)
:TimeSig2MilliSec
    Set TT=%1
    Set HH=!TT:~0,2!
    Set MM=!TT:~3,2!
    Set SS=!TT:~6,2!
    Set MS=!TT:~9,3!
    if !DEBUG! EQU !TRUE! Echo Time:     !TT!
    if !DEBUG! EQU !TRUE! Echo Hour:     !HH!
    if !DEBUG! EQU !TRUE! Echo Minute:   !MM!
    if !DEBUG! EQU !TRUE! Echo Second:   !SS!
    if !DEBUG! EQU !TRUE! Echo Mili-Sec: !MS!
    Set /A MS_SS=(!SS!*1000)
    Set /A MS_MM=(!MM!*60*1000)
    Set /A MS_HH=(!HH!*60*60*1000) 
    if !DEBUG! EQU !TRUE! Echo MS_SS:  !MS_SS!
    if !DEBUG! EQU !TRUE! Echo MS_MM:  !MS_MM!
    if !DEBUG! EQU !TRUE! Echo MS_MM:  !MS_HH!
    Set /A MS_TOT=!MS_HH!+!MS_MM!+!MS_SS!+!MS!
    if !DEBUG! EQU !TRUE! Echo MS_TOT: !MS_TOT!
    Set /A MilliSec=!MS_TOT!
Exit /b

REM # INPUT:  %1 (Milli-Seconds)
REM # OUTPUT: Timex (Time-Signature: 12:34:56,789)
:MilliSec2TimeSig
    Set /A MS_TOT=%1
    Set MS_EXT=!MS_TOT:~-3,3!
    if !DEBUG! EQU !TRUE! Echo Mili-Seconds Extracted: !MS_EXT!
    Set /A Rem_Minus_MS=!MS_TOT!-MS_EXT
    if !DEBUG! EQU !TRUE! Echo Total Minus MS: !Rem_Minus_MS!
    Set /A HH_EXT=(!Rem_Minus_MS!/(60*60*1000))
    if !DEBUG! EQU !TRUE! Echo Hours Extracted: !HH_EXT!
    Set /A Rem_Minus_HH=(!Rem_Minus_MS!%%(60*60*1000))
    if !DEBUG! EQU !TRUE! Echo Total Minus HH: !Rem_Minus_HH!
    Set /A MM_EXT=(!Rem_Minus_HH!/(60*1000))
    if !DEBUG! EQU !TRUE! Echo Minutes Extracted: !MM_EXT!
    Set /A Rem_Minus_MM=(!Rem_Minus_HH!%%(60*1000))
    if !DEBUG! EQU !TRUE! Echo Total Seconds SS: !Rem_Minus_MM!
    Set /A SS_EXT=(!Rem_Minus_MM!/1000)
    if !DEBUG! EQU !TRUE! Echo Seconds Extracted: !SS_EXT!

    if !HH_EXT! lss 10 set HH_EXT=0!HH_EXT!
    if !MM_EXT! lss 10 set MM_EXT=0!MM_EXT!
    if !SS_EXT! lss 10 set SS_EXT=0!SS_EXT!

    if !DEBUG! EQU !TRUE! Echo Time Extracted: !HH_EXT!:!MM_EXT!:!SS_EXT!,!MS_EXT!
    Set Timex=!HH_EXT!:!MM_EXT!:!SS_EXT!,!MS_EXT!
Exit /b

REM # Echo ********[ TEST 1 ]**********

REM # Function ADD_TIMES
REM # INPUT:  %1 %2 (Time-Signatures: 12:34:56,789)
REM # CALL :ADD_TIMES !TIME-VAR-1! !TIME-VAR-2! Total TimexC
REM # OUTPUT: Total (Time-Signature:  12:34:56,789)
:ADD_TIMES
    Set First_Time=%1
    Set Second_Time=%2
    if !DEBUG! EQU !TRUE! Echo First Time:  !First_Time!
    if !DEBUG! EQU !TRUE! Echo Second Time: !Second_Time!
    CALL :TimeSig2MilliSec !First_Time! MilliSec
    Set /A FT=!MilliSec!
    CALL :TimeSig2MilliSec !Second_Time! MilliSec
    Set /A ST=!MilliSec!
    Set /A FT_ST_TOT=!FT!+!ST!
    CALL :MilliSec2TimeSig !FT_ST_TOT! Timex TimexC
    if !DEBUG! EQU !TRUE! Echo Total Time: !Timex!
    Set Total=!Timex!
exit /b

REM # Function SUB_TIMES
REM # INPUT:  %1 %2 (Time-Signatures: 12:34:56,789)
REM # CALL :SUB_TIMES !TIME-VAR-1! !TIME-VAR-2! Total TimexC
REM # OUTPUT: Total (Time-Signature:  12:34:56,789)
:SUB_TIMES
    Set First_Time=%1
    Set Second_Time=%2
    if !DEBUG! EQU !TRUE! Echo First Time:  !First_Time!
    if !DEBUG! EQU !TRUE! Echo Second Time: !Second_Time!
    CALL :TimeSig2MilliSec !First_Time! MilliSec
    Set /A FT=!MilliSec!
    CALL :TimeSig2MilliSec !Second_Time! MilliSec
    Set /A ST=!MilliSec!
    Set /A FT_ST_TOT=!FT!-!ST!
    CALL :MilliSec2TimeSig !FT_ST_TOT! Timex TimexC
    if !DEBUG! EQU !TRUE! Echo Total Time: !Timex!
    Set Total=!Timex!
exit /b

REM # Function TimeSig2MilliSec
REM # INPUT:  %1 (Time-Signature: 12:34:56,789)
REM #         CALL :TimeSig2MilliSec !12.34.56.789! MilliSec
REM # OUTPUT: MilliSec (MilliSeconds: 123456789)
:TimeSig2MilliSec
    Set TT=%1
    Set HH=!TT:~0,2!
    Set MM=!TT:~3,2!
    Set SS=!TT:~6,2!
    Set MS=!TT:~9,3!
    if !DEBUG! EQU !TRUE! Echo Time:     !TT!
    if !DEBUG! EQU !TRUE! Echo Hour:     !HH!
    if !DEBUG! EQU !TRUE! Echo Minute:   !MM!
    if !DEBUG! EQU !TRUE! Echo Second:   !SS!
    if !DEBUG! EQU !TRUE! Echo Mili-Sec: !MS!
    
    Set /A MS_MS=((1!MS! %% 100))
    Set /A MS_SS=((1!SS! %% 100)*1000)
    Set /A MS_MM=((1!MM! %% 100)*60*1000)
    Set /A MS_HH=((1!HH! %% 100)*60*60*1000)
    if !DEBUG! EQU !TRUE! Echo MS_SS:  !MS_SS!
    if !DEBUG! EQU !TRUE! Echo MS_MM:  !MS_MM!
    if !DEBUG! EQU !TRUE! Echo MS_MM:  !MS_HH!
    Set /A MS_TOT=!MS_HH!+!MS_MM!+!MS_SS!+!MS_MS!
    if !DEBUG! EQU !TRUE! Echo MS_TOT: !MS_TOT!
    Set /A MilliSec=!MS_TOT!
    REM Echo MilliSec: !MilliSec!
Exit /b

REM # Function MilliSec2TimeSig
REM # INPUT:  %1 (Milli-Seconds)
REM #          CALL :MilliSec2TimeSig !Milli-Seconds! Timex TimexC
REM # OUTPUT: Timex (Time-Signature: 12:34:56.789)
REM # OUTPUT: TimexC (Time-Signature: 12:34:56,789)
:MilliSec2TimeSig
    Set /A MS_TOT=%1
    Set MS_EXT=!MS_TOT:~-3,3!
    if !DEBUG! EQU !TRUE! Echo Mili-Seconds Extracted: !MS_EXT!
    Set /A Rem_Minus_MS=!MS_TOT!-MS_EXT
    if !DEBUG! EQU !TRUE! Echo Total Minus MS: !Rem_Minus_MS!
    Set /A HH_EXT=(!Rem_Minus_MS!/(60*60*1000))
    if !DEBUG! EQU !TRUE! Echo Hours Extracted: !HH_EXT!
    Set /A Rem_Minus_HH=(!Rem_Minus_MS!%%(60*60*1000))
    if !DEBUG! EQU !TRUE! Echo Total Minus HH: !Rem_Minus_HH!
    Set /A MM_EXT=(!Rem_Minus_HH!/(60*1000))
    if !DEBUG! EQU !TRUE! Echo Minutes Extracted: !MM_EXT!
    Set /A Rem_Minus_MM=(!Rem_Minus_HH!%%(60*1000))
    if !DEBUG! EQU !TRUE! Echo Total Seconds SS: !Rem_Minus_MM!
    Set /A SS_EXT=(!Rem_Minus_MM!/1000)
    if !DEBUG! EQU !TRUE! Echo Seconds Extracted: !SS_EXT!

    if !HH_EXT! lss 10 set HH_EXT=0!HH_EXT!
    if !MM_EXT! lss 10 set MM_EXT=0!MM_EXT!
    if !SS_EXT! lss 10 set SS_EXT=0!SS_EXT!

    if !DEBUG! EQU !TRUE! Echo Time Extracted: !HH_EXT!:!MM_EXT!:!SS_EXT!,!MS_EXT!
    Set Timex=!HH_EXT!:!MM_EXT!:!SS_EXT!.!MS_EXT!
    Set TimexC=!HH_EXT!:!MM_EXT!:!SS_EXT!,!MS_EXT!
Exit /b
