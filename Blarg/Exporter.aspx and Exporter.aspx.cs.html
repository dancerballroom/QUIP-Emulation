using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using System.IO.Compression; // Required for ZipFile
using System.Text;
using System.Web;

public partial class Exporter : System.Web.UI.Page
{
    private class SearchItem
    {
        public string Name { get; set; }
        public string RelativePath { get; set; }
    }

    private List<SearchItem> _searchIndex = new List<SearchItem>();
    private string _rootPath = "";

    protected void Page_Load(object sender, EventArgs e)
    {
        // 1. Define source and a unique temporary working directory
        string sourcePath = Server.MapPath("~/uploads/notes");
        string tempId = Guid.NewGuid().ToString();
        _rootPath = Path.Combine(Path.GetTempPath(), "Export_" + tempId);
        string zipPath = _rootPath + ".zip";

        if (!Directory.Exists(sourcePath))
        {
            Response.Write("Error: Source directory does not exist.");
            return;
        }

        try
        {
            // 2. Copy all original files/folders to our temporary workspace
            CopyDirectory(sourcePath, _rootPath);

            // 3. Run the generator inside the TEMP directory
            GenerateIndices(_rootPath, true);
            GenerateSearchIndexFile(_rootPath);
            GenerateSearchHtmlPage(_rootPath);

            // 4. Compress the temporary directory into a ZIP archive
            ZipFile.CreateFromDirectory(_rootPath, zipPath);

            // 5. Stream the ZIP file directly to the browser
            StreamZipToBrowser(zipPath, "Website_Export.zip");
        }
        catch (Exception ex)
        {
            Response.Write("An error occurred during export: " + ex.Message);
        }
        finally
        {
            // 6. Clean up: Delete temporary files/directories from the server
            if (Directory.Exists(_rootPath))
            {
                Directory.Delete(_rootPath, true);
            }
            if (File.Exists(zipPath))
            {
                File.Delete(zipPath);
            }
        }
    }

    private void StreamZipToBrowser(string zipFilePath, string downloadFileName)
    {
        Response.Clear();
        Response.ContentType = "application/zip";
        Response.AddHeader("Content-Disposition", $"attachment; filename={downloadFileName}");
        
        // Read the file and push it into the response stream
        Response.TransmitFile(zipFilePath);
        
        // Flush and end response to prevent any ASPX markup from corrupting the ZIP
        Response.Flush();
        Response.End();
    }

    // Helper method to copy directories recursively
    private static void CopyDirectory(string sourceDir, string destinationDir)
    {
        DirectoryInfo dir = new DirectoryInfo(sourceDir);
        DirectoryInfo[] dirs = dir.GetDirectories();

        Directory.CreateDirectory(destinationDir);

        foreach (FileInfo file in dir.GetFiles())
        {
            string targetFilePath = Path.Combine(destinationDir, file.Name);
            file.CopyTo(targetFilePath, true);
        }

        foreach (DirectoryInfo subDir in dirs)
        {
            string targetDirPath = Path.Combine(destinationDir, subDir.Name);
            CopyDirectory(subDir.FullName, targetDirPath);
        }
    }

    private void GenerateIndices(string currentDir, bool isRoot)
    {
        // Gather all subdirectories and files in the current folder
        string[] subDirs = Directory.GetDirectories(currentDir);
        string[] files = Directory.GetFiles(currentDir);

        StringBuilder html = new StringBuilder();
        
        // Build a clean, modern, style-agnostic HTML structure
        html.AppendLine("<!DOCTYPE html>");
        html.AppendLine("<html>");
        html.AppendLine("<head>");
        html.AppendLine("    <meta charset='utf-8'>");
        html.AppendLine($"    <title>Index of {GetRelativeUrl(currentDir)}</title>");
        html.AppendLine("    <style>body{font-family:sans-serif;padding:20px;line-height:1.6;} ul{list-style-type:none;padding-left:0;} li{margin:8px 0;} a{text-decoration:none;color:#0066cc;} a:hover{text-decoration:underline;}</style>");
        html.AppendLine("</head>");
        html.AppendLine("<body>");
        
        html.AppendLine($"    <h2>Directory Listing for: {GetRelativeUrl(currentDir)}</h2>");
        html.AppendLine("    <hr>");
        html.AppendLine("    <ul>");

        // 1. Add "Up to Parent" link if we aren't at the root
        if (!isRoot)
        {
            html.AppendLine("        <li>📁 <a href='../'><strong>[..] Up to Parent Directory</strong></a></li>");
        }
        else
        {
            // If it is the root, provide a prominent link to the global search page
            html.AppendLine("        <li>🔍 <a href='./search.html' style='color:#d9534f;'><strong>Search Entire Directory</strong></a></li>");
            html.AppendLine("        <br>");
        }

        // 2. Append Subdirectories
        foreach (string dir in subDirs)
        {
            DirectoryInfo di = new DirectoryInfo(dir);
            // Append a trailing slash for web directory behavior
            html.AppendLine($"        <li>📁 <a href='{di.Name}/'>{di.Name}/</a></li>");

            // RECURSION: Dive into this subdirectory
            GenerateIndices(dir, false);
        }

        // 3. Append Files
        foreach (string file in files)
        {
            FileInfo fi = new FileInfo(file);
            
            // Skip index.html and search components so they don't list themselves
            if (fi.Name.Equals("index.html", StringComparison.OrdinalIgnoreCase) || 
                fi.Name.Equals("search.html", StringComparison.OrdinalIgnoreCase) || 
                fi.Name.Equals("search-index.js", StringComparison.OrdinalIgnoreCase))
            {
                continue;
            }

            html.AppendLine($"        <li>📄 <a href='{fi.Name}' target='_blank'>{fi.Name}</a></li>");

            // Add this file to our global search array tracking
            _searchIndex.Add(new SearchItem
            {
                Name = fi.Name,
                RelativePath = GetRelativeUrl(file)
            });
        }

        html.AppendLine("    </ul>");
        html.AppendLine("</body>");
        html.AppendLine("</html>");

        // Save the index.html file directly into the current directory
        string indexPath = Path.Combine(currentDir, "index.html");
        File.WriteAllText(indexPath, html.ToString(), Encoding.UTF8);
    }

    // Helper method to turn absolute server paths into relative web paths
    private string GetRelativeUrl(string fullPath)
    {
        if (fullPath.Equals(_rootPath, StringComparison.OrdinalIgnoreCase))
        {
            return "/";
        }

        // Strip out the root path to get the relative piece
        string relative = fullPath.Substring(_rootPath.Length).Replace('\\', '/');
        return relative.StartsWith("/") ? relative : "/" + relative;
    }

    // Generates the JSON-like JS file that the client-side search engine reads
    private void GenerateSearchIndexFile(string rootPath)
    {
        StringBuilder js = new StringBuilder();
        js.AppendLine("// Auto-generated search database");
        js.AppendLine("const searchIndex = [");

        foreach (var item in _searchIndex)
        {
            // Clean strings for safe JS deployment
            string safeName = item.Name.Replace("'", "\\'");
            string safePath = item.RelativePath.Replace("'", "\\'");
            
            // Format as a simple JavaScript Object literal
            js.AppendLine($"    {{ name: '{safeName}', path: '.{safePath}' }},");
        }

        js.AppendLine("];");

        string jsPath = Path.Combine(rootPath, "search-index.js");
        File.WriteAllText(jsPath, js.ToString(), Encoding.UTF8);
    }

    // Quick boilerplate generator for search.html
    private void GenerateSearchHtmlPage(string rootPath)
    {
        string searchHtmlPath = Path.Combine(rootPath, "search.html");
        if (File.Exists(searchHtmlPath)) return; // Don't overwrite if you custom designed it

        string html = @"<!DOCTYPE html>
<html>
<head>
    <meta charset='utf-8'>
    <title>Search Site Files</title>
    <style>
        body { font-family: sans-serif; padding: 20px; max-width: 600px; margin: 0 auto; }
        input { width: 100%; padding: 10px; font-size: 16px; box-sizing: border-box; margin-bottom: 20px; }
        ul { list-style: none; padding: 0; }
        li { margin: 10px 0; padding: 10px; border-bottom: 1px solid #eee; }
        a { text-decoration: none; color: #0066cc; font-weight: bold; }
        .path { font-size: 12px; color: #666; margin-top: 4px; }
    </style>
</head>
<body>
    <h2>🔍 Search Directory Files</h2>
    <a href='./index.html'>⬅️ Back to Home</a>
    <br><br>
    <input type='text' id='searchBox' placeholder='Type to search files...' onkeyup='doSearch()'>
    <ul id='results'></ul>

    <script src='./search-index.js'></script>
    <script>
        function doSearch() {
            const query = document.getElementById('searchBox').value.toLowerCase();
            const resultsUl = document.getElementById('results');
            resultsUl.innerHTML = '';
            
            if(query.length < 2) return; // Only search for 2+ characters

            const filtered = searchIndex.filter(item => item.name.toLowerCase().includes(query));

            filtered.forEach(item => {
                const li = document.createElement('li');
                li.innerHTML = `📄 <a href='${item.path}' target='_blank'>${item.name}</a><div class='path'>Location: ${item.path}</div>`;
                resultsUl.appendChild(li);
            });

            if(filtered.length === 0) {
                resultsUl.innerHTML = '<li>No matching files found.</li>';
            }
        }
    </script>
</body>
</html>";

        File.WriteAllText(searchHtmlPath, html, Encoding.UTF8);
    }
