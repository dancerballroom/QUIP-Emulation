Copy#Requires AutoHotkey v2.0 #SingleInstance Force ; Press Ctrl + Shift + S to trigger the script when Chrome is active ^+s:: {     if !WinActive("ahk_exe chrome.exe") {         MsgBox("Please make sure Google Chrome is active before running this script.")         Return     }

    ; --- CONFIGURATION ---     ; Change this path to your preferred folder directory     TargetDir :=
A_Desktop "\LitRPG Recommendations\_Manhwa_Story_URL\_Saved_Chrome_Tabs"      ; ---------------------     ; Create folder if it doesn't exist     if !DirExist(TargetDir)         DirCreate(TargetDir)          CapturedTabs := Map()     DuplicateCount := 0     MaxDuplicatesAllowed := 3 ; Security brake to prevent infinite looping     MsgBox("Capturing tabs... Please do not touch your keyboard or mouse.", "Running", "Iconi T2")     Loop {         ; --- THE FIX: Clear clipboard right before copying ---         A_Clipboard := ""                   ; 1. Highlight and copy the current tab's URL         Send("^l") ; Focus address bar         Sleep(50)         Send("^c") ; Copy URL                  if !ClipWait(1) {             MsgBox("Failed to copy URL from address bar.")             break         }                  CurrentURL := A_Clipboard         CurrentTitle := WinGetTitle("A")         ; Clean " - Google Chrome" from the window title suffix         CleanTitle := RegExReplace(CurrentTitle, " - Google Chrome$")                  ; Remove forbidden Windows file name characters: \ / : * ? " < > |         CleanTitle := RegExReplace(CleanTitle, '[\\/:*?"<>|]', "_")                  ; Limit file name length to prevent Windows filesystem crashes         CleanTitle := SubStr(CleanTitle, 1, 100)         ; 2. Check if we have already captured this exact URL         if CapturedTabs.Has(CurrentURL) {             DuplicateCount++             ; If we hit the same URL multiple times, we've cycled through all open tabs             if (DuplicateCount >= MaxDuplicatesAllowed)                 break         } else {             ; Track the URL and map it to its clean title             CapturedTabs[CurrentURL] := CleanTitle             DuplicateCount := 0 ; Reset consecutive duplicate counter         }         ; 3. Move to the next tab         Send("^{Tab}")         Sleep(250) ; Increased slightly to ensure Chrome updates the window title in time     }     ; 4. Create the Windows URL shortcut files (.url)     SuccessCount := 0     for URL, Title in CapturedTabs {         FilePath := TargetDir "\" Title ".url"                  ; Resolve potential title collisions (different URLs with the exact same title)         Loop {             if !FileExist(FilePath)                 break             FilePath := TargetDir "\" Title " (" A_Index ").url"         }         try {             IniWrite(URL, FilePath, "InternetShortcut", "URL")             SuccessCount++         }         catch {             ; Skip individual write errors gracefully             continue         }     }     MsgBox("Process Finished!`nCaptured " SuccessCount " unique tabs.`nSaved to: " TargetDir, "Success", "Iconi") }
