<%@ Page Language="C#" %>
<%@ Import Namespace="System.IO" %>


<script runat="server">
    protected void Page_Load(object sender, EventArgs e)
    {
        string query = Request.QueryString["q"];
        
        // Display the query exactly as the user typed it in the UI header
        litQuery.Text = HttpUtility.HtmlEncode(query);


        if (!string.IsNullOrEmpty(query))
        {
            string rootPath = Server.MapPath("~/uploads/notes/");
            
            if (!Directory.Exists(rootPath)) return;


            // Replace spaces with underscores to match the FileBrowser saving convention
            string sanitizedQuery = query.Replace(" ", "_");
            string searchPattern = "*" + sanitizedQuery + "*";
            
            // 1. Gather both files and folders matching the underscored pattern
            string[] foundFiles = Directory.GetFiles(rootPath, searchPattern, SearchOption.AllDirectories);
            string[] foundDirs = Directory.GetDirectories(rootPath, searchPattern, SearchOption.AllDirectories);
            
            // 2. Map files to uniform structures
            var filesList = foundFiles.Select(f => {
                string name = Path.GetFileName(f);
                string relFolder = GetRelativeFolder(f);
                return new {
                    Name = name,
                    RelativeFolder = relFolder,
                    IsDirectory = false,
                    SearchLink = "FilingCabinet.aspx?id=" + HttpUtility.UrlEncode(Path.GetFileNameWithoutExtension(name)) + "&folder=" + HttpUtility.UrlEncode(relFolder)
                };
            });


            // 3. Map directories to structures
            var dirsList = foundDirs.Select(d => {
                string name = Path.GetFileName(d);
                string relFolder = GetRelativeFolder(d);
                string fullFolderPath = string.IsNullOrEmpty(relFolder) ? name : relFolder + "/" + name;
                
                return new {
                    Name = name,
                    RelativeFolder = relFolder,
                    IsDirectory = true,
                    SearchLink = "FileBrowser.aspx?path=" + HttpUtility.UrlEncode(fullFolderPath)
                };
            });


            // 4. Combine collections (Folders first, then files)
            var combinedResults = dirsList.Concat(filesList).ToList();
            
            // 5. Bind data
            rptResults.DataSource = combinedResults;
            rptResults.DataBind();
            
            pnlNoResults.Visible = (combinedResults.Count == 0);
        }
        else
        {
            pnlNoResults.Visible = true;
        }
    }


    protected string GetRelativeFolder(string fullPath)
    {
        string rootPath = Server.MapPath("~/uploads/notes/");
        string dir = Path.GetDirectoryName(fullPath);
        if (dir == null) return "";
        return dir.Replace(rootPath, "").Replace("\\", "/").TrimStart('/');
    }
</script>
