using System;
using System.Web;

public static class AppState
{
    public static string CurrentFolder
    {
        get {

            return (sessionValue != null) ? sessionValue.ToString() : ""; 
        }
        set { HttpContext.Current.Session["CurrentFolder"] = value; }
