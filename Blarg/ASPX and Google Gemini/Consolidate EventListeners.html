Consolidate EventListeners

Why you need them:

  • keydown: Essential for your keyboard shortcuts (Ctrl+S, etc.) to work inside the editor.

  • input: Essential for updateLineNumbers and the isFileDirty (dirty tracking) system to function correctly.

  • scroll: Essential for keeping your line numbers aligned with your text as you scroll.

  • contextmenu: Essential for your custom right-click menu to appear.


Quick Tips:

  • Delete Duplicates
CODE:
// Keep these inside your main script block
window.addEventListener("DOMContentLoaded", () => {
    // 1. Keep the Keydown Interceptor (This handles your shortcuts)
    editorBox.addEventListener("keydown", function (e) {
        // ... (your existing shortcut logic for Ctrl+S, Ctrl+H, Tab, etc.)
    });

    // 2. Keep the Input/Update listeners
    editorBox.addEventListener("input", function() {
        setDirty();
        updateLineNumbers();
    });

    // 3. Keep the Scroll listener
    editorBox.addEventListener("scroll", () => {
        document.getElementById("lineNumbers").scrollTop = editorBox.scrollTop;
    });

    // 4. Keep the Context Menu listener
    editorBox.addEventListener("contextmenu", function (e) {
        e.preventDefault(); 
        openContextMenuAt(e.pageX, e.pageY, e.clientX, e.clientY);
    });

    // 5. Keep the MouseDown listener (for your block selection)
    editorBox.addEventListener("mousedown", (e) => {
        if (isShiftAltDown) {
            // ... (your selection logic)
        }
    });
});

The Consolidated Logic:

Replace your existing script blocks with this unified structure. This includes the centralized Escape and Click listeners 
that handle your modal, find/replace tool, and menus.

The Code:

<script type="text/javascript">
    // --- Global State ---
    var rightClickSavedRange = null, isFileDirty = false, isEditorLocked = false;
    var speechRecognitionEngine = null, isCurrentlyRecordingSpeech = false;
    var isShiftAltDown = false, isBlockSelecting = false;
    var startX, startY;

    // --- DOM Elements (Loaded via DOMContentLoaded to be safe) ---
    let editorBox, customMenu, shortcutsModal, findReplaceTool, unsavedModal;

    window.addEventListener("DOMContentLoaded", () => {
        editorBox = document.getElementById("editorBox");
        customMenu = document.getElementById("customContextMenu");
        shortcutsModal = document.getElementById("shortcutsModal");
        findReplaceTool = document.getElementById("findReplaceTool");
        unsavedModal = document.getElementById("unsavedChangesModal");

        // --- Centralized UI Event Handlers ---
        document.addEventListener("click", function(e) {
            // 1. Close menus if clicking outside
            if (!e.target.closest('.dropdown') && !e.target.closest('#customContextMenu')) {
                closeAllDropdownBarMenus();
                customMenu.style.display = "none";
            }
            // 2. Close Modal if clicking backdrop
            if (e.target === shortcutsModal) shortcutsModal.style.display = 'none';
        });

        document.addEventListener("keydown", function(e) {
            if (e.key === "Escape") {
                // Close everything on ESC
                if (shortcutsModal.style.display === "flex") shortcutsModal.style.display = "none";
                if (findReplaceTool && findReplaceTool.style.display === "block") findReplaceTool.style.display = "none";
                if (unsavedModal) unsavedModal.style.display = "none";
                closeAllDropdownBarMenus();
                customMenu.style.display = "none";
            }
        });
    });

    // --- Menu & Modal Controllers ---
    function closeAllDropdownBarMenus() {
        document.querySelectorAll(".dropdown-menu").forEach(m => m.style.display = "none");
        document.querySelectorAll(".menu-trigger").forEach(b => b.classList.remove("active"));
    }

    function toggleFindReplace() {
        if (!findReplaceTool) return;
        const isHidden = findReplaceTool.style.display === 'none' || findReplaceTool.style.display === '';
        findReplaceTool.style.display = isHidden ? 'block' : 'none';
        closeAllDropdownBarMenus();
    }

    // --- Editor Logic (Keep your existing setDirty, updateLineNumbers, etc. here) ---
    function setDirty() { if (!isEditorLocked) isFileDirty = true; }

    // --- Copy-Paste your existing functions below this point ---
    // (insertHyperlink, toggleSpeechToText, updateLineNumbers, etc.)
    // Note: ensure no duplicate 'toggleFindReplace' or 'closeAllDropdownBarMenus' exist.
</script>
